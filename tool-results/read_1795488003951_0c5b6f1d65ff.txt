    86→function NumField({
    87→  label,
    88→  value,
    89→  onCommit,
    90→  min,
    91→  max,
    92→  disabled,
    93→}: {
    94→  label: string;
    95→  value: number;
    96→  onCommit: (v: number) => void;
    97→  min: number;
    98→  max: number;
    99→  disabled?: boolean;
   100→}) {
   101→  const fmt = (v: number) => String(Math.round(v * 10) / 10);
   102→  const commit = (raw: string) => {
   103→    const parsed = Number(raw.replace(",", "."));
   104→    if (!Number.isFinite(parsed)) return;
   105→    onCommit(Math.min(Math.max(parsed, min), max));
   106→  };
   107→  // Uncontrolled input keyed by the external value: dragging / resizing on the
   108→  // canvas remounts it with fresh numbers, while local typing stays untouched.
   109→  return (
   110→    <label className="flex min-w-0 flex-1 flex-col gap-1">
   111→      <span className="font-mono text-[9.5px] uppercase tracking-[0.06em] text-[#6B7280]">
   112→        {label}
   113→      </span>
   114→      <input
   115→        key={`${label}-${fmt(value)}`}
   116→        type="number"
   117→        inputMode="decimal"
   118→        step="0.5"
   119→        defaultValue={fmt(value)}
   120→        disabled={disabled}
   121→        onBlur={(e) => commit(e.target.value)}
   122→        onKeyDown={(e) => {
   123→          if (e.key === "Enter") {
   124→            e.preventDefault();
   125→            (e.target as HTMLInputElement).blur();
   126→          }
   127→        }}
   128→        aria-label={label}
   129→        className="h-8 w-full rounded-md border border-[#E5E7EB] bg-white px-2 font-mono text-[11.5px] tabular-nums text-[#111827] outline-none transition-colors focus:border-[#003399] disabled:opacity-40"
   130→      />
   131→    </label>
   132→  );
   133→}

   251→export function PhotoInspector({
   252→  placements,
   253→  selected,
   254→  selectedIds,
   255→  files,
   256→  settings,
   257→  preset,
   258→  paperWmm,
   259→  paperHmm,
   260→  beyondMargin,
   261→  capacity,

   263→  onBorderless,
   264→  onPreset,

   269→  onArrange,
   270→  onDuplicate,
   271→  onRemove,
   272→  onSelect,
   273→  onSelectAll,
   274→  onClearSelection,
   275→  onClear,
   276→  onAddImages,
   277→}: PhotoInspectorProps) {
   278→  const selectedCount = selectedIds.length;
   279→  // "Jumlah Foto" lives in the shared photo settings so this control, photo
   280→  // uploads and preset changes all drive the same auto-grid count.
   281→  const capSafe = Math.max(1, Math.min(capacity, 200));
   282→  const n = Math.max(1, Math.min(settings.perSheet, capSafe));
   283→  /** Re-tile immediately — bundling is automatic for every sheet control. */
   284→  const setCount = (next: number) => {
   285→    const v = Math.max(1, Math.min(next, capSafe));
   286→    onSetting({ perSheet: v });
   287→    onArrangeCount(v);
   288→  };
   289→  const setAlignV = (v: GridAlignV) => {
   290→    onSetting({ alignV: v });
   291→    onArrangeCount(n, { alignV: v });

   294→    onSetting({ alignH: v });
   295→    onArrangeCount(n, { alignH: v });
   296→  };
   297→  const setMargin = (v: number) => {
   298→    onSetting({ marginMm: v });
   299→    onArrangeCount(n, { marginMm: v });
   300→  };

   302→    onSetting({ gapMm: v });
   303→    onArrangeCount(n, { gapMm: v });
   304→  };

   306→  const unplacedImages = files.filter(
   307→    (f) => f.kind === "image" && f.status === "ready" && !!f.url && !placedIds.has(f.id),
   308→  );
   309→
   310→  const setAdjust = (patch: Partial<PhotoAdjustments>) =>
   311→    onSetting({ adjust: { ...settings.adjust, ...patch } });
   312→  const pctFmt = (v: number) => `${Math.round(v * 100)}%`;
   313→
   314→  const scopeLabel = selected ? "menerapkan ke foto terpilih" : "menerapkan ke semua foto";

   328→            <SelectValue />

   342→                    {p.label}

   344→                ))}
   345→              </SelectGroup>
   346→            ))}

   348→        </Select>

   350→          {PRESET_CHIPS.map((c) => {

   364→                {c.label}

   370→          {presetDimsLabel(preset)} • {scopeLabel}
   371→        </p>
   372→

   374→          <div className="flex gap-2">
   375→            <NumField
   376→              label="Lebar (mm)"
   377→              value={settings.customWMm}
   378→              min={10}
   379→              max={paperWmm}
   380→              onCommit={(v) => onPreset("custom", { wMm: v, hMm: settings.customHMm })}

   382→            <NumField
   383→              label="Tinggi (mm)"
   384→              value={settings.customHMm}
   385→              min={10}
   386→              max={paperHmm}
   387→              onCommit={(v) => onPreset("custom", { wMm: settings.customWMm, hMm: v })}

   396→            aria-label="Kunci rasio aspek foto"
   397→            className="data-[state=checked]:bg-[#003399]"
   398→          />
   399→        </Row>
   400→      </Section>
   401→
   402→      {/* Posisi & Ukuran (Word-like) */}
   403→      <Section
   404→        title={
   405→          selectedCount > 1
   406→            ? `Posisi & Ukuran — ${selectedCount} Foto Dipilih`
   407→            : "Posisi & Ukuran Foto"
   408→        }
   409→        className={selected ? "" : "opacity-60"}
   410→      >

   414→              <NumField
   415→                label="Posisi X"
   416→                value={selected.xMm}
   417→                min={0}
   418→                max={paperWmm}
   419→                onCommit={(v) => onUpdateSelected({ xMm: v })}

   422→                label="Posisi Y"
   423→                value={selected.yMm}
   424→                min={0}
   425→                max={paperHmm}
   426→                onCommit={(v) => onUpdateSelected({ yMm: v })}

   428→            </div>
   429→            <div className="flex gap-2">

   431→                label="Lebar"
   432→                value={selected.wMm}

   445→              />
   446→              <NumField
   447→                label="Tinggi"
   448→                value={selected.hMm}
   449→                min={10}
   450→                max={paperHmm}
   451→                onCommit={(v) =>
   452→                  onUpdateSelected(
   453→                    settings.lockAspect
   454→                      ? {
   455→                          hMm: v,
   456→                          wMm: (v * selected.wMm) / selected.hMm,

   459→                  )

   461→              />
   462→            </div>
   463→            <Row label="Rotasi">
