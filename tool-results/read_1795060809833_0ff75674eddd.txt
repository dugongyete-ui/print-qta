    91→    91→    91→function placementBBox(p: {
    92→    92→    92→  xMm: number;
    93→    93→    93→  yMm: number;
    94→    94→    94→  wMm: number;
    95→    95→    95→  hMm: number;
    96→    96→    96→  rotation: number;
    97→    97→    97→}): { x: number; y: number; w: number; h: number } {
    98→    98→    98→  const swapped = swappedBBox(p.rotation);
    99→    99→    99→  const w = swapped ? p.hMm : p.wMm;
   100→   100→   100→  const h = swapped ? p.wMm : p.hMm;
   101→   101→   101→  const cx = p.xMm + p.wMm / 2;
   102→   102→   102→  const cy = p.yMm + p.hMm / 2;
   103→   103→   103→  return { x: cx - w / 2, y: cy - h / 2, w, h };
   104→   104→   104→}

   107→   107→   107→function clampPlacement(
   108→   108→   108→  p: PhotoPlacement,
   109→   109→   109→  paperWmm: number,
   110→   110→   110→  paperHmm: number,
   111→   111→   111→): PhotoPlacement {
   112→   112→   112→  const bb = placementBBox(p);
   113→   113→   113→  const cx = p.xMm + p.wMm / 2;
   114→   114→   114→  const cy = p.yMm + p.hMm / 2;
   115→   115→   115→  const ncx = Math.min(
   116→   116→   116→    Math.max(cx, bb.w / 2),
   117→   117→   117→    Math.max(bb.w / 2, paperWmm - bb.w / 2),
   118→   118→   118→  );
   119→   119→   119→  const ncy = Math.min(
   120→   120→   120→    Math.max(cy, bb.h / 2),
   121→   121→   121→    Math.max(bb.h / 2, paperHmm - bb.h / 2),
   122→   122→   122→  );
   123→   123→   123→  return { ...p, xMm: ncx - p.wMm / 2, yMm: ncy - p.hMm / 2 };
   124→   124→   124→}

   126→   126→   126→function toBase64(bytes: Uint8Array): string {
   127→   127→   127→  let binary = "";
   128→   128→   128→  const chunk = 0x8000;
   129→   129→   129→  for (let i = 0; i < bytes.length; i += chunk) {
   130→   130→   130→    binary += String.fromCharCode(
   131→   131→   131→      ...Array.from(bytes.subarray(i, i + chunk)),
   132→   132→   132→    );
   133→   133→   133→  }
   134→   134→   134→  return btoa(binary);
   135→   135→   135→}

   212→   212→   212→async function renderDocx(file: LoadedFile): Promise<DocxRender> {
   213→   213→   213→  const cached = docxRenderCache.get(file.id);
   214→   214→   214→  if (cached) return cached;

   220→   220→   220→    document.body.appendChild(host);

   223→   223→   223→        new Blob([file.bytes!.slice().buffer as ArrayBuffer], { type: file.mime }),
   224→   224→   224→        host,

   226→   226→   226→        {

   234→   234→   234→          ignoreHeight: false,

   238→   238→   238→      const sections = wrapper
   239→   239→   239→        ? Array.from(wrapper.querySelectorAll("section.docx"))
   240→   240→   240→        : Array.from(host.querySelectorAll("section.docx"));
   241→   241→   241→      if (sections.length === 0) throw new Error("no-sections");
   242→   242→   242→      const first = sections[0] as HTMLElement;
   243→   243→   243→      const wpx = first.offsetWidth || Math.round(210 * (96 / 25.4));
   244→   244→   244→      const hpx = first.offsetHeight || Math.round(297 * (96 / 25.4));
   245→   245→   245→      return {
   246→   246→   246→        sectionsHtml: sections.map((s) => (s as HTMLElement).outerHTML),
   247→   247→   247→        pageWpx: wpx,
   248→   248→   248→        pageHpx: hpx,
   249→   249→   249→        pageWmm: wpx * PX96_TO_MM,
   250→   250→   250→        pageHmm: hpx * PX96_TO_MM,
   251→   251→   251→        pageCount: sections.length,
   252→   252→   252→      } satisfies DocxRender;
   253→   253→   253→    } finally {
   254→   254→   254→      host.remove();
   255→   255→   255→    }
   256→   256→   256→  })();
   257→   257→   257→  docxRenderCache.set(file.id, task);
   258→   258→   258→  return task;
   259→   259→   259→}
