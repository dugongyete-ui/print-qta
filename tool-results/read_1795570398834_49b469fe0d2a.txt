    83→function normRotation(rot: number): number {
    84→  const r = Math.round(rot / 90) * 90;
    85→  return ((r % 360) + 360) % 360;
    86→}
    87→
    88→function swappedBBox(rotation: number): boolean {
    89→  const r = normRotation(rotation);
    90→  return r === 90 || r === 270;
    91→}

    93→/** Visual bounding box (mm) of a placement, accounting for rotation. */
    94→function placementBBox(p: {
    95→  xMm: number;
    96→  yMm: number;
    97→  wMm: number;
    98→  hMm: number;
    99→  rotation: number;
   100→}): { x: number; y: number; w: number; h: number } {
   101→  const swapped = swappedBBox(p.rotation);
   102→  const w = swapped ? p.hMm : p.wMm;
   103→  const h = swapped ? p.wMm : p.hMm;
   104→  const cx = p.xMm + p.wMm / 2;
   105→  const cy = p.yMm + p.hMm / 2;
   106→  return { x: cx - w / 2, y: cy - h / 2, w, h };
   107→}
   108→
   109→/** Keep the visual bbox fully on the paper by clamping the rotation center. */
   110→function clampPlacement(
   111→  p: PhotoPlacement,
   112→  paperWmm: number,
   113→  paperHmm: number,
   114→): PhotoPlacement {
   115→  const bb = placementBBox(p);
   116→  const cx = p.xMm + p.wMm / 2;
   117→  const cy = p.yMm + p.hMm / 2;
   118→  const ncx = Math.min(
   119→    Math.max(cx, bb.w / 2),
   120→    Math.max(bb.w / 2, paperWmm - bb.w / 2),
   121→  );
   122→  const ncy = Math.min(
   123→    Math.max(cy, bb.h / 2),
   124→    Math.max(bb.h / 2, paperHmm - bb.h / 2),
   125→  );
   126→  return { ...p, xMm: ncx - p.wMm / 2, yMm: ncy - p.hMm / 2 };
   127→}

   129→function toBase64(bytes: Uint8Array): string {
   130→  let binary = "";
   131→  const chunk = 0x8000;
   132→  for (let i = 0; i < bytes.length; i += chunk) {
   133→    binary += String.fromCharCode(
   134→      ...Array.from(bytes.subarray(i, i + chunk)),
   135→    );
   136→  }
   137→  return btoa(binary);
   138→}

   144→function loadPdfTask(file: LoadedFile): Promise<PDFDocumentLoadingTask> {
   145→  const cached = pdfTaskCache.get(file.id);
   146→  if (cached) return cached;
   147→  const task = (async () => {
   148→    const pdfjs = await import("pdfjs-dist");
   149→    pdfjs.GlobalWorkerOptions.workerSrc = "/pdf.worker.min.mjs";
   150→    // pdf.js may transfer the buffer to the worker — always hand it a copy.
   151→    const data = file.bytes!.slice();
   152→    return pdfjs.getDocument({ data });
   153→  })();
   154→  pdfTaskCache.set(file.id, task);
   155→  return task;
   156→}

   158→async function loadPdfDoc(file: LoadedFile): Promise<PDFDocumentProxy> {
   159→  const task = await loadPdfTask(file);
   160→  return task.promise;
   161→}

   163→async function inspectPdf(file: LoadedFile): Promise<Partial<LoadedFile>> {
   164→  try {
   165→    const doc = await loadPdfDoc(file);
   166→    const page1 = await doc.getPage(1);
   167→    const vp = page1.getViewport({ scale: 1 });
   168→    const wMm = vp.width * PT_TO_MM;
   169→    const hMm = vp.height * PT_TO_MM;
   170→    const matched = matchPaper(wMm, hMm);
   171→    return {
   172→      status: "ready",
   173→      pageCount: doc.numPages,
   174→      docWidthMm: wMm,
   175→      docHeightMm: hMm,
   176→      docLabel: matched ? getPaper(matched).shortLabel : sizeLabel(wMm, hMm),
   177→    };
   178→  } catch {
   179→    pdfTaskCache.delete(file.id);
   180→    return {
   181→      status: "error",
   182→      error: "Dokumen PDF rusak, terproteksi, atau tidak dapat dibaca.",
   183→    };
   184→  }
   185→}

   187→async function inspectImage(file: LoadedFile): Promise<Partial<LoadedFile>> {
   188→  if (!file.bytes) {
   189→    return { status: "error", error: "Data gambar kosong." };
   190→  }
   191→  const url = URL.createObjectURL(
   192→    new Blob([file.bytes.slice().buffer as ArrayBuffer], { type: file.mime }),
   193→  );
   194→  try {
   195→    const img = await new Promise<HTMLImageElement>((resolve, reject) => {
   196→      const el = new Image();
   197→      el.onload = () => resolve(el);
   198→      el.onerror = () => reject(new Error("decode"));
   199→      el.src = url;
   200→    });
   201→    return {
   202→      status: "ready",
   203→      url,
   204→      pageCount: 1,
   205→      docWidthMm: img.naturalWidth * PX96_TO_MM,
   206→      docHeightMm: img.naturalHeight * PX96_TO_MM,
   207→      docLabel: `${img.naturalWidth} × ${img.naturalHeight} px`,
   208→    };
   209→  } catch {
   210→    URL.revokeObjectURL(url);
   211→    return { status: "error", error: "Gambar tidak dapat dibaca atau rusak." };
   212→  }
   213→}

   215→async function renderDocx(file: LoadedFile): Promise<DocxRender> {
   216→  const cached = docxRenderCache.get(file.id);
   217→  if (cached) return cached;
   218→  const task = (async () => {
   219→    const docxPreview = await import("docx-preview");
   220→    const host = document.createElement("div");
   221→    host.style.cssText =
   222→      "position:fixed;left:-100000px;top:0;width:1200px;visibility:hidden;";
   223→    document.body.appendChild(host);
   224→    try {
   225→      await docxPreview.renderAsync(
   226→        new Blob([file.bytes!.slice().buffer as ArrayBuffer], { type: file.mime }),
   227→        host,
   228→        host,
   229→        {
   230→          inWrapper: true,
   231→          breakPages: true,
   232→          experimental: true,
   233→          renderHeaders: true,
   234→          renderFooters: true,
   235→          useBase64URL: true,
   236→          ignoreWidth: false,
   237→          ignoreHeight: false,
   238→        },
   239→      );
   240→      const wrapper = host.querySelector(".docx-wrapper");
   241→      const sections = wrapper
   242→        ? Array.from(wrapper.querySelectorAll("section.docx"))
   243→        : Array.from(host.querySelectorAll("section.docx"));
   244→      if (sections.length === 0) throw new Error("no-sections");
   245→      const first = sections[0] as HTMLElement;
   246→      const wpx = first.offsetWidth || Math.round(210 * (96 / 25.4));
   247→      const hpx = first.offsetHeight || Math.round(297 * (96 / 25.4));
   248→      return {
   249→        sectionsHtml: sections.map((s) => (s as HTMLElement).outerHTML),
   250→        pageWpx: wpx,
   251→        pageHpx: hpx,
   252→        pageWmm: wpx * PX96_TO_MM,
   253→        pageHmm: hpx * PX96_TO_MM,
   254→        pageCount: sections.length,
   255→      } satisfies DocxRender;
   256→    } finally {
   257→      host.remove();
   258→    }
   259→  })();
   260→  docxRenderCache.set(file.id, task);
   261→  return task;
   262→}

   264→async function convertViaServer(file: LoadedFile): Promise<string> {
   265→  const cached = docHtmlCache.get(file.id);
   266→  if (cached) return cached;
   267→  const task = (async () => {
   268→    const form = new FormData();
   269→    form.append(
   270→      "file",
   271→      new Blob([file.bytes!.slice().buffer as ArrayBuffer], { type: file.mime }),
   272→      file.name,
   273→    );
   274→    const res = await fetch("/api/convert-docx", { method: "POST", body: form });
   275→    if (!res.ok) {
   276→      const body = (await res.json().catch(() => null)) as { message?: string } | null;
   277→      throw new Error(body?.message ?? "Konversi dokumen gagal di server.");
   278→    }
   279→    const body = (await res.json()) as { html: string };
   280→    return body.html;
   281→  })();
   282→  docHtmlCache.set(file.id, task);
   283→  return task;
   284→}

   286→async function inspectWord(file: LoadedFile): Promise<Partial<LoadedFile>> {
   287→  // 1st choice: client-side docx-preview (true pagination + layout)
   288→  if (file.kind === "docx") {
   289→    try {
   290→      const render = await renderDocx(file);
   291→      const matched = matchPaper(render.pageWmm, render.pageHmm);
   292→      return {
   293→        status: "ready",
   294→        pageCount: render.pageCount,
   295→        docWidthMm: render.pageWmm,
   296→        docHeightMm: render.pageHmm,
   297→        docLabel: matched ? getPaper(matched).shortLabel : sizeLabel(render.pageWmm, render.pageHmm),
   298→      };
   299→    } catch {
   300→      docxRenderCache.delete(file.id);
   301→      // fall through to server-side mammoth
   302→    }
   303→  }
   304→  // 2nd choice: server conversion (mammoth / LibreOffice headless)
   305→  try {
   306→    await convertViaServer(file);
   307→    return {
   308→      status: "ready",
   309→      continuous: true,
   310→      pageCount: 1,
   311→      docWidthMm: 210,
   312→      docHeightMm: 297,
   313→      docLabel: "Halaman Mengalir",
   314→    };
   315→  } catch (err) {
   316→    return {
   317→      status: "error",
   318→      error:
   319→        err instanceof Error
   320→          ? err.message
   321→          : "Konversi dokumen gagal. Simpan ulang sebagai .docx atau PDF.",
   322→    };
   323→  }
   324→}

   332→export function usePrintStation() {
   333→  const [files, setFiles] = useState<LoadedFile[]>([]);
   334→  const [activeId, setActiveId] = useState<string | null>(null);
   335→  const [settings, setSettings] = useState<PrintSettings>(DEFAULT_SETTINGS);
   336→  const [photoMode, setPhotoModeState] = useState(false);
   337→  const [photoPlacements, setPhotoPlacements] = useState<PhotoPlacement[]>([]);
   338→  /** Word-like multi-selection — order matters: last id is the primary. */
   339→  const [photoSelectedIds, setPhotoSelectedIds] = useState<string[]>([]);
   340→  const [photoSettings, setPhotoSettings] = useState<PhotoSettings>(DEFAULT_PHOTO_SETTINGS);
   341→  const [page, setPage] = useState(1);
   342→  const [zoom, setZoom] = useState<"fit" | "100">("fit");
   343→  const [showMargins, setShowMargins] = useState(false);
   344→  const [printing, setPrinting] = useState(false);
   345→  const [duplexAssist, setDuplexAssist] = useState<DuplexAssist | null>(null);
   346→  const [spooler, setSpooler] = useState<SpoolerState>({
   347→    state: "checking",
   348→    info: null,
   349→  });
   350→
   351→  const activeFile = useMemo(
   352→    () => files.find((f) => f.id === activeId) ?? null,
   353→    [files, activeId],
   354→  );
   355→  const activeFileRef = useRef(activeFile);
   356→  activeFileRef.current = activeFile;
   357→  const settingsRef = useRef(settings);
   358→  settingsRef.current = settings;
   359→  const filesRef = useRef(files);
   360→  filesRef.current = files;
   361→  const photoModeRef = useRef(photoMode);
   362→  photoModeRef.current = photoMode;
   363→  const photoPlacementsRef = useRef(photoPlacements);
   364→  photoPlacementsRef.current = photoPlacements;
   365→  const photoSelectedId = photoSelectedIds[photoSelectedIds.length - 1] ?? null;
   366→  const photoSelectedIdsRef = useRef(photoSelectedIds);
   367→  photoSelectedIdsRef.current = photoSelectedIds;
   368→  const photoSelectedIdRef = useRef<string | null>(photoSelectedId);
   369→  photoSelectedIdRef.current = photoSelectedId;
   370→  const photoSettingsRef = useRef(photoSettings);
   371→  photoSettingsRef.current = photoSettings;
   372→
   373→  // -- derived paper geometry -----------------------------------------------
   374→  const paper = getPaper(settings.paperId);
   375→  const { wMm: paperWmm, hMm: paperHmm } = paperDims(
   376→    settings.paperId,
   377→    settings.orientation,
   378→  );
   379→  const paperDimsRef = useRef({ wMm: paperWmm, hMm: paperHmm });
   380→  paperDimsRef.current = { wMm: paperWmm, hMm: paperHmm };
   381→  // Assigned after the photo actions below — used by addFiles to auto-place images.
   382→  const intakeImageRef = useRef<(itemId: string) => void>(() => {});
   383→  // True when the user hand-edited the sheet since the last auto-grid arrange;
   384→  // intake auto-arrange never reshuffles a hand-tuned layout.
   385→  const layoutTouchedRef = useRef(false);
   386→  const intakeTimerRef = useRef<number | null>(null);
   387→
   388→  const contentScale = useMemo(() => {
   389→    const f = activeFile;
   390→    if (!f || f.docWidthMm == null || f.docWidthMm <= 0) return 1;
   391→    if (settings.scaleMode !== "fit" || f.continuous) return 1;
   392→    const paperW = paperWmm;
   393→    const paperH = paperHmm;
   394→    return Math.min(paperW / f.docWidthMm, paperH / (f.docHeightMm || 1));
   395→  }, [activeFile, settings.scaleMode, paperWmm, paperHmm]);
   396→
   397→  const paperMismatch = useMemo(() => {
   398→    const f = activeFile;
   399→    if (!f || f.docWidthMm == null || f.continuous) return false;
   400→    return matchPaper(f.docWidthMm, f.docHeightMm ?? 0) !== settings.paperId;
   401→  }, [activeFile, settings.paperId]);
   402→
   403→  const contentOverflows = useMemo(() => {
   404→    const f = activeFile;
   405→    if (!f || f.docWidthMm == null || f.continuous) return false;
   406→    const s = settings.scaleMode === "fit" ? contentScale : 1;
   407→    return (
   408→      f.docWidthMm * s > paperWmm + 0.5 || (f.docHeightMm ?? 0) * s > paperHmm + 0.5
   409→    );
   410→  }, [activeFile, contentScale, settings.scaleMode, paperWmm, paperHmm]);
   411→
   412→  // -- page range -------------------------------------------------------------
   413→  const selectedPages = useMemo(() => {
   414→    const f = activeFile;
   415→    if (!f || f.status !== "ready" || !f.pageCount) return [];
   416→    if (settings.rangeMode === "all") {
   417→      return Array.from({ length: f.pageCount }, (_, i) => i + 1);
   418→    }
   419→    const parsed = parsePageRange(settings.rangeText, f.pageCount);
   420→    return parsed.valid ? parsed.pages : [];
   421→  }, [activeFile, settings.rangeMode, settings.rangeText]);
   422→
   423→  const rangeInvalid =
   424→    settings.rangeMode === "custom" &&
   425→    !parsePageRange(settings.rangeText, activeFile?.pageCount ?? 1).valid;
   426→
   427→  const selectedPagesRef = useRef<number[]>([]);
   428→  selectedPagesRef.current = selectedPages;
   429→
   430→  const photoReady =
   431→    photoPlacements.length > 0 &&
   432→    photoPlacements.every((p) => {
   433→      const it = files.find((f) => f.id === p.itemId);
   434→      return !!it && it.kind === "image" && it.status === "ready" && !!it.url;
   435→    });
   436→  const photoPreset = getPhotoPreset(photoSettings.presetId);
   437→  const photoSelected =
   438→    photoPlacements.find((p) => p.id === photoSelectedId) ?? null;
   439→
   440→  /**
   441→   * Max copies of the current preset that fit on one sheet (best of portrait
   442→   * / landscape orientation), clamped by the engine's placement cap.
   443→   */
   444→  const photoCapacity = useMemo(() => {
   445→    const w = photoPreset.group === "custom" ? photoSettings.customWMm : photoPreset.wMm;
   446→    const h = photoPreset.group === "custom" ? photoSettings.customHMm : photoPreset.hMm;
   447→    const margin = photoSettings.marginMm;
   448→    const gap = photoSettings.gapMm;
   449→    const printW = paperWmm - 2 * margin;
   450→    const printH = paperHmm - 2 * margin;
   451→    const capFor = (pw: number, ph: number) => {
   452→      if (pw > printW + PHOTO_TOL || ph > printH + PHOTO_TOL) return 0;
   453→      return (
   454→        Math.max(1, Math.floor((printW + gap) / (pw + gap))) *
   455→        Math.max(1, Math.floor((printH + gap) / (ph + gap)))
   456→      );
   457→    };
   458→    return Math.min(Math.max(capFor(w, h), capFor(h, w)), MAX_PLACEMENTS);
   459→  }, [photoPreset, photoSettings, paperWmm, paperHmm]);
   460→
   461→  const photoBeyondMargin = photoPlacements.some((p) => {
   462→    const bb = placementBBox(p);
   463→    return (
   464→      bb.x < photoSettings.marginMm - PHOTO_TOL ||
   465→      bb.y < photoSettings.marginMm - PHOTO_TOL ||
   466→      bb.x + bb.w > paperWmm - photoSettings.marginMm + PHOTO_TOL ||
   467→      bb.y + bb.h > paperHmm - photoSettings.marginMm + PHOTO_TOL
   468→    );
   469→  });
   470→
   471→  const canPrint = photoMode
   472→    ? photoReady && !printing
   473→    : !!activeFile &&
   474→      activeFile.status === "ready" &&
   475→      selectedPages.length > 0 &&
   476→      !printing;
   477→
   478→  // -- spooler detection ------------------------------------------------------
   479→  useEffect(() => {
   480→    let alive = true;
   481→    fetch("/api/print")
   482→      .then((r) => r.json())
   483→      .then((body: { spooler: { type: "ipp" | "lp"; printer: string | null } | null }) => {
   484→        if (!alive) return;
   485→        setSpooler(
   486→          body.spooler
   487→            ? { state: "available", info: body.spooler }
   488→            : { state: "unavailable", info: null },
   489→        );
   490→      })
   491→      .catch(() => alive && setSpooler({ state: "unavailable", info: null }));
   492→    return () => {
   493→      alive = false;
   494→    };
   495→  }, []);
   496→
   497→  // -- auto-fit to loaded tray ------------------------------------------------
   498→  const autoFitKey = useRef<string | null>(null);
   499→  useEffect(() => {
   500→    const f = activeFile;
   501→    if (!f || f.status !== "ready" || f.docWidthMm == null || f.continuous) return;
   502→    const key = `${f.id}:${settings.paperId}`;
   503→    if (autoFitKey.current === key) return;
   504→    const matched = matchPaper(f.docWidthMm, f.docHeightMm ?? 0);
   505→    if (matched !== settings.paperId) {
   506→      autoFitKey.current = key;
   507→      setSettings((s) => (s.scaleMode === "fit" ? s : { ...s, scaleMode: "fit" }));
   508→    }
   509→  }, [activeFile, settings.paperId]);
   510→
   511→  // -- file intake ------------------------------------------------------------
   512→  const patchFile = useCallback((id: string, patch: Partial<LoadedFile>) => {
   513→    setFiles((prev) => prev.map((f) => (f.id === id ? { ...f, ...patch } : f)));
   514→  }, []);
   515→
   516→  const addFiles = useCallback(
   517→    (incoming: File[]) => {
   518→      for (const file of incoming) {
   519→        const ext = (file.name.split(".").pop() ?? "").toLowerCase();
   520→        const accepted = ACCEPTED_EXT[ext];
   521→        if (!accepted) {
   522→          toast.error(`Format .${ext || "?"} tidak didukung.`, {
   523→            description: `${file.name} — gunakan PDF, DOC, DOCX, PNG, atau JPG.`,
   524→          });
   525→          continue;
   526→        }
   527→        if (file.size > MAX_FILE_BYTES) {
   528→          toast.error(`Ukuran file melebihi 50 MB.`, {
   529→            description: file.name,
   530→          });
   531→          continue;
   532→        }
   533→        const id = newId();
   534→        const loaded: LoadedFile = {
   535→          id,
   536→          name: file.name,
   537→          ext,
   538→          sizeBytes: file.size,
   539→          kind: accepted.kind,
   540→          mime: accepted.mime,
   541→          bytes: null,
   542→          url: null,
   543→          status: "loading",
   544→          pageCount: null,
   545→          docWidthMm: null,
   546→          docHeightMm: null,

   931→  /**
   932→   * Auto-grid arrange — the print-shop "ngejajar N foto" flow. Tiles N slots
   933→   * sized to the current preset (or explicit dims) in a balanced grid
   934→   * (6 → 3×2, 8 → 4×2, 9 → 3×3) aligned inside the printable area
   935→   * (top / center / bottom × left / center / right); portrait / landscape is
   936→   * picked by whichever capacity fits N. Slots are always filled round-robin
   937→   * with every distinct ready photo on the sheet — changing the count never
   938→   * collapses the sheet into copies of a hand-picked photo (use "Isi Penuh"
   939→   * for that).
   940→   */
   941→  const arrangePhotoCopies = useCallback(
   942→    (
   943→      count: number,
   944→      options?: {
   945→        dims?: { wMm: number; hMm: number };
   946→        auto?: boolean;
   947→        /** Grow the slot count so every distinct uploaded photo keeps a slot
   948→         * (several customers, one collective sheet). Intake-driven arranges
   949→         * always set this so a newly added face photo is never dropped. */
   950→        coverAll?: boolean;
   951→        /** Explicit margin override — used when margin changes and re-tile
   952→         * happens in the same tick (borderless toggle), where the settings
   953→         * ref is still stale. */
   954→        marginMm?: number;
   955→        gapMm?: number;
   956→        alignV?: GridAlignV;
   957→        alignH?: GridAlignH;
   958→      },
   959→    ) => {
   960→      const list = photoPlacementsRef.current;
   961→      if (list.length === 0) return;
   962→      const ps = photoSettingsRef.current;
   963→      const { wMm: paperW, hMm: paperH } = paperDimsRef.current;
   964→      const preset = getPhotoPreset(ps.presetId);
   965→      const presetW = preset.group === "custom" ? ps.customWMm : preset.wMm;
   966→      const presetH = preset.group === "custom" ? ps.customHMm : preset.hMm;
   967→      const pw = Math.min(options?.dims ? options.dims.wMm : presetW, paperW);
   968→      const ph = Math.min(options?.dims ? options.dims.hMm : presetH, paperH);
   969→      const margin = options?.marginMm ?? ps.marginMm;
   970→      const gap = options?.gapMm ?? ps.gapMm;
   971→      const alignV = options?.alignV ?? ps.alignV;
   972→      const alignH = options?.alignH ?? ps.alignH;
   973→      const printW = paperW - 2 * margin;
   974→      const printH = paperH - 2 * margin;
   975→
   976→      const fitFor = (w: number, h: number) => {
   977→        if (w > printW + PHOTO_TOL || h > printH + PHOTO_TOL) return null;
   978→        const cols = Math.max(1, Math.floor((printW + gap) / (w + gap)));
   979→        const rows = Math.max(1, Math.floor((printH + gap) / (h + gap)));
   980→        return { cols, rows, cap: cols * rows };
   981→      };
   982→
   983→      const portrait = fitFor(pw, ph);
   984→      const landscape = fitFor(ph, pw);
   985→      const n0 = Math.max(1, Math.round(count));
   986→      // Keep the preset's own orientation whenever it can fit N copies —
   987→      // rotating a pas-foto frame would crop faces. Only fall back to the
   988→      // rotated fit when portrait cannot hold N (or doesn't fit at all).
   989→      let useLandscape: boolean;
   990→      if (portrait && landscape) {
   991→        useLandscape =
   992→          portrait.cap >= n0
   993→            ? false
   994→            : landscape.cap >= n0
   995→              ? true
   996→              : landscape.cap > portrait.cap;
   997→      } else {
   998→        useLandscape = !portrait && !!landscape;
   999→      }
  1000→      const fit = useLandscape ? landscape : portrait;
  1001→      if (!fit) {
  1002→        if (!options?.auto) {
  1003→          toast.error("Foto melebihi area cetak lembar.", {
  1004→            description:
  1005→              "Kecilkan ukuran foto atau margin lembar, lalu susun ulang.",
  1006→          });
  1007→        }
  1008→        return;
  1009→      }
  1010→      // Slot sources: the distinct ready images already on the sheet fill
  1011→      // slots round-robin — every uploaded photo stays on the grid. Sources
  1012→      // resolve first so intake arranges can grow the slot count to cover
  1013→      // every distinct photo (several customers, one collective sheet).
  1014→      let sources: string[] = [];
  1015→      const seen = new Set<string>();
  1016→      for (const p of list) {
  1017→        if (seen.has(p.itemId)) continue;
  1018→        const it = filesRef.current.find(
  1019→          (f) =>
  1020→            f.id === p.itemId &&
  1021→            f.kind === "image" &&
  1022→            f.status === "ready" &&
  1023→            !!f.url,
  1024→        );
  1025→        if (it) {
  1026→          seen.add(p.itemId);
  1027→          sources.push(p.itemId);
  1028→        }
  1029→      }
  1030→      if (sources.length === 0) sources = [list[0].itemId];
  1031→
  1032→      const baseTotal = Math.min(n0, fit.cap, MAX_PLACEMENTS);
  1033→      let total = baseTotal;
  1034→      // Intake-driven arrange ("coverAll"): never drop a customer's photo —
  1035→      // when more distinct photos are on the sheet than slots, the grid grows
  1036→      // to fit all of them (up to sheet capacity).
  1037→      let grown = false;
  1038→      if (options?.coverAll && sources.length > total) {
  1039→        total = Math.min(sources.length, fit.cap, MAX_PLACEMENTS);
  1040→        grown = total > baseTotal;
  1041→      }
  1042→      const overCap = Math.max(0, sources.length - total);
  1043→
  1044→      // Balanced grid: fewest wasted slots first, then the most square block,
  1045→      // preferring the wider option on ties (6 → 3×2, 9 → 3×3, 8 → 4×2).
  1046→      let cols = fit.cols;
  1047→      let bestScore = Infinity;
  1048→      for (let c = 1; c <= fit.cols; c++) {
  1049→        const r = Math.ceil(total / c);
  1050→        if (r > fit.rows) continue;
  1051→        const score = (c * r - total) * 1000 + Math.abs(c - r);
  1052→        if (score < bestScore || (score === bestScore && c > cols)) {
  1053→          bestScore = score;
  1054→          cols = c;
  1055→        }
  1056→      }
  1057→      const w = useLandscape ? ph : pw;
  1058→      const h = useLandscape ? pw : ph;
  1059→      const rows = Math.ceil(total / cols);
  1060→
  1061→      const gridH = rows * h + (rows - 1) * gap;
  1062→      const gy =
  1063→        alignV === "top"
  1064→          ? margin
  1065→          : alignV === "bottom"
  1066→            ? margin + printH - gridH
  1067→            : margin + (printH - gridH) / 2;
  1068→      const out: PhotoPlacement[] = [];
  1069→      for (let r = 0; r < rows; r++) {
  1070→        const inRow = Math.min(total - r * cols, cols);
  1071→        const rowW = inRow * w + (inRow - 1) * gap;
  1072→        const gx =
  1073→          alignH === "left"
  1074→            ? margin
  1075→            : alignH === "right"
