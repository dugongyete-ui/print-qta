    80→    80→function normRotation(rot: number): number {
    81→    81→  const r = Math.round(rot / 90) * 90;
    82→    82→  return ((r % 360) + 360) % 360;
    83→    83→}

    85→    85→function swappedBBox(rotation: number): boolean {
    86→    86→  const r = normRotation(rotation);
    87→    87→  return r === 90 || r === 270;
    88→    88→}

    90→    90→/** Visual bounding box (mm) of a placement, accounting for rotation. */
    91→    91→function placementBBox(p: {
    92→    92→  xMm: number;
    93→    93→  yMm: number;
    94→    94→  wMm: number;
    95→    95→  hMm: number;
    96→    96→  rotation: number;
    97→    97→}): { x: number; y: number; w: number; h: number } {
    98→    98→  const swapped = swappedBBox(p.rotation);
    99→    99→  const w = swapped ? p.hMm : p.wMm;
   100→   100→  const h = swapped ? p.wMm : p.hMm;
   101→   101→  const cx = p.xMm + p.wMm / 2;
   102→   102→  const cy = p.yMm + p.hMm / 2;
   103→   103→  return { x: cx - w / 2, y: cy - h / 2, w, h };
   104→   104→}

   106→   106→/** Keep the visual bbox fully on the paper by clamping the rotation center. */
   107→   107→function clampPlacement(
   108→   108→  p: PhotoPlacement,
   109→   109→  paperWmm: number,
   110→   110→  paperHmm: number,
   111→   111→): PhotoPlacement {
   112→   112→  const bb = placementBBox(p);
   113→   113→  const cx = p.xMm + p.wMm / 2;
   114→   114→  const cy = p.yMm + p.hMm / 2;
   115→   115→  const ncx = Math.min(
   116→   116→    Math.max(cx, bb.w / 2),
   117→   117→    Math.max(bb.w / 2, paperWmm - bb.w / 2),
   118→   118→  );
   119→   119→  const ncy = Math.min(
   120→   120→    Math.max(cy, bb.h / 2),
   121→   121→    Math.max(bb.h / 2, paperHmm - bb.h / 2),
   122→   122→  );
   123→   123→  return { ...p, xMm: ncx - p.wMm / 2, yMm: ncy - p.hMm / 2 };
   124→   124→}

   126→   126→function toBase64(bytes: Uint8Array): string {
   127→   127→  let binary = "";
   128→   128→  const chunk = 0x8000;
   129→   129→  for (let i = 0; i < bytes.length; i += chunk) {
   130→   130→    binary += String.fromCharCode(
   131→   131→      ...Array.from(bytes.subarray(i, i + chunk)),
   132→   132→    );
   133→   133→  }
   134→   134→  return btoa(binary);
   135→   135→}

   141→   141→function loadPdfTask(file: LoadedFile): Promise<PDFDocumentLoadingTask> {
   142→   142→  const cached = pdfTaskCache.get(file.id);
   143→   143→  if (cached) return cached;
   144→   144→  const task = (async () => {
   145→   145→    const pdfjs = await import("pdfjs-dist");
   146→   146→    pdfjs.GlobalWorkerOptions.workerSrc = "/pdf.worker.min.mjs";
   147→   147→    // pdf.js may transfer the buffer to the worker — always hand it a copy.
   148→   148→    const data = file.bytes!.slice();
   149→   149→    return pdfjs.getDocument({ data });
   150→   150→  })();
   151→   151→  pdfTaskCache.set(file.id, task);
   152→   152→  return task;
   153→   153→}

   184→   184→async function inspectImage(file: LoadedFile): Promise<Partial<LoadedFile>> {
   185→   185→  if (!file.bytes) {
   186→   186→    return { status: "error", error: "Data gambar kosong." };
   187→   187→  }
   188→   188→  const url = URL.createObjectURL(
   189→   189→    new Blob([file.bytes.slice().buffer as ArrayBuffer], { type: file.mime }),
   190→   190→  );
   191→   191→  try {
   192→   192→    const img = await new Promise<HTMLImageElement>((resolve, reject) => {
   193→   193→      const el = new Image();
   194→   194→      el.onload = () => resolve(el);
   195→   195→      el.onerror = () => reject(new Error("decode"));
   196→   196→      el.src = url;
   197→   197→    });
   198→   198→    return {
   199→   199→      status: "ready",
   200→   200→      url,
   201→   201→      pageCount: 1,
   202→   202→      docWidthMm: img.naturalWidth * PX96_TO_MM,
   203→   203→      docHeightMm: img.naturalHeight * PX96_TO_MM,
   204→   204→      docLabel: `${img.naturalWidth} × ${img.naturalHeight} px`,
   205→   205→    };
   206→   206→  } catch {
   207→   207→    URL.revokeObjectURL(url);
   208→   208→    return { status: "error", error: "Gambar tidak dapat dibaca atau rusak." };
   209→   209→  }
   210→   210→}

   212→   212→async function renderDocx(file: LoadedFile): Promise<DocxRender> {
   213→   213→  const cached = docxRenderCache.get(file.id);
   214→   214→  if (cached) return cached;
   215→   215→  const task = (async () => {
   216→   216→    const docxPreview = await import("docx-preview");
   217→   217→    const host = document.createElement("div");
   218→   218→    host.style.cssText =
   219→   219→      "position:fixed;left:-100000px;top:0;width:1200px;visibility:hidden;";
   220→   220→    document.body.appendChild(host);
   221→   221→    try {
   222→   222→      await docxPreview.renderAsync(
   223→   223→        new Blob([file.bytes!.slice().buffer as ArrayBuffer], { type: file.mime }),
   224→   224→        host,
   225→   225→        host,
   226→   226→        {
   227→   227→          inWrapper: true,
   228→   228→          breakPages: true,
   229→   229→          experimental: true,
   230→   230→          renderHeaders: true,
   231→   231→          renderFooters: true,
   232→   232→          useBase64URL: true,
   233→   233→          ignoreWidth: false,
   234→   234→          ignoreHeight: false,
   235→   235→        },
   236→   236→      );
   237→   237→      const wrapper = host.querySelector(".docx-wrapper");
   238→   238→      const sections = wrapper
   239→   239→        ? Array.from(wrapper.querySelectorAll("section.docx"))
   240→   240→        : Array.from(host.querySelectorAll("section.docx"));
   241→   241→      if (sections.length === 0) throw new Error("no-sections");
   242→   242→      const first = sections[0] as HTMLElement;
   243→   243→      const wpx = first.offsetWidth || Math.round(210 * (96 / 25.4));
   244→   244→      const hpx = first.offsetHeight || Math.round(297 * (96 / 25.4));
   245→   245→      return {
   246→   246→        sectionsHtml: sections.map((s) => (s as HTMLElement).outerHTML),
   247→   247→        pageWpx: wpx,
   248→   248→        pageHpx: hpx,
   249→   249→        pageWmm: wpx * PX96_TO_MM,
   250→   250→        pageHmm: hpx * PX96_TO_MM,
   251→   251→        pageCount: sections.length,
   252→   252→      } satisfies DocxRender;
   253→   253→    } finally {
   254→   254→      host.remove();
   255→   255→    }
   256→   256→  })();
   257→   257→  docxRenderCache.set(file.id, task);
   258→   258→  return task;
   259→   259→}

   261→   261→async function convertViaServer(file: LoadedFile): Promise<string> {
   262→   262→  const cached = docHtmlCache.get(file.id);
   263→   263→  if (cached) return cached;
   264→   264→  const task = (async () => {
   265→   265→    const form = new FormData();
   266→   266→    form.append(
   267→   267→      "file",
   268→   268→      new Blob([file.bytes!.slice().buffer as ArrayBuffer], { type: file.mime }),
   269→   269→      file.name,
   270→   270→    );
   271→   271→    const res = await fetch("/api/convert-docx", { method: "POST", body: form });
   272→   272→    if (!res.ok) {
   273→   273→      const body = (await res.json().catch(() => null)) as { message?: string } | null;
   274→   274→      throw new Error(body?.message ?? "Konversi dokumen gagal di server.");
   275→   275→    }
   276→   276→    const body = (await res.json()) as { html: string };
   277→   277→    return body.html;
   278→   278→  })();
   279→   279→  docHtmlCache.set(file.id, task);
   280→   280→  return task;
   281→   281→}

   433→   433→      bb.x + bb.w > paperWmm - photoSettings.marginMm + PHOTO_TOL ||
   434→   434→      bb.y + bb.h > paperHmm - photoSettings.marginMm + PHOTO_TOL
   435→   435→    );
   436→   436→  });
   437→   437→
   438→   438→  const canPrint = photoMode
   439→   439→    ? photoReady && !printing
   440→   440→    : !!activeFile &&
   441→   441→      activeFile.status === "ready" &&
   442→   442→      selectedPages.length > 0 &&
   443→   443→      !printing;
   444→   444→
   445→   445→  // -- spooler detection ------------------------------------------------------
   446→   446→  useEffect(() => {
   447→   447→    let alive = true;
   448→   448→    fetch("/api/print")

   450→   450→      .then((body: { spooler: { type: "ipp" | "lp"; printer: string | null } | null }) => {

   452→   452→        setSpooler(
   453→   453→          body.spooler
   454→   454→            ? { state: "available", info: body.spooler }
   455→   455→            : { state: "unavailable", info: null },
   456→   456→        );
   457→   457→      })
   458→   458→      .catch(() => alive && setSpooler({ state: "unavailable", info: null }));
   459→   459→    return () => {
   460→   460→      alive = false;
