    81→function normRotation(rot: number): number {
    82→  const r = Math.round(rot / 90) * 90;
    83→  return ((r % 360) + 360) % 360;
    84→}
    85→
    86→function swappedBBox(rotation: number): boolean {
    87→  const r = normRotation(rotation);
    88→  return r === 90 || r === 270;
    89→}

    91→/** Visual bounding box (mm) of a placement, accounting for rotation. */
    92→function placementBBox(p: {
    93→  xMm: number;
    94→  yMm: number;
    95→  wMm: number;
    96→  hMm: number;
    97→  rotation: number;
    98→}): { x: number; y: number; w: number; h: number } {
    99→  const swapped = swappedBBox(p.rotation);
   100→  const w = swapped ? p.hMm : p.wMm;
   101→  const h = swapped ? p.wMm : p.hMm;
   102→  const cx = p.xMm + p.wMm / 2;
   103→  const cy = p.yMm + p.hMm / 2;
   104→  return { x: cx - w / 2, y: cy - h / 2, w, h };
   105→}
   106→
   107→/** Keep the visual bbox fully on the paper by clamping the rotation center. */
   108→function clampPlacement(
   109→  p: PhotoPlacement,
   110→  paperWmm: number,
   111→  paperHmm: number,
   112→): PhotoPlacement {
   113→  const bb = placementBBox(p);
   114→  const cx = p.xMm + p.wMm / 2;
   115→  const cy = p.yMm + p.hMm / 2;
   116→  const ncx = Math.min(
   117→    Math.max(cx, bb.w / 2),
   118→    Math.max(bb.w / 2, paperWmm - bb.w / 2),
   119→  );
   120→  const ncy = Math.min(
   121→    Math.max(cy, bb.h / 2),
   122→    Math.max(bb.h / 2, paperHmm - bb.h / 2),
   123→  );
   124→  return { ...p, xMm: ncx - p.wMm / 2, yMm: ncy - p.hMm / 2 };
   125→}

   127→function toBase64(bytes: Uint8Array): string {
   128→  let binary = "";
   129→  const chunk = 0x8000;
   130→  for (let i = 0; i < bytes.length; i += chunk) {
   131→    binary += String.fromCharCode(
   132→      ...Array.from(bytes.subarray(i, i + chunk)),
   133→    );
   134→  }
   135→  return btoa(binary);
   136→}

   142→function loadPdfTask(file: LoadedFile): Promise<PDFDocumentLoadingTask> {
   143→  const cached = pdfTaskCache.get(file.id);
   144→  if (cached) return cached;
   145→  const task = (async () => {
   146→    const pdfjs = await import("pdfjs-dist");
   147→    pdfjs.GlobalWorkerOptions.workerSrc = "/pdf.worker.min.mjs";
   148→    // pdf.js may transfer the buffer to the worker — always hand it a copy.
   149→    const data = file.bytes!.slice();
   150→    return pdfjs.getDocument({ data });
   151→  })();
   152→  pdfTaskCache.set(file.id, task);
   153→  return task;
   154→}

   156→async function loadPdfDoc(file: LoadedFile): Promise<PDFDocumentProxy> {
   157→  const task = await loadPdfTask(file);
   158→  return task.promise;
   159→}

   161→async function inspectPdf(file: LoadedFile): Promise<Partial<LoadedFile>> {
   162→  try {
   163→    const doc = await loadPdfDoc(file);
   164→    const page1 = await doc.getPage(1);
   165→    const vp = page1.getViewport({ scale: 1 });
   166→    const wMm = vp.width * PT_TO_MM;
   167→    const hMm = vp.height * PT_TO_MM;
   168→    const matched = matchPaper(wMm, hMm);
   169→    return {
   170→      status: "ready",
   171→      pageCount: doc.numPages,
   172→      docWidthMm: wMm,
   173→      docHeightMm: hMm,
   174→      docLabel: matched ? getPaper(matched).shortLabel : sizeLabel(wMm, hMm),
   175→    };
   176→  } catch {
   177→    pdfTaskCache.delete(file.id);
   178→    return {
   179→      status: "error",
   180→      error: "Dokumen PDF rusak, terproteksi, atau tidak dapat dibaca.",
   181→    };
   182→  }
   183→}

   185→async function inspectImage(file: LoadedFile): Promise<Partial<LoadedFile>> {
   186→  if (!file.bytes) {
   187→    return { status: "error", error: "Data gambar kosong." };
   188→  }
   189→  const url = URL.createObjectURL(
   190→    new Blob([file.bytes.slice().buffer as ArrayBuffer], { type: file.mime }),
   191→  );
   192→  try {
   193→    const img = await new Promise<HTMLImageElement>((resolve, reject) => {
   194→      const el = new Image();
   195→      el.onload = () => resolve(el);
   196→      el.onerror = () => reject(new Error("decode"));
   197→      el.src = url;
   198→    });
   199→    return {
   200→      status: "ready",
   201→      url,
   202→      pageCount: 1,
   203→      docWidthMm: img.naturalWidth * PX96_TO_MM,
   204→      docHeightMm: img.naturalHeight * PX96_TO_MM,
   205→      docLabel: `${img.naturalWidth} × ${img.naturalHeight} px`,
   206→    };
   207→  } catch {
   208→    URL.revokeObjectURL(url);
   209→    return { status: "error", error: "Gambar tidak dapat dibaca atau rusak." };
   210→  }
   211→}

   213→async function renderDocx(file: LoadedFile): Promise<DocxRender> {
   214→  const cached = docxRenderCache.get(file.id);
   215→  if (cached) return cached;
   216→  const task = (async () => {
   217→    const docxPreview = await import("docx-preview");
   218→    const host = document.createElement("div");
   219→    host.style.cssText =
   220→      "position:fixed;left:-100000px;top:0;width:1200px;visibility:hidden;";
   221→    document.body.appendChild(host);
   222→    try {
   223→      await docxPreview.renderAsync(
   224→        new Blob([file.bytes!.slice().buffer as ArrayBuffer], { type: file.mime }),
   225→        host,
   226→        host,
   227→        {
   228→          inWrapper: true,
   229→          breakPages: true,
   230→          experimental: true,
   231→          renderHeaders: true,
   232→          renderFooters: true,
   233→          useBase64URL: true,
   234→          ignoreWidth: false,
   235→          ignoreHeight: false,
   236→        },
   237→      );
   238→      const wrapper = host.querySelector(".docx-wrapper");
   239→      const sections = wrapper
   240→        ? Array.from(wrapper.querySelectorAll("section.docx"))
   241→        : Array.from(host.querySelectorAll("section.docx"));
   242→      if (sections.length === 0) throw new Error("no-sections");
   243→      const first = sections[0] as HTMLElement;
   244→      const wpx = first.offsetWidth || Math.round(210 * (96 / 25.4));
   245→      const hpx = first.offsetHeight || Math.round(297 * (96 / 25.4));
   246→      return {
   247→        sectionsHtml: sections.map((s) => (s as HTMLElement).outerHTML),
   248→        pageWpx: wpx,
   249→        pageHpx: hpx,
   250→        pageWmm: wpx * PX96_TO_MM,
   251→        pageHmm: hpx * PX96_TO_MM,
   252→        pageCount: sections.length,
   253→      } satisfies DocxRender;
   254→    } finally {
   255→      host.remove();
   256→    }
   257→  })();
   258→  docxRenderCache.set(file.id, task);
   259→  return task;
   260→}

   262→async function convertViaServer(file: LoadedFile): Promise<string> {
   263→  const cached = docHtmlCache.get(file.id);
   264→  if (cached) return cached;
   265→  const task = (async () => {
   266→    const form = new FormData();
   267→    form.append(
   268→      "file",
   269→      new Blob([file.bytes!.slice().buffer as ArrayBuffer], { type: file.mime }),
   270→      file.name,
   271→    );
   272→    const res = await fetch("/api/convert-docx", { method: "POST", body: form });
   273→    if (!res.ok) {
   274→      const body = (await res.json().catch(() => null)) as { message?: string } | null;
   275→      throw new Error(body?.message ?? "Konversi dokumen gagal di server.");
   276→    }
   277→    const body = (await res.json()) as { html: string };
   278→    return body.html;
   279→  })();
   280→  docHtmlCache.set(file.id, task);
   281→  return task;
   282→}

   284→async function inspectWord(file: LoadedFile): Promise<Partial<LoadedFile>> {
   285→  // 1st choice: client-side docx-preview (true pagination + layout)
   286→  if (file.kind === "docx") {
   287→    try {
   288→      const render = await renderDocx(file);
   289→      const matched = matchPaper(render.pageWmm, render.pageHmm);
   290→      return {
   291→        status: "ready",
   292→        pageCount: render.pageCount,
   293→        docWidthMm: render.pageWmm,
   294→        docHeightMm: render.pageHmm,
   295→        docLabel: matched ? getPaper(matched).shortLabel : sizeLabel(render.pageWmm, render.pageHmm),
   296→      };
   297→    } catch {
   298→      docxRenderCache.delete(file.id);
   299→      // fall through to server-side mammoth
   300→    }
   301→  }
   302→  // 2nd choice: server conversion (mammoth / LibreOffice headless)
   303→  try {
   304→    await convertViaServer(file);
   305→    return {
   306→      status: "ready",
   307→      continuous: true,
   308→      pageCount: 1,
   309→      docWidthMm: 210,
   310→      docHeightMm: 297,
   311→      docLabel: "Halaman Mengalir",
   312→    };
   313→  } catch (err) {
   314→    return {
   315→      status: "error",
   316→      error:
   317→        err instanceof Error
   318→          ? err.message
   319→          : "Konversi dokumen gagal. Simpan ulang sebagai .docx atau PDF.",
   320→    };
   321→  }
   322→}

   330→export function usePrintStation() {
   331→  const [files, setFiles] = useState<LoadedFile[]>([]);
   332→  const [activeId, setActiveId] = useState<string | null>(null);
   333→  const [settings, setSettings] = useState<PrintSettings>(DEFAULT_SETTINGS);
   334→  const [photoMode, setPhotoModeState] = useState(false);
   335→  const [photoPlacements, setPhotoPlacements] = useState<PhotoPlacement[]>([]);
   336→  const [photoSelectedId, setPhotoSelectedId] = useState<string | null>(null);
   337→  const [photoSettings, setPhotoSettings] = useState<PhotoSettings>(DEFAULT_PHOTO_SETTINGS);
   338→  const [page, setPage] = useState(1);
   339→  const [zoom, setZoom] = useState<"fit" | "100">("fit");
   340→  const [showMargins, setShowMargins] = useState(false);
   341→  const [printing, setPrinting] = useState(false);
   342→  const [duplexAssist, setDuplexAssist] = useState<DuplexAssist | null>(null);
   343→  const [spooler, setSpooler] = useState<SpoolerState>({
   344→    state: "checking",
   345→    info: null,
   346→  });
   347→
   348→  const activeFile = useMemo(
   349→    () => files.find((f) => f.id === activeId) ?? null,
   350→    [files, activeId],
   351→  );
   352→  const activeFileRef = useRef(activeFile);
   353→  activeFileRef.current = activeFile;
   354→  const settingsRef = useRef(settings);
   355→  settingsRef.current = settings;
   356→  const filesRef = useRef(files);
   357→  filesRef.current = files;
   358→  const photoModeRef = useRef(photoMode);
   359→  photoModeRef.current = photoMode;
   360→  const photoPlacementsRef = useRef(photoPlacements);
   361→  photoPlacementsRef.current = photoPlacements;
   362→  const photoSelectedIdRef = useRef(photoSelectedId);
   363→  photoSelectedIdRef.current = photoSelectedId;
   364→  const photoSettingsRef = useRef(photoSettings);
   365→  photoSettingsRef.current = photoSettings;
   366→
   367→  // -- derived paper geometry -----------------------------------------------
   368→  const paper = getPaper(settings.paperId);
   369→  const { wMm: paperWmm, hMm: paperHmm } = paperDims(
   370→    settings.paperId,
   371→    settings.orientation,
   372→  );
   373→  const paperDimsRef = useRef({ wMm: paperWmm, hMm: paperHmm });
   374→  paperDimsRef.current = { wMm: paperWmm, hMm: paperHmm };
   375→  // Assigned after the photo actions below — used by addFiles to auto-place images.
   376→  const intakeImageRef = useRef<(itemId: string) => void>(() => {});
   377→  // True when the user hand-edited the sheet since the last auto-grid arrange;
   378→  // intake auto-arrange never reshuffles a hand-tuned layout.
   379→  const layoutTouchedRef = useRef(false);
   380→  const intakeTimerRef = useRef<number | null>(null);
   381→
   382→  const contentScale = useMemo(() => {
   383→    const f = activeFile;
   384→    if (!f || f.docWidthMm == null || f.docWidthMm <= 0) return 1;
   385→    if (settings.scaleMode !== "fit" || f.continuous) return 1;
   386→    const paperW = paperWmm;
   387→    const paperH = paperHmm;
   388→    return Math.min(paperW / f.docWidthMm, paperH / (f.docHeightMm || 1));
   389→  }, [activeFile, settings.scaleMode, paperWmm, paperHmm]);
   390→
   391→  const paperMismatch = useMemo(() => {
   392→    const f = activeFile;
   393→    if (!f || f.docWidthMm == null || f.continuous) return false;
   394→    return matchPaper(f.docWidthMm, f.docHeightMm ?? 0) !== settings.paperId;
   395→  }, [activeFile, settings.paperId]);
   396→
   397→  const contentOverflows = useMemo(() => {
   398→    const f = activeFile;
   399→    if (!f || f.docWidthMm == null || f.continuous) return false;
   400→    const s = settings.scaleMode === "fit" ? contentScale : 1;
   401→    return (
   402→      f.docWidthMm * s > paperWmm + 0.5 || (f.docHeightMm ?? 0) * s > paperHmm + 0.5
   403→    );
   404→  }, [activeFile, contentScale, settings.scaleMode, paperWmm, paperHmm]);
   405→
   406→  // -- page range -------------------------------------------------------------
   407→  const selectedPages = useMemo(() => {
   408→    const f = activeFile;
   409→    if (!f || f.status !== "ready" || !f.pageCount) return [];
   410→    if (settings.rangeMode === "all") {
   411→      return Array.from({ length: f.pageCount }, (_, i) => i + 1);
   412→    }
   413→    const parsed = parsePageRange(settings.rangeText, f.pageCount);
   414→    return parsed.valid ? parsed.pages : [];
   415→  }, [activeFile, settings.rangeMode, settings.rangeText]);
   416→
   417→  const rangeInvalid =
   418→    settings.rangeMode === "custom" &&
   419→    !parsePageRange(settings.rangeText, activeFile?.pageCount ?? 1).valid;
   420→
   421→  const selectedPagesRef = useRef<number[]>([]);
   422→  selectedPagesRef.current = selectedPages;
   423→
   424→  const photoReady =
   425→    photoPlacements.length > 0 &&
   426→    photoPlacements.every((p) => {
   427→      const it = files.find((f) => f.id === p.itemId);
   428→      return !!it && it.kind === "image" && it.status === "ready" && !!it.url;
   429→    });
   430→  const photoPreset = getPhotoPreset(photoSettings.presetId);
   431→  const photoSelected =
   432→    photoPlacements.find((p) => p.id === photoSelectedId) ?? null;
   433→
   434→  /**
   435→   * Max copies of the current preset that fit on one sheet (best of portrait
   436→   * / landscape orientation), clamped by the engine's placement cap.
   437→   */
   438→  const photoCapacity = useMemo(() => {
   439→    const w = photoPreset.group === "custom" ? photoSettings.customWMm : photoPreset.wMm;
   440→    const h = photoPreset.group === "custom" ? photoSettings.customHMm : photoPreset.hMm;
   441→    const margin = photoSettings.marginMm;
   442→    const gap = photoSettings.gapMm;
   443→    const printW = paperWmm - 2 * margin;
   444→    const printH = paperHmm - 2 * margin;
   445→    const capFor = (pw: number, ph: number) => {
   446→      if (pw > printW + PHOTO_TOL || ph > printH + PHOTO_TOL) return 0;
   447→      return (
   448→        Math.max(1, Math.floor((printW + gap) / (pw + gap))) *
   449→        Math.max(1, Math.floor((printH + gap) / (ph + gap)))
   450→      );
   451→    };
   452→    return Math.min(Math.max(capFor(w, h), capFor(h, w)), MAX_PLACEMENTS);
   453→  }, [photoPreset, photoSettings, paperWmm, paperHmm]);
   454→
   455→  const photoBeyondMargin = photoPlacements.some((p) => {
   456→    const bb = placementBBox(p);
   457→    return (
   458→      bb.x < photoSettings.marginMm - PHOTO_TOL ||
   459→      bb.y < photoSettings.marginMm - PHOTO_TOL ||
   460→      bb.x + bb.w > paperWmm - photoSettings.marginMm + PHOTO_TOL ||
   461→      bb.y + bb.h > paperHmm - photoSettings.marginMm + PHOTO_TOL
   462→    );
   463→  });
   464→
   465→  const canPrint = photoMode
   466→    ? photoReady && !printing
   467→    : !!activeFile &&
   468→      activeFile.status === "ready" &&
   469→      selectedPages.length > 0 &&
   470→      !printing;
   471→
   472→  // -- spooler detection ------------------------------------------------------
   473→  useEffect(() => {
   474→    let alive = true;
   475→    fetch("/api/print")
   476→      .then((r) => r.json())
   477→      .then((body: { spooler: { type: "ipp" | "lp"; printer: string | null } | null }) => {
   478→        if (!alive) return;
   479→        setSpooler(
   480→          body.spooler
   481→            ? { state: "available", info: body.spooler }
   482→            : { state: "unavailable", info: null },
   483→        );
   484→      })
   485→      .catch(() => alive && setSpooler({ state: "unavailable", info: null }));
   486→    return () => {
   487→      alive = false;
   488→    };
   489→  }, []);
   490→
   491→  // -- auto-fit to loaded tray ------------------------------------------------
   492→  const autoFitKey = useRef<string | null>(null);
   493→  useEffect(() => {
   494→    const f = activeFile;
   495→    if (!f || f.status !== "ready" || f.docWidthMm == null || f.continuous) return;
   496→    const key = `${f.id}:${settings.paperId}`;
   497→    if (autoFitKey.current === key) return;
   498→    const matched = matchPaper(f.docWidthMm, f.docHeightMm ?? 0);
   499→    if (matched !== settings.paperId) {
   500→      autoFitKey.current = key;
   501→      setSettings((s) => (s.scaleMode === "fit" ? s : { ...s, scaleMode: "fit" }));
   502→    }
   503→  }, [activeFile, settings.paperId]);
   504→
   505→  // -- file intake ------------------------------------------------------------
   506→  const patchFile = useCallback((id: string, patch: Partial<LoadedFile>) => {
   507→    setFiles((prev) => prev.map((f) => (f.id === id ? { ...f, ...patch } : f)));
   508→  }, []);
   509→
   510→  const addFiles = useCallback(
   511→    (incoming: File[]) => {
   512→      for (const file of incoming) {
   513→        const ext = (file.name.split(".").pop() ?? "").toLowerCase();
   514→        const accepted = ACCEPTED_EXT[ext];
   515→        if (!accepted) {
   516→          toast.error(`Format .${ext || "?"} tidak didukung.`, {
   517→            description: `${file.name} — gunakan PDF, DOC, DOCX, PNG, atau JPG.`,
   518→          });
   519→          continue;
   520→        }
   521→        if (file.size > MAX_FILE_BYTES) {
   522→          toast.error(`Ukuran file melebihi 50 MB.`, {
   523→            description: file.name,
   524→          });
   525→          continue;
   526→        }
   527→        const id = newId();
   528→        const loaded: LoadedFile = {
   529→          id,
   530→          name: file.name,
   531→          ext,
   532→          sizeBytes: file.size,
   533→          kind: accepted.kind,
   534→          mime: accepted.mime,
   535→          bytes: null,
   536→          url: null,
   537→          status: "loading",
   538→          pageCount: null,
   539→          docWidthMm: null,
   540→          docHeightMm: null,
   541→          docLabel: null,
   542→        };
   543→        setFiles((prev) => [...prev, loaded]);
   544→        setActiveId((cur) => cur ?? id);
   545→        setPage(1);
   546→
   547→        file
   548→          .arrayBuffer()
   549→          .then(async (buf) => {
   550→            const bytes = new Uint8Array(buf);
   551→            patchFile(id, { bytes });

   762→  /**
   763→   * Auto-grid arrange — the print-shop "ngejajar N foto" flow. Tiles N slots
   764→   * sized to the current preset (or explicit dims) in a balanced grid
   765→   * (6 → 3×2, 8 → 4×2, 9 → 3×3) centered inside the printable area;
   766→   * portrait / landscape is picked by whichever capacity fits N. Slots are
   767→   * filled round-robin with the distinct ready photos on the sheet; a
   768→   * hand-picked photo duplicates itself across the whole grid instead
   769→   * (unless ignoreSelection — used by preset changes & intake re-tiles).
   770→   */
   771→  const arrangePhotoCopies = useCallback(
   772→    (
   773→      count: number,
   774→      options?: {
   775→        dims?: { wMm: number; hMm: number };
   776→        auto?: boolean;
   777→        ignoreSelection?: boolean;
   778→        /** Explicit margin override — used when margin changes and re-tile
   779→         * happens in the same tick (borderless toggle), where the settings
   780→         * ref is still stale. */
   781→        marginMm?: number;
   782→      },
   783→    ) => {
   784→      const list = photoPlacementsRef.current;
   785→      if (list.length === 0) return;
   786→      const ps = photoSettingsRef.current;
   787→      const { wMm: paperW, hMm: paperH } = paperDimsRef.current;
   788→      const preset = getPhotoPreset(ps.presetId);
   789→      const presetW = preset.group === "custom" ? ps.customWMm : preset.wMm;
   790→      const presetH = preset.group === "custom" ? ps.customHMm : preset.hMm;
   791→      const pw = Math.min(options?.dims ? options.dims.wMm : presetW, paperW);
   792→      const ph = Math.min(options?.dims ? options.dims.hMm : presetH, paperH);
   793→      const margin = options?.marginMm ?? ps.marginMm;
   794→      const gap = ps.gapMm;
   795→      const printW = paperW - 2 * margin;
   796→      const printH = paperH - 2 * margin;
   797→
   798→      const fitFor = (w: number, h: number) => {
   799→        if (w > printW + PHOTO_TOL || h > printH + PHOTO_TOL) return null;
   800→        const cols = Math.max(1, Math.floor((printW + gap) / (w + gap)));
   801→        const rows = Math.max(1, Math.floor((printH + gap) / (h + gap)));
   802→        return { cols, rows, cap: cols * rows };
   803→      };
   804→
   805→      const portrait = fitFor(pw, ph);
   806→      const landscape = fitFor(ph, pw);
   807→      const n0 = Math.max(1, Math.round(count));
   808→      // Keep the preset's own orientation whenever it can fit N copies —
   809→      // rotating a pas-foto frame would crop faces. Only fall back to the
   810→      // rotated fit when portrait cannot hold N (or doesn't fit at all).
   811→      let useLandscape: boolean;
   812→      if (portrait && landscape) {
   813→        useLandscape =
   814→          portrait.cap >= n0
   815→            ? false
   816→            : landscape.cap >= n0
   817→              ? true
   818→              : landscape.cap > portrait.cap;
   819→      } else {
   820→        useLandscape = !portrait && !!landscape;
   821→      }
   822→      const fit = useLandscape ? landscape : portrait;
   823→      if (!fit) {
   824→        if (!options?.auto) {
   825→          toast.error("Foto melebihi area cetak lembar.", {
   826→            description:
   827→              "Kecilkan ukuran foto atau margin lembar, lalu susun ulang.",
   828→          });
   829→        }
   830→        return;
   831→      }
   832→      const total = Math.min(n0, fit.cap, MAX_PLACEMENTS);
   833→
   834→      // Balanced grid: fewest wasted slots first, then the most square block,
   835→      // preferring the wider option on ties (6 → 3×2, 9 → 3×3, 8 → 4×2).
   836→      let cols = fit.cols;
   837→      let bestScore = Infinity;
   838→      for (let c = 1; c <= fit.cols; c++) {
   839→        const r = Math.ceil(total / c);
   840→        if (r > fit.rows) continue;
   841→        const score = (c * r - total) * 1000 + Math.abs(c - r);
   842→        if (score < bestScore || (score === bestScore && c > cols)) {
   843→          bestScore = score;
   844→          cols = c;
   845→        }
   846→      }
   847→      const w = useLandscape ? ph : pw;
   848→      const h = useLandscape ? pw : ph;
   849→      const rows = Math.ceil(total / cols);
   850→
   851→      // Slot sources: a hand-picked photo duplicates itself; otherwise the
   852→      // distinct ready images already on the sheet fill slots round-robin.
   853→      let sources: string[] = [];
   854→      const sel = options?.ignoreSelection
   855→        ? null
   856→        : list.find((x) => x.id === photoSelectedIdRef.current);
   857→      if (sel) {
   858→        sources = [sel.itemId];
   859→      } else {
   860→        const seen = new Set<string>();
   861→        for (const p of list) {
   862→          if (seen.has(p.itemId)) continue;
   863→          const it = filesRef.current.find(
   864→            (f) =>
   865→              f.id === p.itemId &&
   866→              f.kind === "image" &&
   867→              f.status === "ready" &&
   868→              !!f.url,
   869→          );
   870→          if (it) {
   871→            seen.add(p.itemId);
   872→            sources.push(p.itemId);
   873→          }
   874→        }
   875→      }
   876→      if (sources.length === 0) sources = [list[0].itemId];
   877→
   878→      const gy = (printH - (rows * h + (rows - 1) * gap)) / 2;
   879→      const out: PhotoPlacement[] = [];
   880→      for (let r = 0; r < rows; r++) {
   881→        const inRow = Math.min(total - r * cols, cols);
   882→        const rowW = inRow * w + (inRow - 1) * gap;
   883→        const gx = (printW - rowW) / 2;
   884→        for (let c = 0; c < inRow; c++) {
   885→          const itemId = sources[(r * cols + c) % sources.length];
   886→          const src = list.find((p) => p.itemId === itemId);
   887→          out.push({
   888→            id: newPhotoId(),
   889→            itemId,
   890→            xMm: margin + gx + c * (w + gap),
   891→            yMm: margin + gy + r * (h + gap),
   892→            wMm: w,
   893→            hMm: h,
   894→            rotation: 0,
   895→            fit: src ? src.fit : "cover",
   896→            cropX: src?.cropX ?? 0,
   897→            cropY: src?.cropY ?? 0,
   898→          });
   899→        }
   900→      }
   901→      setPhotoPlacements(out);
   902→      setPhotoSelectedId(null);
   903→      layoutTouchedRef.current = false;
   904→      if (sources.length > total) {
   905→        toast.info(
   906→          `${total} foto tersusun dari ${sources.length} foto yang diunggah`,
   907→          {
   908→            description: `Naikkan jumlah foto ke ${sources.length} agar semua foto ikut tersusun.`,
   909→          },
   910→        );
   911→        return;
   912→      }
   913→      if (n0 > fit.cap) {
   914→        toast.info(`Lembar penuh — maksimal ${fit.cap} foto ukuran ini`, {
   915→          description:
   916→            "Kurangi jumlah foto, pilih preset lebih kecil, atau ganti kertas lebih besar.",
   917→        });
   918→        return;
   919→      }
   920→      if (!options?.auto || out.length > 1) {
   921→        toast.success(`${out.length} foto tersusun otomatis`, {
   922→          description: `${useLandscape ? "Lanskap" : "Potret"} • ${cols} kolom × ${rows} baris • jarak ${gap} mm`,
   923→        });
   924→      }
   925→    },
   926→    [],
   927→  );
   928→
   929→  /**
   930→   * Debounced intake arrange: uploads become ready one file at a time, so the
   931→   * re-tile waits for a short quiet window before regenerating the grid with
   932→   * every photo from the batch.
   933→   */
   934→  const scheduleIntakeArrange = useCallback(() => {

   950→   * Borderless toggle — zeroes / restores the sheet margin and re-tiles the

   979→  /** Apply a photo size preset to the selected placement (or all when none). */

  1023→          return clampPlacement(
  1024→            { ...p, wMm: w, hMm: h, xMm: cx - w / 2, yMm: cy - h / 2 },

  1029→      );
  1030→    },

  1034→  /** Fill the whole sheet with copies of the selected photo (pas-foto sheets). */
  1035→  const fillPhotoSheet = useCallback(() => {
  1036→    const list = photoPlacementsRef.current;
  1037→    if (list.length === 0) return;
  1038→    layoutTouchedRef.current = true;
  1039→    const selId = photoSelectedIdRef.current;
  1040→    const p = list.find((x) => x.id === selId) ?? list[0];
  1041→    const ps = photoSettingsRef.current;
  1042→    const { wMm: paperW, hMm: paperH } = paperDimsRef.current;
  1043→    const margin = ps.marginMm;
  1044→    const gap = ps.gapMm;
  1045→    const printW = paperW - 2 * margin;
  1046→    const printH = paperH - 2 * margin;
  1047→    const bb = placementBBox(p);
  1048→    if (bb.w > printW + PHOTO_TOL || bb.h > printH + PHOTO_TOL) {
  1049→      toast.error("Foto lebih besar dari area cetak.", {
  1050→        description: "Perkecil ukuran foto atau margin lembar terlebih dahulu.",
  1051→      });
  1052→      return;
  1053→    }
  1054→    const cols = Math.max(1, Math.floor((printW + gap) / (bb.w + gap)));
  1055→    const rowsBase = Math.max(1, Math.floor((printH + gap) / (bb.h + gap)));
  1056→    const rows =
  1057→      cols * rowsBase > MAX_PLACEMENTS
  1058→        ? Math.max(1, Math.floor(MAX_PLACEMENTS / cols))
  1059→        : rowsBase;
  1060→    const gx = (printW - (cols * bb.w + (cols - 1) * gap)) / 2;
  1061→    const gy = (printH - (rows * bb.h + (rows - 1) * gap)) / 2;
  1062→    const out: PhotoPlacement[] = [];
  1063→    for (let r = 0; r < rows; r++) {
  1064→      for (let c = 0; c < cols; c++) {
  1065→        const vx = margin + gx + c * (bb.w + gap);

  1068→          id: newPhotoId(),
  1069→          itemId: p.itemId,

  1086→  /** Tidy the sheet: pack every placement into rows within the printable area. */
  1087→  const autoArrangePhotos = useCallback(() => {

  1091→    const ps = photoSettingsRef.current;
  1092→    const { wMm: paperW, hMm: paperH } = paperDimsRef.current;

  1094→    const gap = ps.gapMm;
  1095→    const printW = paperW - 2 * margin;

  1097→    let y = margin;
  1098→    let rowH = 0;

  1100→      const bb = placementBBox(p);
  1101→      const w = Math.min(bb.w, printW);
  1102→      if (x > margin && x + w > margin + printW + PHOTO_TOL) {
  1103→        x = margin;
  1104→        y += rowH + gap;
  1105→        rowH = 0;
  1106→      }
  1107→      const cx = x + bb.w / 2;
  1108→      const cy = y + bb.h / 2;
  1109→      const np = { ...p, xMm: cx - p.wMm / 2, yMm: cy - p.hMm / 2 };
  1110→      x += bb.w + gap;
  1111→      rowH = Math.max(rowH, bb.h);
  1112→      return np;
  1113→    });
  1114→    setPhotoPlacements(out.map((pl) => clampPlacement(pl, paperW, paperH)));
  1115→  }, []);

  1126→    let cx = p.xMm + p.wMm / 2;
  1127→    let cy = p.yMm + p.hMm / 2;

  1129→    if (dir === "centerX") cx = paperW / 2;
  1130→    if (dir === "right") cx = paperW - ps.marginMm - bb.w / 2;
  1131→    if (dir === "top") cy = ps.marginMm + bb.h / 2;

  1299→      } finally {
  1300→        setPrinting(false);
  1301→      }

  1303→    }

  1305→    const s = settingsRef.current;
  1306→    const pages = selectedPagesRef.current;

  1327→      });
