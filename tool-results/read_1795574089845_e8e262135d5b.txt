     4→     4→// Free drag + 8-handle resize on a true-to-scale paper canvas, with smart
     5→     5→// snapping to the printable margins, page center and sibling photo edges.

    96→    96→/** Visual bounding box of a placement (accounts for 90-step rotation). */
    97→    97→function bboxOf(p: PhotoPlacement): { x: number; y: number; w: number; h: number } {
    98→    98→  const rot = ((Math.round(p.rotation / 90) * 90) % 360 + 360) % 360;
    99→    99→  const swapped = rot === 90 || rot === 270;
   100→   100→  const w = swapped ? p.hMm : p.wMm;
   101→   101→  const h = swapped ? p.wMm : p.hMm;
   102→   102→  const cx = p.xMm + p.wMm / 2;
   103→   103→  const cy = p.yMm + p.hMm / 2;
   104→   104→  return { x: cx - w / 2, y: cy - h / 2, w, h };
   105→   105→}

   107→   107→/** Resize with center-based rotation: the opposite anchor stays fixed. */
   108→   108→function resizePlacement(
   109→   109→  orig: PhotoPlacement,
   110→   110→  dir: HandleDir,
   111→   111→  dxMm: number,
   112→   112→  dyMm: number,
   113→   113→  lockAspect: boolean,
   114→   114→): { xMm: number; yMm: number; wMm: number; hMm: number } {
   115→   115→  const theta = (orig.rotation * Math.PI) / 180;
   116→   116→  const c = Math.cos(theta);
   117→   117→  const s = Math.sin(theta);
   118→   118→  // screen delta -> placement-local delta (rotate by -theta)
   119→   119→  const lx = c * dxMm + s * dyMm;
   120→   120→  const ly = -s * dxMm + c * dyMm;
   121→   121→
   122→   122→  let dw = 0;
   123→   123→  let dh = 0;
   124→   124→  if (dir.includes("e")) dw = lx;
   125→   125→  if (dir.includes("w")) dw = -lx;
   126→   126→  if (dir.includes("s")) dh = ly;
   127→   127→  if (dir.includes("n")) dh = -ly;
   128→   128→
   129→   129→  const ratio = orig.hMm / orig.wMm;
   130→   130→  if (lockAspect && dw !== 0 && dh !== 0) {
   131→   131→    if (Math.abs(dw) / orig.wMm > Math.abs(dh) / orig.hMm) dh = dw * ratio;
   132→   132→    else dw = dh / ratio;
   133→   133→  }
   134→   134→  let w2 = Math.max(MIN_PHOTO_MM, orig.wMm + dw);
   135→   135→  let h2 = Math.max(MIN_PHOTO_MM, orig.hMm + dh);
   136→   136→  if (lockAspect && dw !== 0 && dh !== 0) {
   137→   137→    if (Math.abs(dw) / orig.wMm >= Math.abs(dh) / orig.hMm) h2 = w2 * ratio;
   138→   138→    else w2 = h2 / ratio;
   139→   139→  }
   140→   140→
   141→   141→  // anchor = opposite handle point in local (center-relative) coords
   142→   142→  const fx = dir.includes("w") ? orig.wMm / 2 : dir.includes("e") ? -orig.wMm / 2 : 0;
   143→   143→  const fy = dir.includes("n") ? orig.hMm / 2 : dir.includes("s") ? -orig.hMm / 2 : 0;
   144→   144→  const cx0 = orig.xMm + orig.wMm / 2;
   145→   145→  const cy0 = orig.yMm + orig.hMm / 2;
   146→   146→  const fpx = cx0 + (c * fx - s * fy);
   147→   147→  const fpy = cy0 + (s * fx + c * fy);
   148→   148→  const nfx = dir.includes("w") ? w2 / 2 : dir.includes("e") ? -w2 / 2 : 0;
   149→   149→  const nfy = dir.includes("n") ? h2 / 2 : dir.includes("s") ? -h2 / 2 : 0;
   150→   150→  const ncx = fpx - (c * nfx - s * nfy);
   151→   151→  const ncy = fpy - (s * nfx + c * nfy);
   152→   152→  return { xMm: ncx - w2 / 2, yMm: ncy - h2 / 2, wMm: w2, hMm: h2 };
   153→   153→}

   155→   155→function attachWindowListeners(
   156→   156→  onMove: (ev: PointerEvent) => void,
   157→   157→  onUp: (ev?: Event) => void,
   158→   158→): void {
   159→   159→  window.addEventListener("pointermove", onMove);
   160→   160→  window.addEventListener("pointerup", onUp);
   161→   161→  window.addEventListener("pointercancel", onUp);
   162→   162→  window.addEventListener("blur", onUp);
   163→   163→}
   164→   164→
   165→   165→function detachWindowListeners(
   166→   166→  onMove: (ev: PointerEvent) => void,
   167→   167→  onUp: (ev?: Event) => void,
   168→   168→): void {
   169→   169→  window.removeEventListener("pointermove", onMove);
   170→   170→  window.removeEventListener("pointerup", onUp);
   171→   171→  window.removeEventListener("pointercancel", onUp);
   172→   172→  window.removeEventListener("blur", onUp);
   173→   173→}

   175→   175→export function PhotoCanvas(props: PhotoCanvasProps) {
   176→   176→  const {
   177→   177→    paperWmm,
   178→   178→    paperHmm,
   179→   179→    k,
   180→   180→    placements,
   181→   181→    files,
   182→   182→    selectedIds,
   183→   183→    marginMm,
   184→   184→    lockAspect,
   185→   185→    cutLines,
   186→   186→    grayscale,
   187→   187→    adjust,
   188→   188→    multiMode,
   189→   189→    onSelect,
   190→   190→    onSelectIds,
   191→   191→    onSelectAll,
   192→   192→    onUpdateMany,
   193→   193→    onRemove,
   194→   194→    onDuplicate,
   195→   195→    onChangePhoto,
   196→   196→    onCropPhoto,
   197→   197→    onCoverPhoto,
   198→   198→  } = props;
   199→   199→
   200→   200→  const gestureRef = useRef<Gesture | null>(null);
   201→   201→  const placementsRef = useRef(placements);
   202→   202→  const kRef = useRef(k);
   203→   203→  const lockRef = useRef(lockAspect);
   204→   204→  const paperRef = useRef({ wMm: paperWmm, hMm: paperHmm });
   205→   205→  const selectedIdsRef = useRef(selectedIds);
   206→   206→  const multiModeRef = useRef(multiMode);
   207→   207→  // Sync latest props into refs after commit — gesture/keyboard handlers read
   208→   208→  // them at event time (never during render).
   209→   209→  useEffect(() => {
   210→   210→    placementsRef.current = placements;
   211→   211→    kRef.current = k;
   212→   212→    lockRef.current = lockAspect;
   213→   213→    paperRef.current = { wMm: paperWmm, hMm: paperHmm };
   214→   214→    selectedIdsRef.current = selectedIds;
   215→   215→    multiModeRef.current = multiMode;
   216→   216→  }, [placements, k, lockAspect, paperWmm, paperHmm, selectedIds, multiMode]);
   217→   217→  const [guides, setGuides] = useState<{ v: number | null; h: number | null }>({
   218→   218→    v: null,
   219→   219→    h: null,
   220→   220→  });
   221→   221→  const [marquee, setMarquee] = useState<{
   222→   222→    x: number;
   223→   223→    y: number;
   224→   224→    w: number;
   225→   225→    h: number;
   226→   226→  } | null>(null);
   227→   227→
   228→   228→  const primaryId = selectedIds.length > 0 ? selectedIds[selectedIds.length - 1] : null;
   229→   229→
   230→   230→  const sheetW = Math.round(paperWmm * k);
   231→   231→  const sheetH = Math.round(paperHmm * k);
   232→   232→
   233→   233→  // -- pointer gesture engine ------------------------------------------------
   234→   234→  const endGesture = useCallback(() => {
   235→   235→    gestureRef.current = null;
   236→   236→    setGuides({ v: null, h: null });
   237→   237→    setMarquee(null);
   238→   238→  }, []);
   239→   239→
   240→   240→  /** Group move: every id in `ids` translates by the same mm delta. */
   241→   241→  const beginMoveGesture = useCallback(
   242→   242→    (
   243→   243→      ids: string[],
   244→   244→      grabId: string,
   245→   245→      e: React.PointerEvent,
   246→   246→      tapToggleId: string | null,
   247→   247→    ) => {
   248→   248→      e.stopPropagation();
   249→   249→      e.preventDefault();
   250→   250→      const origins = new Map<string, { x: number; y: number }>();
   251→   251→      for (const p of placementsRef.current) {
   252→   252→        if (ids.includes(p.id)) origins.set(p.id, { x: p.xMm, y: p.yMm });
   253→   253→      }
   254→   254→      gestureRef.current = {
   255→   255→        kind: "move",
   256→   256→        ids,
   257→   257→        grabId,
   258→   258→        px0: e.clientX,
   259→   259→        py0: e.clientY,
   260→   260→        origins,
   261→   261→        tapToggleId,
   262→   262→        movedPx: 0,
   263→   263→      };
   264→   264→      const onMove = (ev: PointerEvent) => {
   265→   265→        const cur = gestureRef.current;
   266→   266→        if (!cur || cur.kind !== "move") return;
   267→   267→        const dxPx = ev.clientX - cur.px0;
   268→   268→        const dyPx = ev.clientY - cur.py0;
   269→   269→        cur.movedPx = Math.max(cur.movedPx, Math.abs(dxPx), Math.abs(dyPx));
   270→   270→        const scale = kRef.current || 1;
   271→   271→        let dx = dxPx / scale;
   272→   272→        let dy = dyPx / scale;
   273→   273→        // smart snapping (Word-like guides) — anchored on the grabbed photo
   274→   274→        const { wMm: paperW, hMm: paperH } = paperRef.current;
   275→   275→        const tolerance = 6 / scale;
   276→   276→        const me = placementsRef.current.find((p) => p.id === cur.grabId);
   277→   277→        const grabOrigin = cur.origins.get(cur.grabId);
   278→   278→        if (me && grabOrigin) {
   279→   279→          const moved = { ...me, xMm: grabOrigin.x + dx, yMm: grabOrigin.y + dy };
   280→   280→          const bb = bboxOf(moved);
   281→   281→          const tx: number[] = [0, marginMm, paperW - marginMm, paperW / 2, paperW];
   282→   282→          const ty: number[] = [0, marginMm, paperH - marginMm, paperH / 2, paperH];
   283→   283→          for (const o of placementsRef.current) {
   284→   284→            if (cur.ids.includes(o.id)) continue;
   285→   285→            const ob = bboxOf(o);
   286→   286→            tx.push(ob.x, ob.x + ob.w, ob.x + ob.w / 2);
   287→   287→            ty.push(ob.y, ob.y + ob.h, ob.y + ob.h / 2);
   288→   288→          }
   289→   289→          let bestX: { d: number; t: number } | null = null;
   290→   290→          let bestY: { d: number; t: number } | null = null;
   291→   291→          for (const t of tx) {
   292→   292→            for (const e2 of [bb.x, bb.x + bb.w, bb.x + bb.w / 2]) {
   293→   293→              const d = t - e2;
   294→   294→              if (Math.abs(d) <= tolerance && (!bestX || Math.abs(d) < Math.abs(bestX.d))) {
   295→   295→                bestX = { d, t };
   296→   296→              }
   297→   297→            }
   298→   298→          }
   299→   299→          for (const t of ty) {
   300→   300→            for (const e2 of [bb.y, bb.y + bb.h, bb.y + bb.h / 2]) {
   301→   301→              const d = t - e2;
   302→   302→              if (Math.abs(d) <= tolerance && (!bestY || Math.abs(d) < Math.abs(bestY.d))) {
   303→   303→                bestY = { d, t };
   304→   304→              }
   305→   305→            }
   306→   306→          }
   307→   307→          if (bestX) dx += bestX.d;
   308→   308→          if (bestY) dy += bestY.d;
   309→   309→          setGuides({ v: bestX ? bestX.t : null, h: bestY ? bestY.t : null });
   310→   310→        }
   311→   311→        onUpdateMany(
   312→   312→          cur.ids
   313→   313→            .filter((id) => cur.origins.has(id))
   314→   314→            .map((id) => {
   315→   315→              const o = cur.origins.get(id)!;
   316→   316→              return { id, patch: { xMm: o.x + dx, yMm: o.y + dy } };
   317→   317→            }),
   318→   318→        );
   319→   319→      };
   320→   320→      const onUp = () => {
   321→   321→        detachWindowListeners(onMove, onUp);
   322→   322→        const cur = gestureRef.current;
   323→   323→        if (
   324→   324→          cur &&
   325→   325→          cur.kind === "move" &&
   326→   326→          cur.tapToggleId &&
   327→   327→          cur.movedPx < 4
   328→   328→        ) {
   329→   329→          // touch "Pilih Banyak": a tap (no drag) toggles the photo in
   330→   330→          onSelect(cur.tapToggleId, true);
   331→   331→        }
   332→   332→        endGesture();
   333→   333→      };
   334→   334→      attachWindowListeners(onMove, onUp);
   335→   335→    },
   336→   336→    [onUpdateMany, onSelect, marginMm, endGesture],
   337→   337→  );
   338→   338→
   339→   339→  const beginMove = useCallback(
   340→   340→    (p: PhotoPlacement, e: React.PointerEvent) => {
   341→   341→      const additive = e.shiftKey || e.ctrlKey || e.metaKey;
   342→   342→      if (additive) {
   343→   343→        // Ctrl/Shift+click toggles into the multi-selection (Word-like);
   344→   344→        // no drag so building a group never accidentally moves photos.
   345→   345→        e.stopPropagation();
   346→   346→        e.preventDefault();
   347→   347→        onSelect(p.id, true);
   348→   348→        return;
   349→   349→      }
   350→   350→      const sel = selectedIdsRef.current;
   351→   351→      if (multiModeRef.current) {
   352→   352→        if (sel.includes(p.id)) {
   353→   353→          beginMoveGesture([...sel], p.id, e, null);
   354→   354→        } else {
   355→   355→          // tap adds it to the selection, drag moves just this photo
   356→   356→          beginMoveGesture([p.id], p.id, e, p.id);
   357→   357→        }
   358→   358→        return;
   359→   359→      }
   360→   360→      if (sel.includes(p.id)) {
   361→   361→        // already in the (possibly multi) selection — drag moves the group
   362→   362→        beginMoveGesture([...sel], p.id, e, null);
   363→   363→      } else {
   364→   364→        onSelect(p.id);
   365→   365→        beginMoveGesture([p.id], p.id, e, null);
   366→   366→      }
   367→   367→    },
   368→   368→    [beginMoveGesture, onSelect],
   369→   369→  );
   370→   370→
   371→   371→  const beginResize = useCallback(
   372→   372→    (p: PhotoPlacement, dir: HandleDir, e: React.PointerEvent) => {
   373→   373→      e.stopPropagation();
   374→   374→      e.preventDefault();
   375→   375→      gestureRef.current = {
   376→   376→        kind: "resize",
   377→   377→        id: p.id,
   378→   378→        dir,
   379→   379→        px0: e.clientX,
   380→   380→        py0: e.clientY,
   381→   381→        orig: p,
   382→   382→      };
   383→   383→      const onMove = (ev: PointerEvent) => {
   384→   384→        const cur = gestureRef.current;
   385→   385→        if (!cur || cur.kind !== "resize") return;
   386→   386→        const scale = kRef.current || 1;
   387→   387→        const dx = (ev.clientX - cur.px0) / scale;
   388→   388→        const dy = (ev.clientY - cur.py0) / scale;
   389→   389→        const r = resizePlacement(cur.orig, cur.dir, dx, dy, lockRef.current);
   390→   390→        onUpdateMany([{ id: cur.id, patch: r }]);
   391→   391→      };
   392→   392→      const onUp = () => {
   393→   393→        detachWindowListeners(onMove, onUp);
   394→   394→        endGesture();
   395→   395→      };
   396→   396→      attachWindowListeners(onMove, onUp);
   397→   397→    },
   398→   398→    [onUpdateMany, endGesture],
   399→   399→  );
   400→   400→
   401→   401→  // -- marquee lasso on the empty sheet --------------------------------------
   402→   402→  const onSheetPointerDown = useCallback(
   403→   403→    (e: React.PointerEvent<HTMLDivElement>) => {
   404→   404→      if (e.target !== e.currentTarget) return;
   405→   405→      if (e.pointerType === "mouse" && e.button !== 0) return;
   406→   406→      const rect = e.currentTarget.getBoundingClientRect();
   407→   407→      const scale = kRef.current || 1;
   408→   408→      const x0 = (e.clientX - rect.left) / scale;
   409→   409→      const y0 = (e.clientY - rect.top) / scale;
   410→   410→      gestureRef.current = {
   411→   411→        kind: "marquee",
   412→   412→        rect,
   413→   413→        x0,
   414→   414→        y0,
   415→   415→        px0: e.clientX,
   416→   416→        py0: e.clientY,
   417→   417→        moved: false,
   418→   418→      };
   419→   419→      const onMove = (ev: PointerEvent) => {
   420→   420→        const cur = gestureRef.current;
   421→   421→        if (!cur || cur.kind !== "marquee") return;
   422→   422→        if (
   423→   423→          Math.abs(ev.clientX - cur.px0) > 4 ||
   424→   424→          Math.abs(ev.clientY - cur.py0) > 4
   425→   425→        ) {
   426→   426→          cur.moved = true;
   427→   427→        }
   428→   428→        const x = (ev.clientX - cur.rect.left) / scale;
   429→   429→        const y = (ev.clientY - cur.rect.top) / scale;
   430→   430→        setMarquee({
   431→   431→          x: Math.min(x, cur.x0),
   432→   432→          y: Math.min(y, cur.y0),
   433→   433→          w: Math.abs(x - cur.x0),
   434→   434→          h: Math.abs(y - cur.y0),
   435→   435→        });
   436→   436→      };
   437→   437→      const onUp = (ev?: Event) => {
   438→   438→        detachWindowListeners(onMove, onUp);
   439→   439→        const cur = gestureRef.current;
   440→   440→        if (cur && cur.kind === "marquee" && cur.moved && ev) {
