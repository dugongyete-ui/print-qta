    86→function NumField({
    87→  label,
    88→  value,
    89→  onCommit,
    90→  min,
    91→  max,
    92→  disabled,
    93→}: {
    94→  label: string;
    95→  value: number;
    96→  onCommit: (v: number) => void;
    97→  min: number;
    98→  max: number;
    99→  disabled?: boolean;
   100→}) {
   101→  const fmt = (v: number) => String(Math.round(v * 10) / 10);
   102→  const commit = (raw: string) => {
   103→    const parsed = Number(raw.replace(",", "."));
   104→    if (!Number.isFinite(parsed)) return;
   105→    onCommit(Math.min(Math.max(parsed, min), max));
   106→  };
   107→  // Uncontrolled input keyed by the external value: dragging / resizing on the
   108→  // canvas remounts it with fresh numbers, while local typing stays untouched.
   109→  return (
   110→    <label className="flex min-w-0 flex-1 flex-col gap-1">
   111→      <span className="font-mono text-[9.5px] uppercase tracking-[0.06em] text-[#6B7280]">
   112→        {label}
   113→      </span>
   114→      <input
   115→        key={`${label}-${fmt(value)}`}
   116→        type="number"
   117→        inputMode="decimal"
   118→        step="0.5"
   119→        defaultValue={fmt(value)}
   120→        disabled={disabled}
   121→        onBlur={(e) => commit(e.target.value)}
   122→        onKeyDown={(e) => {
   123→          if (e.key === "Enter") {
   124→            e.preventDefault();
   125→            (e.target as HTMLInputElement).blur();
   126→          }
   127→        }}
   128→        aria-label={label}
   129→        className="h-8 w-full rounded-md border border-[#E5E7EB] bg-white px-2 font-mono text-[11.5px] tabular-nums text-[#111827] outline-none transition-colors focus:border-[#003399] disabled:opacity-40"
   130→      />
   131→    </label>
   132→  );
   133→}

   258→  onBorderless,

   263→  onArrangeCount,
   264→  onArrange,

   273→  const capSafe = Math.max(1, Math.min(capacity, 200));

   275→  /** Re-tile immediately — bundling is automatic for every sheet control. */
   276→  const setCount = (next: number) => {
   277→    const v = Math.max(1, Math.min(next, capSafe));
   278→    onSetting({ perSheet: v });
   279→    onArrangeCount(v);
   280→  };
   281→  const setAlignV = (v: GridAlignV) => {
   282→    onSetting({ alignV: v });

   284→  };
   285→  const setAlignH = (v: GridAlignH) => {
   286→    onSetting({ alignH: v });
   287→    onArrangeCount(n, { alignH: v });
   288→  };
   289→  const setMargin = (v: number) => {
   290→    onSetting({ marginMm: v });
   291→    onArrangeCount(n, { marginMm: v });
   292→  };

   294→    onSetting({ gapMm: v });
   295→    onArrangeCount(n, { gapMm: v });
   296→  };

   298→  const unplacedImages = files.filter(
   299→    (f) => f.kind === "image" && f.status === "ready" && !!f.url && !placedIds.has(f.id),
   300→  );
   301→
   302→  const setAdjust = (patch: Partial<PhotoAdjustments>) =>
   303→    onSetting({ adjust: { ...settings.adjust, ...patch } });
   304→  const pctFmt = (v: number) => `${Math.round(v * 100)}%`;
   305→
   306→  const scopeLabel = selected ? "menerapkan ke foto terpilih" : "menerapkan ke semua foto";

   336→                ))}

   338→            ))}

   340→        </Select>

   342→          {PRESET_CHIPS.map((c) => {
   343→            const active = preset.id === c.id;

   346→                key={c.id}
   347→                type="button"
   348→                aria-pressed={active}

   350→                className={`h-7 rounded-md border px-2 font-mono text-[11px] font-medium tabular-nums transition-colors ${
   351→                  active
   352→                    ? "border-[#003399] bg-[#003399] text-white"
   353→                    : "border-[#E5E7EB] bg-white text-[#374151] hover:bg-[#F9FAFB]"
   354→                }`}
   355→              >
   356→                {c.label}
   357→              </button>
   358→            );
   359→          })}
   360→        </div>
   361→        <p className="font-mono text-[10px] tabular-nums text-[#6B7280]">
   362→          {presetDimsLabel(preset)} • {scopeLabel}
   363→        </p>
   364→
   365→        {preset.group === "custom" ? (
   366→          <div className="flex gap-2">
   367→            <NumField
   368→              label="Lebar (mm)"
   369→              value={settings.customWMm}

   378→              max={paperHmm}

   388→            aria-label="Kunci rasio aspek foto"
   389→            className="data-[state=checked]:bg-[#003399]"
   390→          />
   391→        </Row>
   392→      </Section>
   393→
   394→      {/* Posisi & Ukuran (Word-like) */}
   395→      <Section
   396→        title="Posisi & Ukuran Foto"
   397→        className={selected ? "" : "opacity-60"}
   398→      >
   399→        {selected ? (
   400→          <>
   401→            <div className="flex gap-2">
   402→              <NumField
   403→                label="Posisi X"
   404→                value={selected.xMm}
   405→                min={0}
   406→                max={paperWmm}
   407→                onCommit={(v) => onUpdateSelected({ xMm: v })}
   408→              />
   409→              <NumField
   410→                label="Posisi Y"
   411→                value={selected.yMm}
   412→                min={0}
   413→                max={paperHmm}
   414→                onCommit={(v) => onUpdateSelected({ yMm: v })}
   415→              />
   416→            </div>
   417→            <div className="flex gap-2">
   418→              <NumField
   419→                label="Lebar"
   420→                value={selected.wMm}
   421→                min={10}
   422→                max={paperWmm}
   423→                onCommit={(v) =>
   424→                  onUpdateSelected(
   425→                    settings.lockAspect
   426→                      ? {
   427→                          wMm: v,
   428→                          hMm: (v * selected.hMm) / selected.wMm,

   430→                      : { wMm: v },
   431→                  )
   432→                }
   433→              />
   434→              <NumField
   435→                label="Tinggi"
   436→                value={selected.hMm}
   437→                min={10}
   438→                max={paperHmm}
   439→                onCommit={(v) =>
   440→                  onUpdateSelected(
   441→                    settings.lockAspect
   442→                      ? {
   443→                          hMm: v,
   444→                          wMm: (v * selected.wMm) / selected.hMm,
   445→                        }
   446→                      : { hMm: v },
   447→                  )
   448→                }
   449→              />
   450→            </div>
   451→            <Row label="Rotasi">

   572→          min={0.5}

   574→          step={0.01}
   575→          onChange={(v) => setAdjust({ contrast: v })}

   577→        />
