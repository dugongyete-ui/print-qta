    92→    92→function AutoFitBadge({
    93→    93→  mismatch,
    94→    94→  overflow,
    95→    95→  docLabel,
    96→    96→  paperLabel,
    97→    97→  scaleMode,
    98→    98→}: {
    99→    99→  mismatch: boolean;
   100→   100→  overflow: boolean;
   101→   101→  docLabel: string;
   102→   102→  paperLabel: string;
   103→   103→  scaleMode: "fit" | "original";
   104→   104→}) {
   105→   105→  const warn = mismatch || overflow;
   106→   106→  const text =
   107→   107→    overflow && !mismatch
   108→   108→      ? `Asli: ${docLabel} — melebihi ${paperLabel}`
   109→   109→      : mismatch && scaleMode === "fit"
   110→   110→        ? `Asli: ${docLabel} → Penyesuaian Otomatis: ${paperLabel}`
   111→   111→        : mismatch
   112→   112→          ? `Asli: ${docLabel} — tidak sesuai tray ${paperLabel}`
   113→   113→          : `Asli: ${docLabel} — Sesuai Tray`;
   114→   114→  return (
   115→   115→    <div
   116→   116→      className={`flex h-7 items-center gap-1.5 whitespace-nowrap rounded-full border px-3 font-mono text-[10.5px] tabular-nums ${
   117→   117→        warn
   118→   118→          ? "border-[#FDE68A] bg-[#FFFBEB] text-[#92400E]"
   119→   119→          : "border-[#E5E7EB] bg-white text-[#6B7280]"
   120→   120→      }`}
   121→   121→      role="status"
   122→   122→    >
   123→   123→      {warn ? <TriangleAlert size={11} strokeWidth={1.5} /> : null}
   124→   124→      {text}
   125→   125→    </div>
   126→   126→  );
   127→   127→}

   129→   129→export function PreviewStage(props: PreviewStageProps) {
   130→   130→  const {
   131→   131→    file,
   132→   132→    paperWmm,
   133→   133→    paperHmm,
   134→   134→    scale,
   135→   135→    grayscale,
   136→   136→    showMargins,
   137→   137→    zoom,
   138→   138→    page,
   139→   139→    pageCount,
   140→   140→    continuous,
   141→   141→    mismatch,
   142→   142→    overflow,
   143→   143→    scaleMode,
   144→   144→    onFiles,
   145→   145→    onSamples,
   146→   146→    onSelect,
   147→   147→    onRemove,
   148→   148→    onPageChange,
   149→   149→    onZoomChange,
   150→   150→    onToggleMargins,
   151→   151→    onEstimatePages,
   152→   152→    getPdfDoc,
   153→   153→    getDocxRender,
   154→   154→    getDocHtml,
   155→   155→    duplexAssist,
   156→   156→    onPrintEven,
   157→   157→    onCancelDuplex,
   158→   158→  } = props;
   159→   159→
   160→   160→  const scrollRef = useRef<HTMLDivElement>(null);
   161→   161→  const canvasRef = useRef<HTMLCanvasElement>(null);
   162→   162→  const renderTaskRef = useRef<{ cancel: () => void } | null>(null);
   163→   163→  const docxHostRef = useRef<HTMLDivElement>(null);
   164→   164→  const flowHostRef = useRef<HTMLDivElement>(null);
   165→   165→  const [box, setBox] = useState({ w: 0, h: 0 });
   166→   166→  const [dragging, setDragging] = useState(false);
   167→   167→  const [docxRender, setDocxRender] = useState<DocxRender | null>(null);
   168→   168→  /** Touch-friendly multi-select mode: tap photos one by one to add. */
   169→   169→  const [photoMulti, setPhotoMulti] = useState(false);
   170→   170→
   171→   171→  // -- stage calibration: px per mm ------------------------------------------
   172→   172→  useEffect(() => {
   173→   173→    const el = scrollRef.current;
   174→   174→    if (!el) return;
   175→   175→    const ro = new ResizeObserver((entries) => {
   176→   176→      const r = entries[0].contentRect;
   177→   177→      setBox({ w: r.width, h: r.height });
   178→   178→    });
   179→   179→    ro.observe(el);
   180→   180→    return () => ro.disconnect();
   181→   181→  }, []);
   182→   182→
   183→   183→  const k = useMemo(() => {
   184→   184→    if (zoom === "100") return MM_TO_PX96;
   185→   185→    if (box.w < 40 || box.h < 40) return 0;
   186→   186→    const fit = Math.min((box.w - 72) / paperWmm, (box.h - 88) / paperHmm);
   187→   187→    return Math.max(0.4, Math.min(fit, 7));
   188→   188→  }, [zoom, box, paperWmm, paperHmm]);
   189→   189→
   190→   190→  const sheetW = Math.round(paperWmm * k);
   191→   191→  const sheetH = Math.round(paperHmm * k);
   192→   192→
   193→   193→  // -- content geometry -------------------------------------------------------
   194→   194→  const docWmm = file?.docWidthMm ?? paperWmm;
   195→   195→  const docHmm = file?.docHeightMm ?? paperHmm;
   196→   196→  const contentW = Math.round(docWmm * scale * k);
   197→   197→  const contentH = Math.round(docHmm * scale * k);
   198→   198→  const flowWidth = Math.round((paperWmm - 36) * k);
   199→   199→  const flowPadX = Math.round(18 * k);
   200→   200→  const flowPadY = Math.round(20 * k);
   201→   201→  const flowPageH = Math.round((paperHmm - 40) * k);
   202→   202→
   203→   203→  // -- PDF rendering ----------------------------------------------------------
   204→   204→  useEffect(() => {
   205→   205→    let cancelled = false;
   206→   206→    if (!file || file.kind !== "pdf" || file.status !== "ready" || k <= 0) return;
   207→   207→    const render = async () => {
   208→   208→      try {
   209→   209→        // Serialize renders on the shared canvas: abort any in-flight paint first.
   210→   210→        renderTaskRef.current?.cancel();
   211→   211→        renderTaskRef.current = null;
   212→   212→        const doc = await getPdfDoc(file);
   213→   213→        if (cancelled) return;
   214→   214→        const docPage = await doc.getPage(Math.min(page, doc.numPages));
   215→   215→        if (cancelled) return;
   216→   216→        const base = docPage.getViewport({ scale: 1 });
   217→   217→        const dpr = Math.min(window.devicePixelRatio || 1, 2);
   218→   218→        const targetW = Math.max(40, contentW * dpr);
   219→   219→        const capped = Math.min(targetW, 3200);
   220→   220→        const viewport = docPage.getViewport({ scale: capped / base.width });
   221→   221→        const canvas = canvasRef.current;
   222→   222→        if (!canvas) return;
   223→   223→        canvas.width = Math.max(1, Math.floor(viewport.width));
   224→   224→        canvas.height = Math.max(1, Math.floor(viewport.height));
   225→   225→        const ctx = canvas.getContext("2d");
   226→   226→        if (ctx) {
   227→   227→          ctx.fillStyle = "#ffffff";
   228→   228→          ctx.fillRect(0, 0, canvas.width, canvas.height);
   229→   229→        }
   230→   230→        canvas.style.width = `${contentW}px`;
   231→   231→        canvas.style.height = `${contentH}px`;
   232→   232→        const task = docPage.render({ canvas, viewport });
   233→   233→        renderTaskRef.current = task;
   234→   234→        await task.promise;
   235→   235→        if (renderTaskRef.current === task) renderTaskRef.current = null;
   236→   236→        docPage.cleanup();
   237→   237→      } catch {
   238→   238→        /* aborted (superseded) or decode issue — preview keeps last frame */
   239→   239→      }
   240→   240→    };
   241→   241→    void render();
   242→   242→    return () => {
   243→   243→      cancelled = true;
   244→   244→      renderTaskRef.current?.cancel();
   245→   245→      renderTaskRef.current = null;
   246→   246→    };
   247→   247→  }, [file, file?.id, file?.status, page, contentW, contentH, k, getPdfDoc]);
   248→   248→
   249→   249→  // -- DOCX section rendering -------------------------------------------------
   250→   250→  useEffect(() => {
   251→   251→    let cancelled = false;
   252→   252→    if (!file || file.kind !== "docx" || file.status !== "ready" || !file.docWidthMm) return;
   253→   253→    if (docxHostRef.current && !continuous) docxHostRef.current.innerHTML = "";
   254→   254→    const render = async () => {
   255→   255→      try {
   256→   256→        const r = await getDocxRender(file);
   257→   257→        if (cancelled) return;
   258→   258→        setDocxRender(r);
   259→   259→        const host = docxHostRef.current;
   260→   260→        if (!host || continuous) return;
   261→   261→        const idx = Math.min(Math.max(page - 1, 0), r.sectionsHtml.length - 1);
   262→   262→        host.innerHTML = r.sectionsHtml[idx] ?? "";
   263→   263→      } catch {
   264→   264→        /* handled via file.status */
   265→   265→      }
   266→   266→    };
   267→   267→    void render();
   268→   268→    return () => {
   269→   269→      cancelled = true;
   270→   270→    };
   271→   271→  }, [file, file?.id, file?.status, page, continuous, getDocxRender]);
   272→   272→
   273→   273→  // -- Flow HTML (mammoth / .doc conversion) -----------------------------------
   274→   274→  useEffect(() => {
   275→   275→    let cancelled = false;
   276→   276→    if (!file || !(file.kind === "docx" || file.kind === "doc") || file.status !== "ready") return;
   277→   277→    if (!file.continuous) return;
   278→   278→    const render = async () => {
   279→   279→      try {
   280→   280→        const html = await getDocHtml(file);
   281→   281→        if (cancelled) return;
   282→   282→        const host = flowHostRef.current;
   283→   283→        if (!host) return;
   284→   284→        host.innerHTML = html;
   285→   285→        // estimate page count from rendered height
   286→   286→        requestAnimationFrame(() => {
   287→   287→          if (cancelled || !host) return;
   288→   288→          const contentH2 = host.scrollHeight - flowPadY * 2;
   289→   289→          const pages = Math.max(1, Math.ceil(contentH2 / Math.max(1, flowPageH)));
   290→   290→          onEstimatePages(file.id, pages);
   291→   291→        });
   292→   292→      } catch {
   293→   293→        /* handled by file.status */
   294→   294→      }
   295→   295→    };
   296→   296→    void render();
   297→   297→    return () => {
   298→   298→      cancelled = true;
   299→   299→    };
   300→   300→  }, [file, file?.id, file?.status, flowPadY, flowPageH, getDocHtml, onEstimatePages]);
   301→   301→
   302→   302→  const hasDocxFrame =
   303→   303→    !!file && file.kind === "docx" && file.status === "ready" && !file.continuous;
   304→   304→  const docxScale =
   305→   305→    hasDocxFrame && docxRender && docxRender.pageWpx > 0 && contentW > 0
   306→   306→      ? contentW / docxRender.pageWpx
   307→   307→      : 0;
   308→   308→
   309→   309→  const dragHandlers = useMemo(
   310→   310→    () => ({
   311→   311→      onDragOver: (e: React.DragEvent) => {
   312→   312→        e.preventDefault();
   313→   313→        setDragging(true);
   314→   314→      },
   315→   315→      onDragLeave: (e: React.DragEvent) => {
   316→   316→        e.preventDefault();
   317→   317→        setDragging(false);
   318→   318→      },
   319→   319→      onDrop: (e: React.DragEvent) => {
   320→   320→        e.preventDefault();
   321→   321→        setDragging(false);
   322→   322→        const list = Array.from(e.dataTransfer.files ?? []);
   323→   323→        if (list.length > 0) onFiles(list);
   324→   324→      },
   325→   325→    }),
   326→   326→    [onFiles],
   327→   327→  );
   328→   328→
   329→   329→  const navDisabled = !file || file.status !== "ready" || !pageCount || pageCount < 2;
   330→   330→
   331→   331→  const photo = props.photo;
   332→   332→  const photoMode = photo !== null;
   333→   333→  // Tap-to-add only applies while the photo editor is mounted.
   334→   334→  const multiActive = photoMode && photoMulti;
   335→   335→
   336→   336→  const zoomControl = (
   337→   337→    <div className="flex items-center rounded-[5px] bg-[#F3F4F6] p-0.5">
   338→   338→      <button
   339→   339→        type="button"
   340→   340→        aria-pressed={zoom === "fit"}
   341→   341→        onClick={() => onZoomChange("fit")}
   342→   342→        className={`h-6 rounded-[4px] px-2 text-[11px] font-medium transition-colors ${
   343→   343→          zoom === "fit" ? "bg-white text-[#111827] shadow-sm" : "text-[#6B7280] hover:text-[#111827]"
   344→   344→        }`}
   345→   345→      >
   346→   346→        Fit
   347→   347→      </button>
   348→   348→      <button
   349→   349→        type="button"
   350→   350→        aria-pressed={zoom === "100"}
   351→   351→        onClick={() => onZoomChange("100")}
   352→   352→        className={`h-6 rounded-[4px] px-2 font-mono text-[11px] font-medium tabular-nums transition-colors ${
   353→   353→          zoom === "100" ? "bg-white text-[#111827] shadow-sm" : "text-[#6B7280] hover:text-[#111827]"
   354→   354→        }`}
   355→   355→      >
   356→   356→        100%
   357→   357→      </button>
   358→   358→    </div>
   359→   359→  );
   360→   360→
   361→   361→  const setSafePage = useCallback(
   362→   362→    (p: number) => {
   363→   363→      const total = pageCount ?? 1;
   364→   364→      onPageChange(Math.min(Math.max(1, p), Math.max(1, total)));
   365→   365→    },
   366→   366→    [pageCount, onPageChange],
   367→   367→  );
   368→   368→
   369→   369→  return (
   370→   370→    <section
   371→   371→      className="relative flex min-w-0 flex-1 flex-col bg-[#F4F5F7]"
   372→   372→      aria-label="Pratinjau Cetak"
   373→   373→      {...dragHandlers}
   374→   374→    >
   375→   375→      <FileQueue files={props.files ?? []} activeId={props.activeQueueId ?? null} onSelect={onSelect} onRemove={onRemove} />
   376→   376→
   377→   377→      {/* Source-size meta row (badge lives here, never overlaps the queue) */}
   378→   378→      {photoMode && photo ? (
   379→   379→        <div className="flex min-h-9 shrink-0 flex-wrap items-center justify-center gap-x-2 gap-y-1 px-4 py-1">
   380→   380→          <div
   381→   381→            className="flex h-7 items-center gap-1.5 whitespace-nowrap rounded-full border border-[#E5E7EB] bg-white px-3 font-mono text-[10.5px] tabular-nums text-[#6B7280]"
   382→   382→            role="status"
   383→   383→          >
   384→   384→            <ImageIcon size={11} strokeWidth={1.5} aria-hidden="true" />
   385→   385→            Mode Foto • {photo.placements.length} foto • {photo.presetShort} • margin {photo.marginMm} mm
   386→   386→          </div>
   387→   387→          {/* Always-visible picker: add more customers' face photos — the
   388→   388→              grid re-tiles automatically so every distinct photo keeps a slot. */}
   389→   389→          <label
   390→   390→            className="flex h-7 cursor-pointer items-center gap-1 whitespace-nowrap rounded-md border border-[#E5E7EB] bg-white px-2 text-[10.5px] font-semibold text-[#003399] transition-colors hover:bg-[#003399]/5"
   391→   391→            title="Tambah foto orang lain — semua foto otomatis dijajarkan dalam satu lembar"
   392→   392→          >
   393→   393→            <input
   394→   394→              type="file"
   395→   395→              multiple
   396→   396→              accept="image/png,image/jpeg"
   397→   397→              className="sr-only"
   398→   398→              aria-label="Tambah foto ke lembar"
   399→   399→              onChange={(e) => {
   400→   400→                const list = Array.from(e.target.files ?? []);
   401→   401→                if (list.length > 0) props.onAddPhotoFiles(list);
   402→   402→                e.target.value = "";
   403→   403→              }}
   404→   404→            />
   405→   405→            <ImagePlus size={11} strokeWidth={1.5} aria-hidden="true" />
   406→   406→            Tambah Foto
   407→   407→          </label>
   408→   408→          {photo.placements.length > 0 ? (
   409→   409→            <div className="flex h-8 items-center gap-0.5 rounded-md border border-[#E5E7EB] bg-white p-0.5 shadow-[0_1px_2px_rgba(17,24,39,0.05)]">
   410→   410→              {photo.selectedIds.length > 0 ? (
   411→   411→                <span
   412→   412→                  className="whitespace-nowrap px-1.5 font-mono text-[10px] font-medium tabular-nums text-[#003399]"
   413→   413→                  aria-live="polite"
   414→   414→                >
   415→   415→                  {photo.selectedIds.length} dipilih
   416→   416→                </span>
   417→   417→              ) : null}
   418→   418→              <button
   419→   419→                type="button"
   420→   420→                onClick={photo.onSelectAll}
   421→   421→                disabled={photo.selectedIds.length === photo.placements.length}
   422→   422→                title="Pilih semua foto — geser sekali untuk memindah semuanya"
   423→   423→                className="h-7 whitespace-nowrap rounded px-2 text-[10.5px] font-medium text-[#374151] transition-colors hover:bg-[#F3F4F6] disabled:opacity-30 disabled:hover:bg-transparent"
   424→   424→              >
   425→   425→                Pilih Semua
   426→   426→              </button>
   427→   427→              <button
   428→   428→                type="button"
   429→   429→                onClick={() => photo.onSelect(null)}
   430→   430→                disabled={photo.selectedIds.length === 0}
   431→   431→                title="Kosongkan pilihan"
   432→   432→                className="h-7 whitespace-nowrap rounded px-2 text-[10.5px] font-medium text-[#374151] transition-colors hover:bg-[#F3F4F6] disabled:opacity-30 disabled:hover:bg-transparent"
   433→   433→              >
   434→   434→                Kosongkan
   435→   435→              </button>
   436→   436→              <button
   437→   437→                type="button"
   438→   438→                aria-pressed={photoMulti}
   439→   439→                onClick={() => setPhotoMulti((v) => !v)}
   440→   440→                title="Mode pilih banyak (layar sentuh): ketuk foto satu per satu untuk menambah ke pilihan"
   441→   441→                className={`h-7 whitespace-nowrap rounded px-2 text-[10.5px] font-semibold transition-colors ${
   442→   442→                  photoMulti
   443→   443→                    ? "bg-[#003399] text-white"
   444→   444→                    : "text-[#003399] hover:bg-[#003399]/5"
   445→   445→                }`}
