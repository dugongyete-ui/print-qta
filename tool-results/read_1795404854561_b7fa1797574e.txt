    83→function normRotation(rot: number): number {
    84→  const r = Math.round(rot / 90) * 90;
    85→  return ((r % 360) + 360) % 360;
    86→}
    87→
    88→function swappedBBox(rotation: number): boolean {
    89→  const r = normRotation(rotation);
    90→  return r === 90 || r === 270;
    91→}

    93→/** Visual bounding box (mm) of a placement, accounting for rotation. */
    94→function placementBBox(p: {
    95→  xMm: number;
    96→  yMm: number;
    97→  wMm: number;
    98→  hMm: number;
    99→  rotation: number;
   100→}): { x: number; y: number; w: number; h: number } {
   101→  const swapped = swappedBBox(p.rotation);
   102→  const w = swapped ? p.hMm : p.wMm;
   103→  const h = swapped ? p.wMm : p.hMm;
   104→  const cx = p.xMm + p.wMm / 2;
   105→  const cy = p.yMm + p.hMm / 2;
   106→  return { x: cx - w / 2, y: cy - h / 2, w, h };
   107→}
   108→
   109→/** Keep the visual bbox fully on the paper by clamping the rotation center. */
   110→function clampPlacement(
   111→  p: PhotoPlacement,
   112→  paperWmm: number,
   113→  paperHmm: number,
   114→): PhotoPlacement {
   115→  const bb = placementBBox(p);
   116→  const cx = p.xMm + p.wMm / 2;
   117→  const cy = p.yMm + p.hMm / 2;
   118→  const ncx = Math.min(
   119→    Math.max(cx, bb.w / 2),
   120→    Math.max(bb.w / 2, paperWmm - bb.w / 2),
   121→  );
   122→  const ncy = Math.min(
   123→    Math.max(cy, bb.h / 2),
   124→    Math.max(bb.h / 2, paperHmm - bb.h / 2),
   125→  );
   126→  return { ...p, xMm: ncx - p.wMm / 2, yMm: ncy - p.hMm / 2 };
   127→}

   129→function toBase64(bytes: Uint8Array): string {
   130→  let binary = "";
   131→  const chunk = 0x8000;
   132→  for (let i = 0; i < bytes.length; i += chunk) {
   133→    binary += String.fromCharCode(
   134→      ...Array.from(bytes.subarray(i, i + chunk)),
   135→    );
   136→  }
   137→  return btoa(binary);
   138→}

   144→function loadPdfTask(file: LoadedFile): Promise<PDFDocumentLoadingTask> {
   145→  const cached = pdfTaskCache.get(file.id);
   146→  if (cached) return cached;
   147→  const task = (async () => {
   148→    const pdfjs = await import("pdfjs-dist");
   149→    pdfjs.GlobalWorkerOptions.workerSrc = "/pdf.worker.min.mjs";
   150→    // pdf.js may transfer the buffer to the worker — always hand it a copy.
   151→    const data = file.bytes!.slice();
   152→    return pdfjs.getDocument({ data });
   153→  })();
   154→  pdfTaskCache.set(file.id, task);
   155→  return task;
   156→}

   158→async function loadPdfDoc(file: LoadedFile): Promise<PDFDocumentProxy> {
   159→  const task = await loadPdfTask(file);
   160→  return task.promise;
   161→}

   163→async function inspectPdf(file: LoadedFile): Promise<Partial<LoadedFile>> {
   164→  try {
   165→    const doc = await loadPdfDoc(file);
   166→    const page1 = await doc.getPage(1);
   167→    const vp = page1.getViewport({ scale: 1 });
   168→    const wMm = vp.width * PT_TO_MM;
   169→    const hMm = vp.height * PT_TO_MM;
   170→    const matched = matchPaper(wMm, hMm);
   171→    return {
   172→      status: "ready",
   173→      pageCount: doc.numPages,
   174→      docWidthMm: wMm,
   175→      docHeightMm: hMm,
   176→      docLabel: matched ? getPaper(matched).shortLabel : sizeLabel(wMm, hMm),
   177→    };
   178→  } catch {
   179→    pdfTaskCache.delete(file.id);
   180→    return {
   181→      status: "error",
   182→      error: "Dokumen PDF rusak, terproteksi, atau tidak dapat dibaca.",
   183→    };
   184→  }
   185→}

   187→async function inspectImage(file: LoadedFile): Promise<Partial<LoadedFile>> {
   188→  if (!file.bytes) {
   189→    return { status: "error", error: "Data gambar kosong." };
   190→  }
   191→  const url = URL.createObjectURL(
   192→    new Blob([file.bytes.slice().buffer as ArrayBuffer], { type: file.mime }),
   193→  );
   194→  try {
   195→    const img = await new Promise<HTMLImageElement>((resolve, reject) => {
   196→      const el = new Image();
   197→      el.onload = () => resolve(el);
   198→      el.onerror = () => reject(new Error("decode"));
   199→      el.src = url;
   200→    });
   201→    return {
   202→      status: "ready",
   203→      url,
   204→      pageCount: 1,
   205→      docWidthMm: img.naturalWidth * PX96_TO_MM,
   206→      docHeightMm: img.naturalHeight * PX96_TO_MM,
   207→      docLabel: `${img.naturalWidth} × ${img.naturalHeight} px`,
   208→    };
   209→  } catch {
   210→    URL.revokeObjectURL(url);
   211→    return { status: "error", error: "Gambar tidak dapat dibaca atau rusak." };
   212→  }
   213→}

   215→async function renderDocx(file: LoadedFile): Promise<DocxRender> {
   216→  const cached = docxRenderCache.get(file.id);
   217→  if (cached) return cached;
   218→  const task = (async () => {
   219→    const docxPreview = await import("docx-preview");
   220→    const host = document.createElement("div");
   221→    host.style.cssText =
   222→      "position:fixed;left:-100000px;top:0;width:1200px;visibility:hidden;";
   223→    document.body.appendChild(host);
   224→    try {
   225→      await docxPreview.renderAsync(
   226→        new Blob([file.bytes!.slice().buffer as ArrayBuffer], { type: file.mime }),
   227→        host,
   228→        host,
   229→        {
   230→          inWrapper: true,
   231→          breakPages: true,
   232→          experimental: true,
   233→          renderHeaders: true,
   234→          renderFooters: true,
   235→          useBase64URL: true,
   236→          ignoreWidth: false,
   237→          ignoreHeight: false,
   238→        },
   239→      );
   240→      const wrapper = host.querySelector(".docx-wrapper");
   241→      const sections = wrapper
   242→        ? Array.from(wrapper.querySelectorAll("section.docx"))
   243→        : Array.from(host.querySelectorAll("section.docx"));
   244→      if (sections.length === 0) throw new Error("no-sections");
   245→      const first = sections[0] as HTMLElement;
   246→      const wpx = first.offsetWidth || Math.round(210 * (96 / 25.4));
   247→      const hpx = first.offsetHeight || Math.round(297 * (96 / 25.4));
   248→      return {
   249→        sectionsHtml: sections.map((s) => (s as HTMLElement).outerHTML),
   250→        pageWpx: wpx,
   251→        pageHpx: hpx,
   252→        pageWmm: wpx * PX96_TO_MM,
   253→        pageHmm: hpx * PX96_TO_MM,
   254→        pageCount: sections.length,
   255→      } satisfies DocxRender;
   256→    } finally {
   257→      host.remove();
   258→    }
   259→  })();
   260→  docxRenderCache.set(file.id, task);
   261→  return task;
   262→}

   264→async function convertViaServer(file: LoadedFile): Promise<string> {
   265→  const cached = docHtmlCache.get(file.id);
   266→  if (cached) return cached;
   267→  const task = (async () => {
   268→    const form = new FormData();
   269→    form.append(
   270→      "file",
   271→      new Blob([file.bytes!.slice().buffer as ArrayBuffer], { type: file.mime }),
   272→      file.name,
   273→    );
   274→    const res = await fetch("/api/convert-docx", { method: "POST", body: form });
   275→    if (!res.ok) {
   276→      const body = (await res.json().catch(() => null)) as { message?: string } | null;
   277→      throw new Error(body?.message ?? "Konversi dokumen gagal di server.");
   278→    }
   279→    const body = (await res.json()) as { html: string };
   280→    return body.html;
   281→  })();
   282→  docHtmlCache.set(file.id, task);
   283→  return task;
   284→}

   286→async function inspectWord(file: LoadedFile): Promise<Partial<LoadedFile>> {
   287→  // 1st choice: client-side docx-preview (true pagination + layout)
   288→  if (file.kind === "docx") {
   289→    try {
   290→      const render = await renderDocx(file);
   291→      const matched = matchPaper(render.pageWmm, render.pageHmm);
   292→      return {
   293→        status: "ready",
   294→        pageCount: render.pageCount,
   295→        docWidthMm: render.pageWmm,
   296→        docHeightMm: render.pageHmm,
   297→        docLabel: matched ? getPaper(matched).shortLabel : sizeLabel(render.pageWmm, render.pageHmm),
   298→      };
   299→    } catch {
   300→      docxRenderCache.delete(file.id);
   301→      // fall through to server-side mammoth
   302→    }
   303→  }
   304→  // 2nd choice: server conversion (mammoth / LibreOffice headless)
   305→  try {
   306→    await convertViaServer(file);
   307→    return {
   308→      status: "ready",
   309→      continuous: true,
   310→      pageCount: 1,
   311→      docWidthMm: 210,
   312→      docHeightMm: 297,
   313→      docLabel: "Halaman Mengalir",
   314→    };
   315→  } catch (err) {
   316→    return {
   317→      status: "error",
   318→      error:
   319→        err instanceof Error
   320→          ? err.message
   321→          : "Konversi dokumen gagal. Simpan ulang sebagai .docx atau PDF.",
   322→    };
   323→  }
   324→}

   332→export function usePrintStation() {
   333→  const [files, setFiles] = useState<LoadedFile[]>([]);
   334→  const [activeId, setActiveId] = useState<string | null>(null);
   335→  const [settings, setSettings] = useState<PrintSettings>(DEFAULT_SETTINGS);
   336→  const [photoMode, setPhotoModeState] = useState(false);
   337→  const [photoPlacements, setPhotoPlacements] = useState<PhotoPlacement[]>([]);
   338→  const [photoSelectedId, setPhotoSelectedId] = useState<string | null>(null);
   339→  const [photoSettings, setPhotoSettings] = useState<PhotoSettings>(DEFAULT_PHOTO_SETTINGS);
   340→  const [page, setPage] = useState(1);
   341→  const [zoom, setZoom] = useState<"fit" | "100">("fit");
   342→  const [showMargins, setShowMargins] = useState(false);
   343→  const [printing, setPrinting] = useState(false);
   344→  const [duplexAssist, setDuplexAssist] = useState<DuplexAssist | null>(null);
   345→  const [spooler, setSpooler] = useState<SpoolerState>({
   346→    state: "checking",
   347→    info: null,
   348→  });
   349→
   350→  const activeFile = useMemo(
   351→    () => files.find((f) => f.id === activeId) ?? null,
   352→    [files, activeId],
   353→  );
   354→  const activeFileRef = useRef(activeFile);
   355→  activeFileRef.current = activeFile;
   356→  const settingsRef = useRef(settings);
   357→  settingsRef.current = settings;
   358→  const filesRef = useRef(files);
   359→  filesRef.current = files;
   360→  const photoModeRef = useRef(photoMode);
   361→  photoModeRef.current = photoMode;
   362→  const photoPlacementsRef = useRef(photoPlacements);
   363→  photoPlacementsRef.current = photoPlacements;
   364→  const photoSelectedIdRef = useRef(photoSelectedId);
   365→  photoSelectedIdRef.current = photoSelectedId;
   366→  const photoSettingsRef = useRef(photoSettings);
   367→  photoSettingsRef.current = photoSettings;
   368→
   369→  // -- derived paper geometry -----------------------------------------------
   370→  const paper = getPaper(settings.paperId);
   371→  const { wMm: paperWmm, hMm: paperHmm } = paperDims(
   372→    settings.paperId,
   373→    settings.orientation,
   374→  );
   375→  const paperDimsRef = useRef({ wMm: paperWmm, hMm: paperHmm });
   376→  paperDimsRef.current = { wMm: paperWmm, hMm: paperHmm };
   377→  // Assigned after the photo actions below — used by addFiles to auto-place images.
   378→  const intakeImageRef = useRef<(itemId: string) => void>(() => {});
   379→  // True when the user hand-edited the sheet since the last auto-grid arrange;
   380→  // intake auto-arrange never reshuffles a hand-tuned layout.
   381→  const layoutTouchedRef = useRef(false);
   382→  const intakeTimerRef = useRef<number | null>(null);
   383→
   384→  const contentScale = useMemo(() => {
   385→    const f = activeFile;
   386→    if (!f || f.docWidthMm == null || f.docWidthMm <= 0) return 1;
   387→    if (settings.scaleMode !== "fit" || f.continuous) return 1;
   388→    const paperW = paperWmm;
   389→    const paperH = paperHmm;
   390→    return Math.min(paperW / f.docWidthMm, paperH / (f.docHeightMm || 1));
   391→  }, [activeFile, settings.scaleMode, paperWmm, paperHmm]);
   392→
   393→  const paperMismatch = useMemo(() => {
   394→    const f = activeFile;
   395→    if (!f || f.docWidthMm == null || f.continuous) return false;
   396→    return matchPaper(f.docWidthMm, f.docHeightMm ?? 0) !== settings.paperId;
   397→  }, [activeFile, settings.paperId]);
   398→
   399→  const contentOverflows = useMemo(() => {
   400→    const f = activeFile;
   401→    if (!f || f.docWidthMm == null || f.continuous) return false;
   402→    const s = settings.scaleMode === "fit" ? contentScale : 1;
   403→    return (
   404→      f.docWidthMm * s > paperWmm + 0.5 || (f.docHeightMm ?? 0) * s > paperHmm + 0.5
   405→    );
   406→  }, [activeFile, contentScale, settings.scaleMode, paperWmm, paperHmm]);
   407→
   408→  // -- page range -------------------------------------------------------------
   409→  const selectedPages = useMemo(() => {
   410→    const f = activeFile;
   411→    if (!f || f.status !== "ready" || !f.pageCount) return [];
   412→    if (settings.rangeMode === "all") {
   413→      return Array.from({ length: f.pageCount }, (_, i) => i + 1);
   414→    }
   415→    const parsed = parsePageRange(settings.rangeText, f.pageCount);
   416→    return parsed.valid ? parsed.pages : [];
   417→  }, [activeFile, settings.rangeMode, settings.rangeText]);
   418→
   419→  const rangeInvalid =
   420→    settings.rangeMode === "custom" &&
   421→    !parsePageRange(settings.rangeText, activeFile?.pageCount ?? 1).valid;
   422→
   423→  const selectedPagesRef = useRef<number[]>([]);
   424→  selectedPagesRef.current = selectedPages;
   425→
   426→  const photoReady =
   427→    photoPlacements.length > 0 &&
   428→    photoPlacements.every((p) => {
   429→      const it = files.find((f) => f.id === p.itemId);
   430→      return !!it && it.kind === "image" && it.status === "ready" && !!it.url;
   431→    });
   432→  const photoPreset = getPhotoPreset(photoSettings.presetId);
   433→  const photoSelected =
   434→    photoPlacements.find((p) => p.id === photoSelectedId) ?? null;
   435→
   436→  /**
   437→   * Max copies of the current preset that fit on one sheet (best of portrait
   438→   * / landscape orientation), clamped by the engine's placement cap.
   439→   */
   440→  const photoCapacity = useMemo(() => {
   441→    const w = photoPreset.group === "custom" ? photoSettings.customWMm : photoPreset.wMm;
   442→    const h = photoPreset.group === "custom" ? photoSettings.customHMm : photoPreset.hMm;
   443→    const margin = photoSettings.marginMm;
   444→    const gap = photoSettings.gapMm;
   445→    const printW = paperWmm - 2 * margin;
   446→    const printH = paperHmm - 2 * margin;
   447→    const capFor = (pw: number, ph: number) => {
   448→      if (pw > printW + PHOTO_TOL || ph > printH + PHOTO_TOL) return 0;
   449→      return (
   450→        Math.max(1, Math.floor((printW + gap) / (pw + gap))) *
   451→        Math.max(1, Math.floor((printH + gap) / (ph + gap)))
   452→      );
   453→    };
   454→    return Math.min(Math.max(capFor(w, h), capFor(h, w)), MAX_PLACEMENTS);
   455→  }, [photoPreset, photoSettings, paperWmm, paperHmm]);
   456→
   457→  const photoBeyondMargin = photoPlacements.some((p) => {
   458→    const bb = placementBBox(p);
   459→    return (
   460→      bb.x < photoSettings.marginMm - PHOTO_TOL ||
   461→      bb.y < photoSettings.marginMm - PHOTO_TOL ||
   462→      bb.x + bb.w > paperWmm - photoSettings.marginMm + PHOTO_TOL ||
   463→      bb.y + bb.h > paperHmm - photoSettings.marginMm + PHOTO_TOL
   464→    );
   465→  });
   466→
   467→  const canPrint = photoMode
   468→    ? photoReady && !printing
   469→    : !!activeFile &&
   470→      activeFile.status === "ready" &&
   471→      selectedPages.length > 0 &&
   472→      !printing;
   473→
   474→  // -- spooler detection ------------------------------------------------------
   475→  useEffect(() => {
   476→    let alive = true;
   477→    fetch("/api/print")
   478→      .then((r) => r.json())
   479→      .then((body: { spooler: { type: "ipp" | "lp"; printer: string | null } | null }) => {
   480→        if (!alive) return;
   481→        setSpooler(
   482→          body.spooler
   483→            ? { state: "available", info: body.spooler }
   484→            : { state: "unavailable", info: null },
   485→        );
   486→      })
   487→      .catch(() => alive && setSpooler({ state: "unavailable", info: null }));
   488→    return () => {
   489→      alive = false;
   490→    };
   491→  }, []);
   492→
   493→  // -- auto-fit to loaded tray ------------------------------------------------
   494→  const autoFitKey = useRef<string | null>(null);
   495→  useEffect(() => {
   496→    const f = activeFile;
   497→    if (!f || f.status !== "ready" || f.docWidthMm == null || f.continuous) return;
   498→    const key = `${f.id}:${settings.paperId}`;
   499→    if (autoFitKey.current === key) return;
   500→    const matched = matchPaper(f.docWidthMm, f.docHeightMm ?? 0);
   501→    if (matched !== settings.paperId) {
   502→      autoFitKey.current = key;
   503→      setSettings((s) => (s.scaleMode === "fit" ? s : { ...s, scaleMode: "fit" }));
   504→    }
   505→  }, [activeFile, settings.paperId]);
   506→
   507→  // -- file intake ------------------------------------------------------------
   508→  const patchFile = useCallback((id: string, patch: Partial<LoadedFile>) => {
   509→    setFiles((prev) => prev.map((f) => (f.id === id ? { ...f, ...patch } : f)));
   510→  }, []);
   511→
   512→  const addFiles = useCallback(
   513→    (incoming: File[]) => {
   514→      for (const file of incoming) {
   515→        const ext = (file.name.split(".").pop() ?? "").toLowerCase();
   516→        const accepted = ACCEPTED_EXT[ext];
   517→        if (!accepted) {
   518→          toast.error(`Format .${ext || "?"} tidak didukung.`, {
   519→            description: `${file.name} — gunakan PDF, DOC, DOCX, PNG, atau JPG.`,
   520→          });
   521→          continue;
   522→        }
   523→        if (file.size > MAX_FILE_BYTES) {
   524→          toast.error(`Ukuran file melebihi 50 MB.`, {
   525→            description: file.name,
   526→          });
   527→          continue;
   528→        }
   529→        const id = newId();
   530→        const loaded: LoadedFile = {
   531→          id,
   532→          name: file.name,
   533→          ext,
   534→          sizeBytes: file.size,
   535→          kind: accepted.kind,
   536→          mime: accepted.mime,
   537→          bytes: null,
   538→          url: null,
   539→          status: "loading",
   540→          pageCount: null,
   541→          docWidthMm: null,
   542→          docHeightMm: null,
   543→          docLabel: null,
   544→        };
   545→        setFiles((prev) => [...prev, loaded]);
   546→        setActiveId((cur) => cur ?? id);
   547→        setPage(1);
   548→
   549→        file
   550→          .arrayBuffer()
   551→          .then(async (buf) => {

   763→
   764→  /**
   765→   * Auto-grid arrange — the print-shop "ngejajar N foto" flow. Tiles N slots
   766→   * sized to the current preset (or explicit dims) in a balanced grid
   767→   * (6 → 3×2, 8 → 4×2, 9 → 3×3) aligned inside the printable area
   768→   * (top / center / bottom × left / center / right); portrait / landscape is
   769→   * picked by whichever capacity fits N. Slots are always filled round-robin
   770→   * with every distinct ready photo on the sheet — changing the count never
   771→   * collapses the sheet into copies of a hand-picked photo (use "Isi Penuh"
   772→   * for that).
   773→   */
   774→  const arrangePhotoCopies = useCallback(
   775→    (
   776→      count: number,
   777→      options?: {
   778→        dims?: { wMm: number; hMm: number };
   779→        auto?: boolean;
   780→        /** Explicit margin override — used when margin changes and re-tile
   781→         * happens in the same tick (borderless toggle), where the settings
   782→         * ref is still stale. */
   783→        marginMm?: number;
   784→        gapMm?: number;
   785→        alignV?: GridAlignV;
   786→        alignH?: GridAlignH;
   787→      },
   788→    ) => {
   789→      const list = photoPlacementsRef.current;
   790→      if (list.length === 0) return;
   791→      const ps = photoSettingsRef.current;
   792→      const { wMm: paperW, hMm: paperH } = paperDimsRef.current;
   793→      const preset = getPhotoPreset(ps.presetId);
   794→      const presetW = preset.group === "custom" ? ps.customWMm : preset.wMm;
   795→      const presetH = preset.group === "custom" ? ps.customHMm : preset.hMm;
   796→      const pw = Math.min(options?.dims ? options.dims.wMm : presetW, paperW);
   797→      const ph = Math.min(options?.dims ? options.dims.hMm : presetH, paperH);
   798→      const margin = options?.marginMm ?? ps.marginMm;
   799→      const gap = options?.gapMm ?? ps.gapMm;
   800→      const alignV = options?.alignV ?? ps.alignV;
   801→      const alignH = options?.alignH ?? ps.alignH;
   802→      const printW = paperW - 2 * margin;
   803→      const printH = paperH - 2 * margin;
   804→
   805→      const fitFor = (w: number, h: number) => {
   806→        if (w > printW + PHOTO_TOL || h > printH + PHOTO_TOL) return null;
   807→        const cols = Math.max(1, Math.floor((printW + gap) / (w + gap)));
   808→        const rows = Math.max(1, Math.floor((printH + gap) / (h + gap)));
   809→        return { cols, rows, cap: cols * rows };
   810→      };
   811→
   812→      const portrait = fitFor(pw, ph);
   813→      const landscape = fitFor(ph, pw);
   814→      const n0 = Math.max(1, Math.round(count));
   815→      // Keep the preset's own orientation whenever it can fit N copies —
   816→      // rotating a pas-foto frame would crop faces. Only fall back to the
   817→      // rotated fit when portrait cannot hold N (or doesn't fit at all).
   818→      let useLandscape: boolean;
   819→      if (portrait && landscape) {
   820→        useLandscape =
   821→          portrait.cap >= n0
   822→            ? false
   823→            : landscape.cap >= n0
   824→              ? true
   825→              : landscape.cap > portrait.cap;
   826→      } else {
   827→        useLandscape = !portrait && !!landscape;
   828→      }
   829→      const fit = useLandscape ? landscape : portrait;
   830→      if (!fit) {
   831→        if (!options?.auto) {
   832→          toast.error("Foto melebihi area cetak lembar.", {
   833→            description:
   834→              "Kecilkan ukuran foto atau margin lembar, lalu susun ulang.",
   835→          });
   836→        }
   837→        return;
   838→      }
   839→      const total = Math.min(n0, fit.cap, MAX_PLACEMENTS);
   840→
   841→      // Balanced grid: fewest wasted slots first, then the most square block,
   842→      // preferring the wider option on ties (6 → 3×2, 9 → 3×3, 8 → 4×2).
   843→      let cols = fit.cols;
   844→      let bestScore = Infinity;
   845→      for (let c = 1; c <= fit.cols; c++) {
   846→        const r = Math.ceil(total / c);
   847→        if (r > fit.rows) continue;
   848→        const score = (c * r - total) * 1000 + Math.abs(c - r);
   849→        if (score < bestScore || (score === bestScore && c > cols)) {
   850→          bestScore = score;
   851→          cols = c;
   852→        }
   853→      }
   854→      const w = useLandscape ? ph : pw;
   855→      const h = useLandscape ? pw : ph;
   856→      const rows = Math.ceil(total / cols);
   857→
   858→      // Slot sources: the distinct ready images already on the sheet fill
   859→      // slots round-robin — every uploaded photo stays on the grid.
   860→      let sources: string[] = [];
   861→      const seen = new Set<string>();
   862→      for (const p of list) {
   863→        if (seen.has(p.itemId)) continue;
   864→        const it = filesRef.current.find(
   865→          (f) =>
   866→            f.id === p.itemId &&
   867→            f.kind === "image" &&
   868→            f.status === "ready" &&
   869→            !!f.url,
   870→        );
   871→        if (it) {
   872→          seen.add(p.itemId);
   873→          sources.push(p.itemId);
   874→        }
   875→      }
   876→      if (sources.length === 0) sources = [list[0].itemId];
   877→
   878→      const gridH = rows * h + (rows - 1) * gap;
   879→      const gy =
   880→        alignV === "top"
   881→          ? margin
   882→          : alignV === "bottom"
   883→            ? margin + printH - gridH
   884→            : margin + (printH - gridH) / 2;
   885→      const out: PhotoPlacement[] = [];
   886→      for (let r = 0; r < rows; r++) {
   887→        const inRow = Math.min(total - r * cols, cols);
   888→        const rowW = inRow * w + (inRow - 1) * gap;
   889→        const gx =
   890→          alignH === "left"
   891→            ? margin
   892→            : alignH === "right"
   893→              ? margin + printW - rowW
   894→              : margin + (printW - rowW) / 2;
   895→        for (let c = 0; c < inRow; c++) {
   896→          const itemId = sources[(r * cols + c) % sources.length];
   897→          const src = list.find((p) => p.itemId === itemId);
   898→          out.push({
   899→            id: newPhotoId(),
   900→            itemId,
   901→            xMm: gx + c * (w + gap),
   902→            yMm: gy + r * (h + gap),
   903→            wMm: w,
   904→            hMm: h,
   905→            rotation: 0,
   906→            fit: src ? src.fit : "cover",
   907→            cropX: src?.cropX ?? 0,
   908→            cropY: src?.cropY ?? 0,
   909→          });
   910→        }
   911→      }
   912→      setPhotoPlacements(out);
   913→      setPhotoSelectedId(null);
   914→      layoutTouchedRef.current = false;
   915→      if (sources.length > total) {
   916→        toast.info(
   917→          `${total} foto tersusun dari ${sources.length} foto yang diunggah`,
   918→          {
   919→            description: `Naikkan jumlah foto ke ${sources.length} agar semua foto ikut tersusun.`,
   920→          },
   921→        );
   922→        return;
   923→      }
   924→      if (n0 > fit.cap) {
   925→        toast.info(`Lembar penuh — maksimal ${fit.cap} foto ukuran ini`, {
   926→          description:
   927→            "Kurangi jumlah foto, pilih preset lebih kecil, atau ganti kertas lebih besar.",
   928→        });
   929→        return;
   930→      }
   931→      if (!options?.auto || out.length > 1) {
   932→        toast.success(`${out.length} foto tersusun otomatis`, {
   933→          description: `${useLandscape ? "Lanskap" : "Potret"} • ${cols} kolom × ${rows} baris • ${
   934→            alignV === "top" ? "rata atas" : alignV === "bottom" ? "rata bawah" : "di tengah"
   935→          } • jarak ${gap} mm`,
   936→        });
   937→      }
   938→    },
   939→    [],
   940→  );
   941→
   942→  /**
   943→   * Debounced intake arrange: uploads become ready one file at a time, so the
   944→   * re-tile waits for a short quiet window before regenerating the grid with
   945→   * every photo from the batch.
   946→   */
   947→  const scheduleIntakeArrange = useCallback(() => {
   948→    if (intakeTimerRef.current !== null) {
   949→      window.clearTimeout(intakeTimerRef.current);
   950→    }
   951→    intakeTimerRef.current = window.setTimeout(() => {

   955→      // the count is the only decision, tiling is always automatic.

   957→        auto: true,
   958→      });
   959→    }, 350);
   960→  }, [arrangePhotoCopies]);
   961→
   962→  /**
   963→   * Borderless toggle — zeroes / restores the sheet margin and re-tiles the
   964→   * grid in the same tick with the explicit new margin (the settings ref is
   965→   * still stale at this point, so it is passed explicitly).
   966→   */
   967→  const setPhotoBorderless = useCallback(

   969→      setPhotoSettings((s) => ({
   970→        ...s,
   971→        borderless: next,
   972→        marginMm: next ? 0 : DEFAULT_PHOTO_SETTINGS.marginMm,

   991→  /** Apply a photo size preset to the selected placement (or all when none). */
   992→  const applyPhotoPreset = useCallback(
   993→    (presetId: string, customDims?: { wMm: number; hMm: number }) => {
   994→      const ps = photoSettingsRef.current;
   995→      const { wMm: paperW, hMm: paperH } = paperDimsRef.current;
   996→      setPhotoSettings((s) => ({
   997→        ...s,
   998→        presetId,
   999→        ...(customDims
  1000→          ? { customWMm: customDims.wMm, customHMm: customDims.hMm }
  1001→          : {}),
  1002→      }));
  1003→      const preset = getPhotoPreset(presetId);
  1004→      const w = Math.min(
  1005→        customDims
  1006→          ? customDims.wMm
  1007→          : preset.group === "custom"
  1008→            ? ps.customWMm
  1009→            : preset.wMm,
  1010→        paperW,
  1011→      );
  1012→      const h = Math.min(
  1013→        customDims
  1014→          ? customDims.hMm
  1015→          : preset.group === "custom"
  1016→            ? ps.customHMm
  1017→            : preset.hMm,
  1018→        paperH,
  1019→      );
  1020→      // Untouched auto-grid sheet: the preset drives the whole layout —
  1021→      // re-tile every copy to the new size at once. A hand-tuned sheet keeps
  1022→      // its placement (sizes change in place) instead of reshuffling.

  1024→        arrangePhotoCopies(ps.perSheet, {

  1029→      const selId = photoSelectedIdRef.current;

  1033→          const cx = p.xMm + p.wMm / 2;
  1034→          const cy = p.yMm + p.hMm / 2;
  1035→          return clampPlacement(
  1036→            { ...p, wMm: w, hMm: h, xMm: cx - w / 2, yMm: cy - h / 2 },
  1037→            paperW,
  1038→            paperH,
  1039→          );
  1040→        }),
  1041→      );
  1042→    },
  1043→    [arrangePhotoCopies],
  1044→  );
  1045→
  1046→  /** Fill the whole sheet with copies of the selected photo (pas-foto sheets). */
  1047→  const fillPhotoSheet = useCallback(() => {
  1048→    const list = photoPlacementsRef.current;
  1049→    if (list.length === 0) return;
  1050→    layoutTouchedRef.current = true;
  1051→    const selId = photoSelectedIdRef.current;
  1052→    const p = list.find((x) => x.id === selId) ?? list[0];
  1053→    const ps = photoSettingsRef.current;
  1054→    const { wMm: paperW, hMm: paperH } = paperDimsRef.current;
  1055→    const margin = ps.marginMm;
  1056→    const gap = ps.gapMm;
  1057→    const printW = paperW - 2 * margin;
  1058→    const printH = paperH - 2 * margin;
  1059→    const bb = placementBBox(p);
  1060→    if (bb.w > printW + PHOTO_TOL || bb.h > printH + PHOTO_TOL) {
  1061→      toast.error("Foto lebih besar dari area cetak.", {
  1062→        description: "Perkecil ukuran foto atau margin lembar terlebih dahulu.",
  1063→      });
  1064→      return;
  1065→    }
  1066→    const cols = Math.max(1, Math.floor((printW + gap) / (bb.w + gap)));

  1080→          id: newPhotoId(),
  1081→          itemId: p.itemId,
  1082→          xMm: vx + bb.w / 2 - p.wMm / 2,

  1124→      return np;
  1125→    });
