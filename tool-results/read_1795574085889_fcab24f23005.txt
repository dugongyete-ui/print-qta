    89→function NumField({
    90→  label,
    91→  value,
    92→  onCommit,
    93→  min,
    94→  max,
    95→  disabled,
    96→}: {
    97→  label: string;
    98→  value: number;
    99→  onCommit: (v: number) => void;
   100→  min: number;
   101→  max: number;
   102→  disabled?: boolean;
   103→}) {
   104→  const fmt = (v: number) => String(Math.round(v * 10) / 10);
   105→  const commit = (raw: string) => {
   106→    const parsed = Number(raw.replace(",", "."));
   107→    if (!Number.isFinite(parsed)) return;
   108→    onCommit(Math.min(Math.max(parsed, min), max));
   109→  };
   110→  // Uncontrolled input keyed by the external value: dragging / resizing on the
   111→  // canvas remounts it with fresh numbers, while local typing stays untouched.
   112→  return (
   113→    <label className="flex min-w-0 flex-1 flex-col gap-1">
   114→      <span className="font-mono text-[9.5px] uppercase tracking-[0.06em] text-[#6B7280]">
   115→        {label}
   116→      </span>
   117→      <input
   118→        key={`${label}-${fmt(value)}`}
   119→        type="number"
   120→        inputMode="decimal"
   121→        step="0.5"
   122→        defaultValue={fmt(value)}
   123→        disabled={disabled}
   124→        onBlur={(e) => commit(e.target.value)}
   125→        onKeyDown={(e) => {
   126→          if (e.key === "Enter") {
   127→            e.preventDefault();
   128→            (e.target as HTMLInputElement).blur();
   129→          }
   130→        }}
   131→        aria-label={label}
   132→        className="h-8 w-full rounded-md border border-[#E5E7EB] bg-white px-2 font-mono text-[11.5px] tabular-nums text-[#111827] outline-none transition-colors focus:border-[#003399] disabled:opacity-40"
   133→      />
   134→    </label>
   135→  );
   136→}

   209→interface PhotoInspectorProps {
   210→  placements: PhotoPlacement[];

   212→  /** All selected ids — actions apply to the whole group. */

   217→  paperWmm: number;

   220→  /** Max copies of the current preset that fit on one sheet. */

   223→  /** Borderless toggle — zeroes margin + re-tiles in one tick. */

   227→  onAlign: (dir: PhotoAlign) => void;
   228→  onFillSheet: () => void;

   231→  onArrangeCount: (
   232→    n: number,
   233→    opts?: {
   234→      marginMm?: number;
   235→      gapMm?: number;
   236→      alignV?: GridAlignV;

   238→    },
   239→  ) => void;
   240→  onArrange: () => void;
   241→  onDuplicate: (ids: string[]) => void;
   242→  onRemove: (ids: string[]) => void;
   243→  onSelect: (id: string | null, additive?: boolean) => void;
   244→  onSelectAll: () => void;
   245→  onClearSelection: () => void;
   246→  onClear: () => void;
   247→  onAddImages: () => void;
   248→  /** "Tambah Foto" picker intake — image-only, batch-friendly. */
   249→  onAddPhotoFiles: (files: File[]) => void;
   250→  /** "Ubah Foto" — swap the source image of one slot in place. */
   251→  onChangePhoto: (placementId: string, file: File) => void;

   253→  onCropPhoto: (placementId: string) => void;
   254→  /** "Penuhi Lembar" — enlarge the selection until it covers the entire
   255→   * sheet (full-bleed, no empty paper left). */
   256→  onCoverSheet: (ids: string[]) => void;
   257→}
