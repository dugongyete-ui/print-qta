    86→    86→function NumField({
    87→    87→  label,
    88→    88→  value,
    89→    89→  onCommit,
    90→    90→  min,
    91→    91→  max,
    92→    92→  disabled,
    93→    93→}: {
    94→    94→  label: string;
    95→    95→  value: number;
    96→    96→  onCommit: (v: number) => void;
    97→    97→  min: number;
    98→    98→  max: number;
    99→    99→  disabled?: boolean;
   100→   100→}) {
   101→   101→  const fmt = (v: number) => String(Math.round(v * 10) / 10);
   102→   102→  const commit = (raw: string) => {
   103→   103→    const parsed = Number(raw.replace(",", "."));
   104→   104→    if (!Number.isFinite(parsed)) return;
   105→   105→    onCommit(Math.min(Math.max(parsed, min), max));
   106→   106→  };
   107→   107→  // Uncontrolled input keyed by the external value: dragging / resizing on the
   108→   108→  // canvas remounts it with fresh numbers, while local typing stays untouched.
   109→   109→  return (
   110→   110→    <label className="flex min-w-0 flex-1 flex-col gap-1">
   111→   111→      <span className="font-mono text-[9.5px] uppercase tracking-[0.06em] text-[#6B7280]">
   112→   112→        {label}
   113→   113→      </span>
   114→   114→      <input
   115→   115→        key={`${label}-${fmt(value)}`}
   116→   116→        type="number"
   117→   117→        inputMode="decimal"
   118→   118→        step="0.5"
   119→   119→        defaultValue={fmt(value)}
   120→   120→        disabled={disabled}
   121→   121→        onBlur={(e) => commit(e.target.value)}
   122→   122→        onKeyDown={(e) => {
   123→   123→          if (e.key === "Enter") {
   124→   124→            e.preventDefault();
   125→   125→            (e.target as HTMLInputElement).blur();
   126→   126→          }
   127→   127→        }}
   128→   128→        aria-label={label}
   129→   129→        className="h-8 w-full rounded-md border border-[#E5E7EB] bg-white px-2 font-mono text-[11.5px] tabular-nums text-[#111827] outline-none transition-colors focus:border-[#003399] disabled:opacity-40"
   130→   130→      />
   131→   131→    </label>
   132→   132→  );
   133→   133→}
