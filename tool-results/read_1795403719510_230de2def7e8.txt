    81→    81→function normRotation(rot: number): number {
    82→    82→  const r = Math.round(rot / 90) * 90;
    83→    83→  return ((r % 360) + 360) % 360;
    84→    84→}

    86→    86→function swappedBBox(rotation: number): boolean {
    87→    87→  const r = normRotation(rotation);
    88→    88→  return r === 90 || r === 270;
    89→    89→}

    91→    91→/** Visual bounding box (mm) of a placement, accounting for rotation. */
    92→    92→function placementBBox(p: {
    93→    93→  xMm: number;
    94→    94→  yMm: number;
    95→    95→  wMm: number;
    96→    96→  hMm: number;
    97→    97→  rotation: number;
    98→    98→}): { x: number; y: number; w: number; h: number } {
    99→    99→  const swapped = swappedBBox(p.rotation);
   100→   100→  const w = swapped ? p.hMm : p.wMm;
   101→   101→  const h = swapped ? p.wMm : p.hMm;
   102→   102→  const cx = p.xMm + p.wMm / 2;
   103→   103→  const cy = p.yMm + p.hMm / 2;
   104→   104→  return { x: cx - w / 2, y: cy - h / 2, w, h };
   105→   105→}

   107→   107→/** Keep the visual bbox fully on the paper by clamping the rotation center. */
   108→   108→function clampPlacement(
   109→   109→  p: PhotoPlacement,
   110→   110→  paperWmm: number,
   111→   111→  paperHmm: number,
   112→   112→): PhotoPlacement {
   113→   113→  const bb = placementBBox(p);
   114→   114→  const cx = p.xMm + p.wMm / 2;
   115→   115→  const cy = p.yMm + p.hMm / 2;
   116→   116→  const ncx = Math.min(
   117→   117→    Math.max(cx, bb.w / 2),
   118→   118→    Math.max(bb.w / 2, paperWmm - bb.w / 2),
   119→   119→  );
   120→   120→  const ncy = Math.min(
   121→   121→    Math.max(cy, bb.h / 2),
   122→   122→    Math.max(bb.h / 2, paperHmm - bb.h / 2),
   123→   123→  );
   124→   124→  return { ...p, xMm: ncx - p.wMm / 2, yMm: ncy - p.hMm / 2 };
   125→   125→}

   127→   127→function toBase64(bytes: Uint8Array): string {
   128→   128→  let binary = "";
   129→   129→  const chunk = 0x8000;
   130→   130→  for (let i = 0; i < bytes.length; i += chunk) {
   131→   131→    binary += String.fromCharCode(
   132→   132→      ...Array.from(bytes.subarray(i, i + chunk)),
   133→   133→    );
   134→   134→  }
   135→   135→  return btoa(binary);
   136→   136→}

   142→   142→function loadPdfTask(file: LoadedFile): Promise<PDFDocumentLoadingTask> {
   143→   143→  const cached = pdfTaskCache.get(file.id);
   144→   144→  if (cached) return cached;
   145→   145→  const task = (async () => {
   146→   146→    const pdfjs = await import("pdfjs-dist");
   147→   147→    pdfjs.GlobalWorkerOptions.workerSrc = "/pdf.worker.min.mjs";
   148→   148→    // pdf.js may transfer the buffer to the worker — always hand it a copy.
   149→   149→    const data = file.bytes!.slice();
   150→   150→    return pdfjs.getDocument({ data });
   151→   151→  })();
   152→   152→  pdfTaskCache.set(file.id, task);
   153→   153→  return task;
   154→   154→}

   185→   185→async function inspectImage(file: LoadedFile): Promise<Partial<LoadedFile>> {
   186→   186→  if (!file.bytes) {
   187→   187→    return { status: "error", error: "Data gambar kosong." };
   188→   188→  }
   189→   189→  const url = URL.createObjectURL(
   190→   190→    new Blob([file.bytes.slice().buffer as ArrayBuffer], { type: file.mime }),
   191→   191→  );
   192→   192→  try {
   193→   193→    const img = await new Promise<HTMLImageElement>((resolve, reject) => {
   194→   194→      const el = new Image();
   195→   195→      el.onload = () => resolve(el);
   196→   196→      el.onerror = () => reject(new Error("decode"));
   197→   197→      el.src = url;
   198→   198→    });
   199→   199→    return {
   200→   200→      status: "ready",
   201→   201→      url,
   202→   202→      pageCount: 1,
   203→   203→      docWidthMm: img.naturalWidth * PX96_TO_MM,
   204→   204→      docHeightMm: img.naturalHeight * PX96_TO_MM,
   205→   205→      docLabel: `${img.naturalWidth} × ${img.naturalHeight} px`,
   206→   206→    };
   207→   207→  } catch {
   208→   208→    URL.revokeObjectURL(url);
   209→   209→    return { status: "error", error: "Gambar tidak dapat dibaca atau rusak." };
   210→   210→  }
   211→   211→}

   213→   213→async function renderDocx(file: LoadedFile): Promise<DocxRender> {
   214→   214→  const cached = docxRenderCache.get(file.id);
   215→   215→  if (cached) return cached;
   216→   216→  const task = (async () => {
   217→   217→    const docxPreview = await import("docx-preview");
   218→   218→    const host = document.createElement("div");
   219→   219→    host.style.cssText =
   220→   220→      "position:fixed;left:-100000px;top:0;width:1200px;visibility:hidden;";
   221→   221→    document.body.appendChild(host);
   222→   222→    try {
   223→   223→      await docxPreview.renderAsync(
   224→   224→        new Blob([file.bytes!.slice().buffer as ArrayBuffer], { type: file.mime }),
   225→   225→        host,
   226→   226→        host,
   227→   227→        {
   228→   228→          inWrapper: true,
   229→   229→          breakPages: true,
   230→   230→          experimental: true,
   231→   231→          renderHeaders: true,
   232→   232→          renderFooters: true,
   233→   233→          useBase64URL: true,
   234→   234→          ignoreWidth: false,
   235→   235→          ignoreHeight: false,
   236→   236→        },
   237→   237→      );
   238→   238→      const wrapper = host.querySelector(".docx-wrapper");
   239→   239→      const sections = wrapper
   240→   240→        ? Array.from(wrapper.querySelectorAll("section.docx"))
   241→   241→        : Array.from(host.querySelectorAll("section.docx"));
   242→   242→      if (sections.length === 0) throw new Error("no-sections");
   243→   243→      const first = sections[0] as HTMLElement;
   244→   244→      const wpx = first.offsetWidth || Math.round(210 * (96 / 25.4));
   245→   245→      const hpx = first.offsetHeight || Math.round(297 * (96 / 25.4));
   246→   246→      return {
   247→   247→        sectionsHtml: sections.map((s) => (s as HTMLElement).outerHTML),
   248→   248→        pageWpx: wpx,
   249→   249→        pageHpx: hpx,
   250→   250→        pageWmm: wpx * PX96_TO_MM,
   251→   251→        pageHmm: hpx * PX96_TO_MM,
   252→   252→        pageCount: sections.length,
   253→   253→      } satisfies DocxRender;
   254→   254→    } finally {
   255→   255→      host.remove();
   256→   256→    }
   257→   257→  })();
   258→   258→  docxRenderCache.set(file.id, task);
   259→   259→  return task;
   260→   260→}

   262→   262→async function convertViaServer(file: LoadedFile): Promise<string> {
   263→   263→  const cached = docHtmlCache.get(file.id);
   264→   264→  if (cached) return cached;
   265→   265→  const task = (async () => {
   266→   266→    const form = new FormData();
   267→   267→    form.append(
   268→   268→      "file",
   269→   269→      new Blob([file.bytes!.slice().buffer as ArrayBuffer], { type: file.mime }),
   270→   270→      file.name,
   271→   271→    );
   272→   272→    const res = await fetch("/api/convert-docx", { method: "POST", body: form });
   273→   273→    if (!res.ok) {
   274→   274→      const body = (await res.json().catch(() => null)) as { message?: string } | null;
   275→   275→      throw new Error(body?.message ?? "Konversi dokumen gagal di server.");
   276→   276→    }
   277→   277→    const body = (await res.json()) as { html: string };
   278→   278→    return body.html;
   279→   279→  })();
   280→   280→  docHtmlCache.set(file.id, task);
   281→   281→  return task;
   282→   282→}

   434→   434→  /**
   435→   435→   * Max copies of the current preset that fit on one sheet (best of portrait
   436→   436→   * / landscape orientation), clamped by the engine's placement cap.
   437→   437→   */
   438→   438→  const photoCapacity = useMemo(() => {
   439→   439→    const w = photoPreset.group === "custom" ? photoSettings.customWMm : photoPreset.wMm;
   440→   440→    const h = photoPreset.group === "custom" ? photoSettings.customHMm : photoPreset.hMm;
   441→   441→    const margin = photoSettings.marginMm;
   442→   442→    const gap = photoSettings.gapMm;
   443→   443→    const printW = paperWmm - 2 * margin;
   444→   444→    const printH = paperHmm - 2 * margin;
   445→   445→    const capFor = (pw: number, ph: number) => {
   446→   446→      if (pw > printW + PHOTO_TOL || ph > printH + PHOTO_TOL) return 0;
   447→   447→      return (
   448→   448→        Math.max(1, Math.floor((printW + gap) / (pw + gap))) *
   449→   449→        Math.max(1, Math.floor((printH + gap) / (ph + gap)))
   450→   450→      );
   451→   451→    };
   452→   452→    return Math.min(Math.max(capFor(w, h), capFor(h, w)), MAX_PLACEMENTS);
   453→   453→  }, [photoPreset, photoSettings, paperWmm, paperHmm]);

   762→   762→  /**
   763→   763→   * Auto-grid arrange — the print-shop "ngejajar N foto" flow. Tiles N slots
   764→   764→   * sized to the current preset (or explicit dims) in a balanced grid
   765→   765→   * (6 → 3×2, 8 → 4×2, 9 → 3×3) centered inside the printable area;
   766→   766→   * portrait / landscape is picked by whichever capacity fits N. Slots are
   767→   767→   * filled round-robin with the distinct ready photos on the sheet; a
   768→   768→   * hand-picked photo duplicates itself across the whole grid instead
   769→   769→   * (unless ignoreSelection — used by preset changes & intake re-tiles).
   770→   770→   */
   771→   771→  const arrangePhotoCopies = useCallback(
   772→   772→    (
   773→   773→      count: number,
   774→   774→      options?: {
   775→   775→        dims?: { wMm: number; hMm: number };
   776→   776→        auto?: boolean;
   777→   777→        ignoreSelection?: boolean;
   778→   778→        /** Explicit margin override — used when margin changes and re-tile
   779→   779→         * happens in the same tick (borderless toggle), where the settings
   780→   780→         * ref is still stale. */
   781→   781→        marginMm?: number;
   782→   782→      },
   783→   783→    ) => {
   784→   784→      const list = photoPlacementsRef.current;
   785→   785→      if (list.length === 0) return;
   786→   786→      const ps = photoSettingsRef.current;
   787→   787→      const { wMm: paperW, hMm: paperH } = paperDimsRef.current;
   788→   788→      const preset = getPhotoPreset(ps.presetId);
   789→   789→      const presetW = preset.group === "custom" ? ps.customWMm : preset.wMm;
   790→   790→      const presetH = preset.group === "custom" ? ps.customHMm : preset.hMm;
   791→   791→      const pw = Math.min(options?.dims ? options.dims.wMm : presetW, paperW);
   792→   792→      const ph = Math.min(options?.dims ? options.dims.hMm : presetH, paperH);
   793→   793→      const margin = options?.marginMm ?? ps.marginMm;
   794→   794→      const gap = ps.gapMm;
   795→   795→      const printW = paperW - 2 * margin;
   796→   796→      const printH = paperH - 2 * margin;
   797→   797→
   798→   798→      const fitFor = (w: number, h: number) => {
   799→   799→        if (w > printW + PHOTO_TOL || h > printH + PHOTO_TOL) return null;
   800→   800→        const cols = Math.max(1, Math.floor((printW + gap) / (w + gap)));
   801→   801→        const rows = Math.max(1, Math.floor((printH + gap) / (h + gap)));
   802→   802→        return { cols, rows, cap: cols * rows };
   803→   803→      };
   804→   804→
   805→   805→      const portrait = fitFor(pw, ph);
   806→   806→      const landscape = fitFor(ph, pw);
   807→   807→      const n0 = Math.max(1, Math.round(count));
   808→   808→      // Keep the preset's own orientation whenever it can fit N copies —
   809→   809→      // rotating a pas-foto frame would crop faces. Only fall back to the
   810→   810→      // rotated fit when portrait cannot hold N (or doesn't fit at all).
   811→   811→      let useLandscape: boolean;
   812→   812→      if (portrait && landscape) {
   813→   813→        useLandscape =
   814→   814→          portrait.cap >= n0
   815→   815→            ? false
   816→   816→            : landscape.cap >= n0
   817→   817→              ? true
   818→   818→              : landscape.cap > portrait.cap;
   819→   819→      } else {
   820→   820→        useLandscape = !portrait && !!landscape;
   821→   821→      }
   822→   822→      const fit = useLandscape ? landscape : portrait;
   823→   823→      if (!fit) {
   824→   824→        if (!options?.auto) {
   825→   825→          toast.error("Foto melebihi area cetak lembar.", {
   826→   826→            description:
   827→   827→              "Kecilkan ukuran foto atau margin lembar, lalu susun ulang.",
   828→   828→          });
   829→   829→        }
   830→   830→        return;
   831→   831→      }
   832→   832→      const total = Math.min(n0, fit.cap, MAX_PLACEMENTS);
   833→   833→
   834→   834→      // Balanced grid: fewest wasted slots first, then the most square block,
   835→   835→      // preferring the wider option on ties (6 → 3×2, 9 → 3×3, 8 → 4×2).
   836→   836→      let cols = fit.cols;
   837→   837→      let bestScore = Infinity;
   838→   838→      for (let c = 1; c <= fit.cols; c++) {
   839→   839→        const r = Math.ceil(total / c);
   840→   840→        if (r > fit.rows) continue;
   841→   841→        const score = (c * r - total) * 1000 + Math.abs(c - r);
   842→   842→        if (score < bestScore || (score === bestScore && c > cols)) {
   843→   843→          bestScore = score;
   844→   844→          cols = c;
   845→   845→        }
   846→   846→      }
   847→   847→      const w = useLandscape ? ph : pw;
   848→   848→      const h = useLandscape ? pw : ph;
   849→   849→      const rows = Math.ceil(total / cols);
   850→   850→
   851→   851→      // Slot sources: a hand-picked photo duplicates itself; otherwise the
   852→   852→      // distinct ready images already on the sheet fill slots round-robin.
   853→   853→      let sources: string[] = [];
   854→   854→      const sel = options?.ignoreSelection
   855→   855→        ? null
   856→   856→        : list.find((x) => x.id === photoSelectedIdRef.current);
   857→   857→      if (sel) {
   858→   858→        sources = [sel.itemId];
   859→   859→      } else {
   860→   860→        const seen = new Set<string>();
   861→   861→        for (const p of list) {
   862→   862→          if (seen.has(p.itemId)) continue;
   863→   863→          const it = filesRef.current.find(
   864→   864→            (f) =>
   865→   865→              f.id === p.itemId &&
   866→   866→              f.kind === "image" &&
   867→   867→              f.status === "ready" &&
   868→   868→              !!f.url,
   869→   869→          );
   870→   870→          if (it) {
   871→   871→            seen.add(p.itemId);
   872→   872→            sources.push(p.itemId);
   873→   873→          }
   874→   874→        }
   875→   875→      }
   876→   876→      if (sources.length === 0) sources = [list[0].itemId];
   877→   877→
   878→   878→      const gy = (printH - (rows * h + (rows - 1) * gap)) / 2;
   879→   879→      const out: PhotoPlacement[] = [];
   880→   880→      for (let r = 0; r < rows; r++) {
   881→   881→        const inRow = Math.min(total - r * cols, cols);
   882→   882→        const rowW = inRow * w + (inRow - 1) * gap;
   883→   883→        const gx = (printW - rowW) / 2;
   884→   884→        for (let c = 0; c < inRow; c++) {
   885→   885→          const itemId = sources[(r * cols + c) % sources.length];
   886→   886→          const src = list.find((p) => p.itemId === itemId);
   887→   887→          out.push({
   888→   888→            id: newPhotoId(),
   889→   889→            itemId,
   890→   890→            xMm: margin + gx + c * (w + gap),
