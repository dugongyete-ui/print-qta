    83→    83→function normRotation(rot: number): number {
    84→    84→  const r = Math.round(rot / 90) * 90;
    85→    85→  return ((r % 360) + 360) % 360;
    86→    86→}

    88→    88→function swappedBBox(rotation: number): boolean {
    89→    89→  const r = normRotation(rotation);
    90→    90→  return r === 90 || r === 270;
    91→    91→}

    93→    93→/** Visual bounding box (mm) of a placement, accounting for rotation. */
    94→    94→function placementBBox(p: {
    95→    95→  xMm: number;
    96→    96→  yMm: number;
    97→    97→  wMm: number;
    98→    98→  hMm: number;
    99→    99→  rotation: number;
   100→   100→}): { x: number; y: number; w: number; h: number } {
   101→   101→  const swapped = swappedBBox(p.rotation);
   102→   102→  const w = swapped ? p.hMm : p.wMm;
   103→   103→  const h = swapped ? p.wMm : p.hMm;
   104→   104→  const cx = p.xMm + p.wMm / 2;
   105→   105→  const cy = p.yMm + p.hMm / 2;
   106→   106→  return { x: cx - w / 2, y: cy - h / 2, w, h };
   107→   107→}

   109→   109→/** Keep the visual bbox fully on the paper by clamping the rotation center. */
   110→   110→function clampPlacement(
   111→   111→  p: PhotoPlacement,
   112→   112→  paperWmm: number,
   113→   113→  paperHmm: number,
   114→   114→): PhotoPlacement {
   115→   115→  const bb = placementBBox(p);
   116→   116→  const cx = p.xMm + p.wMm / 2;
   117→   117→  const cy = p.yMm + p.hMm / 2;
   118→   118→  const ncx = Math.min(
   119→   119→    Math.max(cx, bb.w / 2),
   120→   120→    Math.max(bb.w / 2, paperWmm - bb.w / 2),
   121→   121→  );
   122→   122→  const ncy = Math.min(
   123→   123→    Math.max(cy, bb.h / 2),
   124→   124→    Math.max(bb.h / 2, paperHmm - bb.h / 2),
   125→   125→  );
   126→   126→  return { ...p, xMm: ncx - p.wMm / 2, yMm: ncy - p.hMm / 2 };
   127→   127→}

   129→   129→function toBase64(bytes: Uint8Array): string {
   130→   130→  let binary = "";
   131→   131→  const chunk = 0x8000;
   132→   132→  for (let i = 0; i < bytes.length; i += chunk) {
   133→   133→    binary += String.fromCharCode(
   134→   134→      ...Array.from(bytes.subarray(i, i + chunk)),
   135→   135→    );
   136→   136→  }
   137→   137→  return btoa(binary);
   138→   138→}

   144→   144→function loadPdfTask(file: LoadedFile): Promise<PDFDocumentLoadingTask> {
   145→   145→  const cached = pdfTaskCache.get(file.id);
   146→   146→  if (cached) return cached;
   147→   147→  const task = (async () => {
   148→   148→    const pdfjs = await import("pdfjs-dist");
   149→   149→    pdfjs.GlobalWorkerOptions.workerSrc = "/pdf.worker.min.mjs";
   150→   150→    // pdf.js may transfer the buffer to the worker — always hand it a copy.
   151→   151→    const data = file.bytes!.slice();
   152→   152→    return pdfjs.getDocument({ data });
   153→   153→  })();
   154→   154→  pdfTaskCache.set(file.id, task);
   155→   155→  return task;
   156→   156→}

   187→   187→async function inspectImage(file: LoadedFile): Promise<Partial<LoadedFile>> {
   188→   188→  if (!file.bytes) {
   189→   189→    return { status: "error", error: "Data gambar kosong." };
   190→   190→  }
   191→   191→  const url = URL.createObjectURL(
   192→   192→    new Blob([file.bytes.slice().buffer as ArrayBuffer], { type: file.mime }),
   193→   193→  );
   194→   194→  try {
   195→   195→    const img = await new Promise<HTMLImageElement>((resolve, reject) => {
   196→   196→      const el = new Image();
   197→   197→      el.onload = () => resolve(el);
   198→   198→      el.onerror = () => reject(new Error("decode"));
   199→   199→      el.src = url;
   200→   200→    });
   201→   201→    return {
   202→   202→      status: "ready",
   203→   203→      url,
   204→   204→      pageCount: 1,
   205→   205→      docWidthMm: img.naturalWidth * PX96_TO_MM,
   206→   206→      docHeightMm: img.naturalHeight * PX96_TO_MM,
   207→   207→      docLabel: `${img.naturalWidth} × ${img.naturalHeight} px`,
   208→   208→    };
   209→   209→  } catch {
   210→   210→    URL.revokeObjectURL(url);
   211→   211→    return { status: "error", error: "Gambar tidak dapat dibaca atau rusak." };
   212→   212→  }
   213→   213→}

   215→   215→async function renderDocx(file: LoadedFile): Promise<DocxRender> {
   216→   216→  const cached = docxRenderCache.get(file.id);
   217→   217→  if (cached) return cached;
   218→   218→  const task = (async () => {
   219→   219→    const docxPreview = await import("docx-preview");
   220→   220→    const host = document.createElement("div");
   221→   221→    host.style.cssText =
   222→   222→      "position:fixed;left:-100000px;top:0;width:1200px;visibility:hidden;";
   223→   223→    document.body.appendChild(host);
   224→   224→    try {
   225→   225→      await docxPreview.renderAsync(
   226→   226→        new Blob([file.bytes!.slice().buffer as ArrayBuffer], { type: file.mime }),
   227→   227→        host,
   228→   228→        host,
   229→   229→        {
   230→   230→          inWrapper: true,
   231→   231→          breakPages: true,
   232→   232→          experimental: true,
   233→   233→          renderHeaders: true,
   234→   234→          renderFooters: true,
   235→   235→          useBase64URL: true,
   236→   236→          ignoreWidth: false,
   237→   237→          ignoreHeight: false,
   238→   238→        },
   239→   239→      );
   240→   240→      const wrapper = host.querySelector(".docx-wrapper");
   241→   241→      const sections = wrapper
   242→   242→        ? Array.from(wrapper.querySelectorAll("section.docx"))
   243→   243→        : Array.from(host.querySelectorAll("section.docx"));
   244→   244→      if (sections.length === 0) throw new Error("no-sections");
   245→   245→      const first = sections[0] as HTMLElement;
   246→   246→      const wpx = first.offsetWidth || Math.round(210 * (96 / 25.4));
   247→   247→      const hpx = first.offsetHeight || Math.round(297 * (96 / 25.4));
   248→   248→      return {
   249→   249→        sectionsHtml: sections.map((s) => (s as HTMLElement).outerHTML),
   250→   250→        pageWpx: wpx,
   251→   251→        pageHpx: hpx,
   252→   252→        pageWmm: wpx * PX96_TO_MM,
   253→   253→        pageHmm: hpx * PX96_TO_MM,
   254→   254→        pageCount: sections.length,
   255→   255→      } satisfies DocxRender;
   256→   256→    } finally {
   257→   257→      host.remove();
   258→   258→    }
   259→   259→  })();
   260→   260→  docxRenderCache.set(file.id, task);
   261→   261→  return task;
   262→   262→}

   264→   264→async function convertViaServer(file: LoadedFile): Promise<string> {
   265→   265→  const cached = docHtmlCache.get(file.id);
   266→   266→  if (cached) return cached;
   267→   267→  const task = (async () => {
   268→   268→    const form = new FormData();
   269→   269→    form.append(
   270→   270→      "file",
   271→   271→      new Blob([file.bytes!.slice().buffer as ArrayBuffer], { type: file.mime }),
   272→   272→      file.name,
   273→   273→    );
   274→   274→    const res = await fetch("/api/convert-docx", { method: "POST", body: form });
   275→   275→    if (!res.ok) {
   276→   276→      const body = (await res.json().catch(() => null)) as { message?: string } | null;
   277→   277→      throw new Error(body?.message ?? "Konversi dokumen gagal di server.");
   278→   278→    }
   279→   279→    const body = (await res.json()) as { html: string };
   280→   280→    return body.html;
   281→   281→  })();
   282→   282→  docHtmlCache.set(file.id, task);
   283→   283→  return task;
   284→   284→}

   828→   828→   * sized to the current preset (or explicit dims) in a balanced grid
   829→   829→   * (6 → 3×2, 8 → 4×2, 9 → 3×3) aligned inside the printable area
   830→   830→   * (top / center / bottom × left / center / right); portrait / landscape is
   831→   831→   * picked by whichever capacity fits N. Slots are always filled round-robin
   832→   832→   * with every distinct ready photo on the sheet — changing the count never
   833→   833→   * collapses the sheet into copies of a hand-picked photo (use "Isi Penuh"
   834→   834→   * for that).
   835→   835→   */
   836→   836→  const arrangePhotoCopies = useCallback(
   837→   837→    (
   838→   838→      count: number,
   839→   839→      options?: {
   840→   840→        dims?: { wMm: number; hMm: number };
   841→   841→        auto?: boolean;
   842→   842→        /** Explicit margin override — used when margin changes and re-tile
   843→   843→         * happens in the same tick (borderless toggle), where the settings
   844→   844→         * ref is still stale. */
   845→   845→        marginMm?: number;
   846→   846→        gapMm?: number;
   847→   847→        alignV?: GridAlignV;
   848→   848→        alignH?: GridAlignH;
   849→   849→      },
   850→   850→    ) => {
   851→   851→      const list = photoPlacementsRef.current;
   852→   852→      if (list.length === 0) return;
   853→   853→      const ps = photoSettingsRef.current;
   854→   854→      const { wMm: paperW, hMm: paperH } = paperDimsRef.current;
   855→   855→      const preset = getPhotoPreset(ps.presetId);
   856→   856→      const presetW = preset.group === "custom" ? ps.customWMm : preset.wMm;
   857→   857→      const presetH = preset.group === "custom" ? ps.customHMm : preset.hMm;
   858→   858→      const pw = Math.min(options?.dims ? options.dims.wMm : presetW, paperW);
   859→   859→      const ph = Math.min(options?.dims ? options.dims.hMm : presetH, paperH);
   860→   860→      const margin = options?.marginMm ?? ps.marginMm;
   861→   861→      const gap = options?.gapMm ?? ps.gapMm;
   862→   862→      const alignV = options?.alignV ?? ps.alignV;
   863→   863→      const alignH = options?.alignH ?? ps.alignH;
   864→   864→      const printW = paperW - 2 * margin;
   865→   865→      const printH = paperH - 2 * margin;
   866→   866→
   867→   867→      const fitFor = (w: number, h: number) => {
   868→   868→        if (w > printW + PHOTO_TOL || h > printH + PHOTO_TOL) return null;
   869→   869→        const cols = Math.max(1, Math.floor((printW + gap) / (w + gap)));
   870→   870→        const rows = Math.max(1, Math.floor((printH + gap) / (h + gap)));
   871→   871→        return { cols, rows, cap: cols * rows };
   872→   872→      };
   873→   873→
   874→   874→      const portrait = fitFor(pw, ph);
   875→   875→      const landscape = fitFor(ph, pw);
   876→   876→      const n0 = Math.max(1, Math.round(count));
   877→   877→      // Keep the preset's own orientation whenever it can fit N copies —
   878→   878→      // rotating a pas-foto frame would crop faces. Only fall back to the
   879→   879→      // rotated fit when portrait cannot hold N (or doesn't fit at all).
   880→   880→      let useLandscape: boolean;
   881→   881→      if (portrait && landscape) {
   882→   882→        useLandscape =
   883→   883→          portrait.cap >= n0
   884→   884→            ? false
   885→   885→            : landscape.cap >= n0
   886→   886→              ? true
   887→   887→              : landscape.cap > portrait.cap;
   888→   888→      } else {
   889→   889→        useLandscape = !portrait && !!landscape;
   890→   890→      }
   891→   891→      const fit = useLandscape ? landscape : portrait;
   892→   892→      if (!fit) {
   893→   893→        if (!options?.auto) {
