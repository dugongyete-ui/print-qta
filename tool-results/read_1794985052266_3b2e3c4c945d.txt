    80→function normRotation(rot: number): number {
    81→  const r = Math.round(rot / 90) * 90;
    82→  return ((r % 360) + 360) % 360;
    83→}
    84→
    85→function swappedBBox(rotation: number): boolean {
    86→  const r = normRotation(rotation);
    87→  return r === 90 || r === 270;
    88→}

    90→/** Visual bounding box (mm) of a placement, accounting for rotation. */
    91→function placementBBox(p: {
    92→  xMm: number;
    93→  yMm: number;
    94→  wMm: number;
    95→  hMm: number;
    96→  rotation: number;
    97→}): { x: number; y: number; w: number; h: number } {
    98→  const swapped = swappedBBox(p.rotation);
    99→  const w = swapped ? p.hMm : p.wMm;
   100→  const h = swapped ? p.wMm : p.hMm;
   101→  const cx = p.xMm + p.wMm / 2;
   102→  const cy = p.yMm + p.hMm / 2;
   103→  return { x: cx - w / 2, y: cy - h / 2, w, h };
   104→}
   105→
   106→/** Keep the visual bbox fully on the paper by clamping the rotation center. */
   107→function clampPlacement(
   108→  p: PhotoPlacement,
   109→  paperWmm: number,
   110→  paperHmm: number,
   111→): PhotoPlacement {
   112→  const bb = placementBBox(p);
   113→  const cx = p.xMm + p.wMm / 2;
   114→  const cy = p.yMm + p.hMm / 2;
   115→  const ncx = Math.min(
   116→    Math.max(cx, bb.w / 2),
   117→    Math.max(bb.w / 2, paperWmm - bb.w / 2),
   118→  );
   119→  const ncy = Math.min(
   120→    Math.max(cy, bb.h / 2),
   121→    Math.max(bb.h / 2, paperHmm - bb.h / 2),
   122→  );
   123→  return { ...p, xMm: ncx - p.wMm / 2, yMm: ncy - p.hMm / 2 };
   124→}

   126→function toBase64(bytes: Uint8Array): string {
   127→  let binary = "";
   128→  const chunk = 0x8000;
   129→  for (let i = 0; i < bytes.length; i += chunk) {
   130→    binary += String.fromCharCode(
   131→      ...Array.from(bytes.subarray(i, i + chunk)),
   132→    );
   133→  }
   134→  return btoa(binary);
   135→}

   141→function loadPdfTask(file: LoadedFile): Promise<PDFDocumentLoadingTask> {
   142→  const cached = pdfTaskCache.get(file.id);
   143→  if (cached) return cached;
   144→  const task = (async () => {
   145→    const pdfjs = await import("pdfjs-dist");
   146→    pdfjs.GlobalWorkerOptions.workerSrc = "/pdf.worker.min.mjs";
   147→    // pdf.js may transfer the buffer to the worker — always hand it a copy.
   148→    const data = file.bytes!.slice();
   149→    return pdfjs.getDocument({ data });
   150→  })();
   151→  pdfTaskCache.set(file.id, task);
   152→  return task;
   153→}

   155→async function loadPdfDoc(file: LoadedFile): Promise<PDFDocumentProxy> {
   156→  const task = await loadPdfTask(file);
   157→  return task.promise;
   158→}

   160→async function inspectPdf(file: LoadedFile): Promise<Partial<LoadedFile>> {
   161→  try {
   162→    const doc = await loadPdfDoc(file);
   163→    const page1 = await doc.getPage(1);
   164→    const vp = page1.getViewport({ scale: 1 });
   165→    const wMm = vp.width * PT_TO_MM;
   166→    const hMm = vp.height * PT_TO_MM;
   167→    const matched = matchPaper(wMm, hMm);
   168→    return {
   169→      status: "ready",
   170→      pageCount: doc.numPages,
   171→      docWidthMm: wMm,
   172→      docHeightMm: hMm,
   173→      docLabel: matched ? getPaper(matched).shortLabel : sizeLabel(wMm, hMm),
   174→    };
   175→  } catch {
   176→    pdfTaskCache.delete(file.id);
   177→    return {
   178→      status: "error",
   179→      error: "Dokumen PDF rusak, terproteksi, atau tidak dapat dibaca.",
   180→    };
   181→  }
   182→}

   184→async function inspectImage(file: LoadedFile): Promise<Partial<LoadedFile>> {
   185→  if (!file.bytes) {
   186→    return { status: "error", error: "Data gambar kosong." };
   187→  }
   188→  const url = URL.createObjectURL(
   189→    new Blob([file.bytes.slice().buffer as ArrayBuffer], { type: file.mime }),
   190→  );
   191→  try {
   192→    const img = await new Promise<HTMLImageElement>((resolve, reject) => {
   193→      const el = new Image();
   194→      el.onload = () => resolve(el);
   195→      el.onerror = () => reject(new Error("decode"));
   196→      el.src = url;
   197→    });
   198→    return {
   199→      status: "ready",
   200→      url,
   201→      pageCount: 1,
   202→      docWidthMm: img.naturalWidth * PX96_TO_MM,
   203→      docHeightMm: img.naturalHeight * PX96_TO_MM,
   204→      docLabel: `${img.naturalWidth} × ${img.naturalHeight} px`,
   205→    };
   206→  } catch {
   207→    URL.revokeObjectURL(url);
   208→    return { status: "error", error: "Gambar tidak dapat dibaca atau rusak." };
   209→  }
   210→}

   212→async function renderDocx(file: LoadedFile): Promise<DocxRender> {
   213→  const cached = docxRenderCache.get(file.id);
   214→  if (cached) return cached;
   215→  const task = (async () => {
   216→    const docxPreview = await import("docx-preview");
   217→    const host = document.createElement("div");
   218→    host.style.cssText =
   219→      "position:fixed;left:-100000px;top:0;width:1200px;visibility:hidden;";
   220→    document.body.appendChild(host);
   221→    try {
   222→      await docxPreview.renderAsync(
   223→        new Blob([file.bytes!.slice().buffer as ArrayBuffer], { type: file.mime }),
   224→        host,
   225→        host,
   226→        {
   227→          inWrapper: true,
   228→          breakPages: true,
   229→          experimental: true,
   230→          renderHeaders: true,
   231→          renderFooters: true,
   232→          useBase64URL: true,
   233→          ignoreWidth: false,
   234→          ignoreHeight: false,
   235→        },
   236→      );
   237→      const wrapper = host.querySelector(".docx-wrapper");
   238→      const sections = wrapper
   239→        ? Array.from(wrapper.querySelectorAll("section.docx"))
   240→        : Array.from(host.querySelectorAll("section.docx"));
   241→      if (sections.length === 0) throw new Error("no-sections");
   242→      const first = sections[0] as HTMLElement;
   243→      const wpx = first.offsetWidth || Math.round(210 * (96 / 25.4));
   244→      const hpx = first.offsetHeight || Math.round(297 * (96 / 25.4));
   245→      return {
   246→        sectionsHtml: sections.map((s) => (s as HTMLElement).outerHTML),
   247→        pageWpx: wpx,
   248→        pageHpx: hpx,
   249→        pageWmm: wpx * PX96_TO_MM,
   250→        pageHmm: hpx * PX96_TO_MM,
   251→        pageCount: sections.length,
   252→      } satisfies DocxRender;
   253→    } finally {
   254→      host.remove();
   255→    }
   256→  })();
   257→  docxRenderCache.set(file.id, task);
   258→  return task;
   259→}

   261→async function convertViaServer(file: LoadedFile): Promise<string> {
   262→  const cached = docHtmlCache.get(file.id);
   263→  if (cached) return cached;
   264→  const task = (async () => {
   265→    const form = new FormData();
   266→    form.append(
   267→      "file",
   268→      new Blob([file.bytes!.slice().buffer as ArrayBuffer], { type: file.mime }),
   269→      file.name,
   270→    );
   271→    const res = await fetch("/api/convert-docx", { method: "POST", body: form });
   272→    if (!res.ok) {
   273→      const body = (await res.json().catch(() => null)) as { message?: string } | null;
   274→      throw new Error(body?.message ?? "Konversi dokumen gagal di server.");
   275→    }
   276→    const body = (await res.json()) as { html: string };
   277→    return body.html;
   278→  })();
   279→  docHtmlCache.set(file.id, task);
   280→  return task;
   281→}

   283→async function inspectWord(file: LoadedFile): Promise<Partial<LoadedFile>> {
   284→  // 1st choice: client-side docx-preview (true pagination + layout)
   285→  if (file.kind === "docx") {
   286→    try {
   287→      const render = await renderDocx(file);
   288→      const matched = matchPaper(render.pageWmm, render.pageHmm);
   289→      return {
   290→        status: "ready",
   291→        pageCount: render.pageCount,
   292→        docWidthMm: render.pageWmm,
   293→        docHeightMm: render.pageHmm,
   294→        docLabel: matched ? getPaper(matched).shortLabel : sizeLabel(render.pageWmm, render.pageHmm),
   295→      };
   296→    } catch {
   297→      docxRenderCache.delete(file.id);
   298→      // fall through to server-side mammoth
   299→    }
   300→  }
   301→  // 2nd choice: server conversion (mammoth / LibreOffice headless)
   302→  try {
   303→    await convertViaServer(file);
   304→    return {
   305→      status: "ready",
   306→      continuous: true,
   307→      pageCount: 1,
   308→      docWidthMm: 210,
   309→      docHeightMm: 297,
   310→      docLabel: "Halaman Mengalir",
   311→    };
   312→  } catch (err) {
   313→    return {
   314→      status: "error",
   315→      error:
   316→        err instanceof Error
   317→          ? err.message
   318→          : "Konversi dokumen gagal. Simpan ulang sebagai .docx atau PDF.",
   319→    };
   320→  }
   321→}

   329→export function usePrintStation() {
   330→  const [files, setFiles] = useState<LoadedFile[]>([]);
   331→  const [activeId, setActiveId] = useState<string | null>(null);
   332→  const [settings, setSettings] = useState<PrintSettings>(DEFAULT_SETTINGS);
   333→  const [photoMode, setPhotoModeState] = useState(false);
   334→  const [photoPlacements, setPhotoPlacements] = useState<PhotoPlacement[]>([]);
   335→  const [photoSelectedId, setPhotoSelectedId] = useState<string | null>(null);
   336→  const [photoSettings, setPhotoSettings] = useState<PhotoSettings>(DEFAULT_PHOTO_SETTINGS);
   337→  const [page, setPage] = useState(1);
   338→  const [zoom, setZoom] = useState<"fit" | "100">("fit");
   339→  const [showMargins, setShowMargins] = useState(false);
   340→  const [printing, setPrinting] = useState(false);
   341→  const [duplexAssist, setDuplexAssist] = useState<DuplexAssist | null>(null);
   342→  const [spooler, setSpooler] = useState<SpoolerState>({
   343→    state: "checking",
   344→    info: null,
   345→  });
   346→
   347→  const activeFile = useMemo(
   348→    () => files.find((f) => f.id === activeId) ?? null,
   349→    [files, activeId],
   350→  );
   351→  const activeFileRef = useRef(activeFile);
   352→  activeFileRef.current = activeFile;
   353→  const settingsRef = useRef(settings);
   354→  settingsRef.current = settings;
   355→  const filesRef = useRef(files);
   356→  filesRef.current = files;
   357→  const photoModeRef = useRef(photoMode);
   358→  photoModeRef.current = photoMode;
   359→  const photoPlacementsRef = useRef(photoPlacements);
   360→  photoPlacementsRef.current = photoPlacements;
   361→  const photoSelectedIdRef = useRef(photoSelectedId);
   362→  photoSelectedIdRef.current = photoSelectedId;
   363→  const photoSettingsRef = useRef(photoSettings);
   364→  photoSettingsRef.current = photoSettings;
   365→
   366→  // -- derived paper geometry -----------------------------------------------
   367→  const paper = getPaper(settings.paperId);
   368→  const { wMm: paperWmm, hMm: paperHmm } = paperDims(
   369→    settings.paperId,
   370→    settings.orientation,
   371→  );
   372→  const paperDimsRef = useRef({ wMm: paperWmm, hMm: paperHmm });
   373→  paperDimsRef.current = { wMm: paperWmm, hMm: paperHmm };
   374→  // Assigned after the photo actions below — used by addFiles to auto-place images.
   375→  const placeImageRef = useRef<(itemId: string) => void>(() => {});
   376→
   377→  const contentScale = useMemo(() => {
   378→    const f = activeFile;
   379→    if (!f || f.docWidthMm == null || f.docWidthMm <= 0) return 1;
   380→    if (settings.scaleMode !== "fit" || f.continuous) return 1;
   381→    const paperW = paperWmm;
   382→    const paperH = paperHmm;
   383→    return Math.min(paperW / f.docWidthMm, paperH / (f.docHeightMm || 1));
   384→  }, [activeFile, settings.scaleMode, paperWmm, paperHmm]);
   385→
   386→  const paperMismatch = useMemo(() => {
   387→    const f = activeFile;
   388→    if (!f || f.docWidthMm == null || f.continuous) return false;
   389→    return matchPaper(f.docWidthMm, f.docHeightMm ?? 0) !== settings.paperId;
   390→  }, [activeFile, settings.paperId]);
   391→
   392→  const contentOverflows = useMemo(() => {
   393→    const f = activeFile;
   394→    if (!f || f.docWidthMm == null || f.continuous) return false;
   395→    const s = settings.scaleMode === "fit" ? contentScale : 1;
   396→    return (
   397→      f.docWidthMm * s > paperWmm + 0.5 || (f.docHeightMm ?? 0) * s > paperHmm + 0.5
   398→    );
   399→  }, [activeFile, contentScale, settings.scaleMode, paperWmm, paperHmm]);
   400→
   401→  // -- page range -------------------------------------------------------------
   402→  const selectedPages = useMemo(() => {
   403→    const f = activeFile;
   404→    if (!f || f.status !== "ready" || !f.pageCount) return [];
   405→    if (settings.rangeMode === "all") {
   406→      return Array.from({ length: f.pageCount }, (_, i) => i + 1);
   407→    }
   408→    const parsed = parsePageRange(settings.rangeText, f.pageCount);
   409→    return parsed.valid ? parsed.pages : [];
   410→  }, [activeFile, settings.rangeMode, settings.rangeText]);
   411→
   412→  const rangeInvalid =
   413→    settings.rangeMode === "custom" &&
   414→    !parsePageRange(settings.rangeText, activeFile?.pageCount ?? 1).valid;
   415→
   416→  const selectedPagesRef = useRef<number[]>([]);
   417→  selectedPagesRef.current = selectedPages;
   418→
   419→  const photoReady =
   420→    photoPlacements.length > 0 &&
   421→    photoPlacements.every((p) => {
   422→      const it = files.find((f) => f.id === p.itemId);
   423→      return !!it && it.kind === "image" && it.status === "ready" && !!it.url;
   424→    });
   425→  const photoPreset = getPhotoPreset(photoSettings.presetId);
   426→  const photoSelected =
   427→    photoPlacements.find((p) => p.id === photoSelectedId) ?? null;
   428→
   429→  /**
   430→   * Max copies of the current preset that fit on one sheet (best of portrait
   431→   * / landscape orientation), clamped by the engine's placement cap.
   432→   */
   433→  const photoCapacity = useMemo(() => {
   434→    const w = photoPreset.group === "custom" ? photoSettings.customWMm : photoPreset.wMm;
   435→    const h = photoPreset.group === "custom" ? photoSettings.customHMm : photoPreset.hMm;
   436→    const margin = photoSettings.marginMm;
   437→    const gap = photoSettings.gapMm;
   438→    const printW = paperWmm - 2 * margin;
   439→    const printH = paperHmm - 2 * margin;
   440→    const capFor = (pw: number, ph: number) => {
   441→      if (pw > printW + PHOTO_TOL || ph > printH + PHOTO_TOL) return 0;
   442→      return (
   443→        Math.max(1, Math.floor((printW + gap) / (pw + gap))) *
   444→        Math.max(1, Math.floor((printH + gap) / (ph + gap)))
   445→      );
   446→    };
   447→    return Math.min(Math.max(capFor(w, h), capFor(h, w)), MAX_PLACEMENTS);
   448→  }, [photoPreset, photoSettings, paperWmm, paperHmm]);
   449→
   450→  const photoBeyondMargin = photoPlacements.some((p) => {
   451→    const bb = placementBBox(p);
   452→    return (
   453→      bb.x < photoSettings.marginMm - PHOTO_TOL ||
   454→      bb.y < photoSettings.marginMm - PHOTO_TOL ||
   455→      bb.x + bb.w > paperWmm - photoSettings.marginMm + PHOTO_TOL ||
   456→      bb.y + bb.h > paperHmm - photoSettings.marginMm + PHOTO_TOL
   457→    );
   458→  });
   459→
   460→  const canPrint = photoMode
   461→    ? photoReady && !printing
   462→    : !!activeFile &&
   463→      activeFile.status === "ready" &&
   464→      selectedPages.length > 0 &&
   465→      !printing;
   466→
   467→  // -- spooler detection ------------------------------------------------------
   468→  useEffect(() => {
   469→    let alive = true;
   470→    fetch("/api/print")
   471→      .then((r) => r.json())
   472→      .then((body: { spooler: { type: "ipp" | "lp"; printer: string | null } | null }) => {
   473→        if (!alive) return;
   474→        setSpooler(
   475→          body.spooler
   476→            ? { state: "available", info: body.spooler }
   477→            : { state: "unavailable", info: null },
   478→        );
   479→      })
   480→      .catch(() => alive && setSpooler({ state: "unavailable", info: null }));
   481→    return () => {
   482→      alive = false;
   483→    };
   484→  }, []);
   485→
   486→  // -- auto-fit to loaded tray ------------------------------------------------
   487→  const autoFitKey = useRef<string | null>(null);
   488→  useEffect(() => {
   489→    const f = activeFile;
   490→    if (!f || f.status !== "ready" || f.docWidthMm == null || f.continuous) return;
   491→    const key = `${f.id}:${settings.paperId}`;
   492→    if (autoFitKey.current === key) return;
   493→    const matched = matchPaper(f.docWidthMm, f.docHeightMm ?? 0);
   494→    if (matched !== settings.paperId) {
   495→      autoFitKey.current = key;
   496→      setSettings((s) => (s.scaleMode === "fit" ? s : { ...s, scaleMode: "fit" }));
   497→    }
   498→  }, [activeFile, settings.paperId]);
   499→
   500→  // -- file intake ------------------------------------------------------------
   501→  const patchFile = useCallback((id: string, patch: Partial<LoadedFile>) => {
   502→    setFiles((prev) => prev.map((f) => (f.id === id ? { ...f, ...patch } : f)));
   503→  }, []);
   504→
   505→  const addFiles = useCallback(
   506→    (incoming: File[]) => {
   507→      for (const file of incoming) {
   508→        const ext = (file.name.split(".").pop() ?? "").toLowerCase();
   509→        const accepted = ACCEPTED_EXT[ext];
   510→        if (!accepted) {
   511→          toast.error(`Format .${ext || "?"} tidak didukung.`, {
   512→            description: `${file.name} — gunakan PDF, DOC, DOCX, PNG, atau JPG.`,
   513→          });
   514→          continue;
   515→        }
   516→        if (file.size > MAX_FILE_BYTES) {
   517→          toast.error(`Ukuran file melebihi 50 MB.`, {
   518→            description: file.name,
   519→          });
   520→          continue;
   521→        }
   522→        const id = newId();
   523→        const loaded: LoadedFile = {
   524→          id,
   525→          name: file.name,
   526→          ext,
   527→          sizeBytes: file.size,
   528→          kind: accepted.kind,
   529→          mime: accepted.mime,
   530→          bytes: null,
   531→          url: null,
   532→          status: "loading",
   533→          pageCount: null,
   534→          docWidthMm: null,
   535→          docHeightMm: null,
   536→          docLabel: null,
   537→        };
   538→        setFiles((prev) => [...prev, loaded]);
   539→        setActiveId((cur) => cur ?? id);
   540→        setPage(1);
   541→
   542→        file
   543→          .arrayBuffer()
   544→          .then(async (buf) => {
   545→            const bytes = new Uint8Array(buf);
   546→            patchFile(id, { bytes });
   547→            const probe: LoadedFile = { ...loaded, bytes };
   548→            const patch =
   549→              accepted.kind === "pdf"
   550→                ? await inspectPdf(probe)
   551→                : accepted.kind === "image"
   552→                  ? await inspectImage(probe)
   553→                  : await inspectWord(probe);

   786→  /** Fill the whole sheet with copies of the selected photo (pas-foto sheets). */
   787→  const fillPhotoSheet = useCallback(() => {
   788→    const list = photoPlacementsRef.current;

   790→    const selId = photoSelectedIdRef.current;
   791→    const p = list.find((x) => x.id === selId) ?? list[0];
   792→    const ps = photoSettingsRef.current;

   819→          id: newPhotoId(),
   820→          itemId: p.itemId,

   831→    setPhotoSelectedId(null);
   832→    toast.success(`Lembar terisi — ${cols} × ${rows} foto (${out.length} salinan).`);
   833→  }, []);

   835→  /**
   836→   * Row-align exactly N copies of the selected (or first) photo, sized to the
   837→   * current preset — the print-shop "cetak 6 lembar pas foto" flow. The grid
   838→   * packs rows left-to-right, centers the last row and centers the whole block
   839→   * vertically inside the printable area. Orientation follows the preset;
   840→   * when the preset orientation cannot fit N copies but the rotated one can,
   841→   * the layout rotates automatically.
   842→   */
   843→  const arrangePhotoCopies = useCallback((count: number) => {
   844→    const list = photoPlacementsRef.current;
   845→    if (list.length === 0) return;
   846→    const selId = photoSelectedIdRef.current;
   847→    const src = list.find((x) => x.id === selId) ?? list[0];
   848→    const ps = photoSettingsRef.current;
   849→    const { wMm: paperW, hMm: paperH } = paperDimsRef.current;
   850→    const preset = getPhotoPreset(ps.presetId);
   851→    const pw = Math.min(preset.group === "custom" ? ps.customWMm : preset.wMm, paperW);
   852→    const ph = Math.min(preset.group === "custom" ? ps.customHMm : preset.hMm, paperH);
   853→    const margin = ps.marginMm;
   854→    const gap = ps.gapMm;
   855→    const printW = paperW - 2 * margin;
   856→    const printH = paperH - 2 * margin;
   857→
   858→    const gridFor = (w: number, h: number) => {
   859→      if (w > printW + PHOTO_TOL || h > printH + PHOTO_TOL) {
   860→        return { cols: 0, cap: 0 };
   861→      }
   862→      const cols = Math.max(1, Math.floor((printW + gap) / (w + gap)));
   863→      const rows = Math.max(1, Math.floor((printH + gap) / (h + gap)));
   864→      return { cols, cap: cols * rows };
   865→    };
   866→
   867→    const portrait = gridFor(pw, ph);
   868→    const landscape = gridFor(ph, pw);
   869→    const n0 = Math.max(1, Math.round(count));
   870→    const useLandscape =
   871→      portrait.cap >= n0 ? false : landscape.cap >= n0 ? true : landscape.cap > portrait.cap;
   872→    const grid = useLandscape ? landscape : portrait;
   873→    if (grid.cap === 0) {
   874→      toast.error("Foto melebihi area cetak lembar.", {
   875→        description: "Kecilkan ukuran foto atau margin lembar, lalu susun ulang.",
   876→      });
   877→      return;
   878→    }
   879→    const total = Math.min(n0, grid.cap, MAX_PLACEMENTS);
   880→    const w = useLandscape ? ph : pw;
   881→    const h = useLandscape ? pw : ph;
   882→    const rows = Math.ceil(total / grid.cols);
   883→    const gy = (printH - (rows * h + (rows - 1) * gap)) / 2;
   884→    const out: PhotoPlacement[] = [];
   885→    for (let r = 0; r < rows; r++) {
   886→      const inRow = Math.min(total - r * grid.cols, grid.cols);
   887→      const rowW = inRow * w + (inRow - 1) * gap;
   888→      const gx = (printW - rowW) / 2;
   889→      for (let c = 0; c < inRow; c++) {
   890→        out.push({
   891→          id: newPhotoId(),
   892→          itemId: src.itemId,
   893→          xMm: margin + gx + c * (w + gap),
   894→          yMm: margin + gy + r * (h + gap),
   895→          wMm: w,
   896→          hMm: h,
   897→          rotation: 0,
   898→          fit: src.fit,
   899→        });
   900→      }
   901→    }
   902→    setPhotoPlacements(out);
   903→    setPhotoSelectedId(null);
   904→    if (n0 > grid.cap) {
   905→      toast.info(`Lembar penuh — maksimal ${grid.cap} foto ukuran ini`, {
   906→        description:
   907→          "Kurangi jumlah foto, pilih preset lebih kecil, atau ganti kertas lebih besar.",
   908→      });
   909→    } else {
   910→      toast.success(`${total} foto disusun otomatis`, {
   911→        description: `${useLandscape ? "Lanskap" : "Potret"} • ${grid.cols} kolom × ${rows} baris • jarak ${gap} mm`,
   912→      });
   913→    }
   914→  }, []);

  1061→    },
  1062→    [],
  1063→  );
  1064→
  1065→  const printPages = useCallback(async (pages: number[]) => {
  1066→    const req = buildPrintRequest(pages);
  1067→    if (!req) return;
  1068→    await printViaBrowser(req);
  1069→  }, [buildPrintRequest]);
  1070→
  1071→  const spoolPrint = useCallback(async () => {
  1072→    const f = activeFileRef.current;
  1073→    const s = settingsRef.current;
  1074→    const pages = selectedPagesRef.current;
  1075→    if (!f || !f.bytes) return;
  1076→    const dims = paperDims(s.paperId, s.orientation);
  1077→    const res = await fetch("/api/print", {
  1078→      method: "POST",
  1079→      headers: { "Content-Type": "application/json" },
  1080→      body: JSON.stringify({
  1081→        job: {
  1082→          fileName: f.name,
  1083→          mime: f.mime,
  1084→          kind: f.kind,
  1085→          paperId: s.paperId,
  1086→          ippMedia: getPaper(s.paperId).ippMedia,
  1087→          paperWmm: dims.wMm,
  1088→          paperHmm: dims.hMm,
  1089→          orientation: s.orientation,
  1090→          colorMode: s.colorMode,
  1091→          quality: s.quality,
  1092→          copies: s.copies,
  1093→          duplex: s.duplex,
  1094→          scaleMode: s.scaleMode,
  1095→          pages,
  1096→        },
  1097→        data: toBase64(f.bytes),
  1098→      }),
  1099→    });
  1100→    const body = (await res.json().catch(() => null)) as
  1101→      | { ok?: boolean; jobId?: string; message?: string; spooler?: string }
  1102→      | null;
  1103→    if (!res.ok || !body?.ok) {
  1104→      toast.error("Direct Spooler menolak pekerjaan cetak", {
  1105→        description: body?.message ?? "Spooler lokal tidak tersedia.",
  1106→      });
  1107→      return;
  1108→    }
  1109→    toast.success("Pekerjaan cetak terkirim", {
  1110→      description: `${body.spooler ?? "spooler"} • job ${body.jobId ?? "-"}`,
  1111→    });
  1112→  }, []);
  1113→
  1114→  const triggerPrint = useCallback(async () => {
  1115→    // Photo layout mode has its own print pipeline (single rasterized sheet).
  1116→    if (photoModeRef.current) {
  1117→      if (printing || photoPlacementsRef.current.length === 0) return;
  1118→      setPrinting(true);
  1119→      try {
  1120→        await printPhotoLayout();
  1121→      } catch (err) {
  1122→        toast.error("Cetak foto gagal", {
  1123→          description: err instanceof Error ? err.message : "Kesalahan tidak diketahui.",
  1124→        });
  1125→      } finally {
  1126→        setPrinting(false);
  1127→      }
  1128→      return;
  1129→    }
  1130→    const f = activeFileRef.current;
  1131→    const s = settingsRef.current;
  1132→    const pages = selectedPagesRef.current;
  1133→    if (!f || f.status !== "ready" || pages.length === 0 || printing) return;
  1134→
  1135→    setPrinting(true);
  1136→    try {
  1137→      if (s.printMode === "spooler") {
  1138→        await spoolPrint();
  1139→        return;
  1140→      }
  1141→      // Manual duplex: pass 1 = odd pages, pass 2 (user-driven) = even pages reversed
  1142→      if (s.duplex && pages.length > 1) {
  1143→        const odd = pages.filter((p) => p % 2 === 1);
  1144→        await printPages(odd.length > 0 ? odd : pages);
  1145→        const even = pages.filter((p) => p % 2 === 0).reverse();
  1146→        if (even.length > 0) setDuplexAssist({ evenPages: even });
  1147→        return;
  1148→      }
  1149→      await printPages(pages);
  1150→    } catch (err) {
  1151→      toast.error("Cetak gagal", {
  1152→        description: err instanceof Error ? err.message : "Kesalahan tidak diketahui.",
  1153→      });
  1154→    } finally {
  1155→      setPrinting(false);
  1156→    }
  1157→  }, [printing, printPages, spoolPrint, printPhotoLayout]);
  1158→
  1159→  const printEvenPass = useCallback(async () => {
  1160→    if (!duplexAssist) return;
  1161→    setPrinting(true);
  1162→    try {
  1163→      await printPages(duplexAssist.evenPages);
  1164→    } catch (err) {
  1165→      toast.error("Cetak sisi genap gagal", {
  1166→        description: err instanceof Error ? err.message : "Kesalahan tidak diketahui.",
  1167→      });
  1168→    } finally {
  1169→      setPrinting(false);
  1170→      setDuplexAssist(null);
  1171→    }
  1172→  }, [duplexAssist, printPages]);
  1173→
  1174→  const cancelDuplex = useCallback(() => setDuplexAssist(null), []);
  1175→
  1176→  // Ctrl+P / Cmd+P shortcut
  1177→  const triggerRef = useRef(triggerPrint);
  1178→  triggerRef.current = triggerPrint;
  1179→  useEffect(() => {
  1180→    const handler = (e: KeyboardEvent) => {
  1181→      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === "p") {
  1182→        e.preventDefault();
  1183→        void triggerRef.current();
  1184→      }
  1185→    };
  1186→    window.addEventListener("keydown", handler);
  1187→    return () => window.removeEventListener("keydown", handler);
  1188→  }, []);
  1189→
  1190→  const loadSamples = useCallback(async () => {
  1191→    try {
  1192→      const samples: Array<[string, string]> = [
  1193→        ["/samples/sample-letter.pdf", "Surat-Kwitansi-Letter.pdf"],
  1194→        ["/samples/sample-warna.png", "Grafik-Warna.png"],
  1195→        ["/samples/sample-dokumen.docx", "Laporan-Dokumen.docx"],
  1196→      ];
  1197→      const fetched: File[] = [];
  1198→      for (const [url, name] of samples) {
  1199→        const res = await fetch(url);
  1200→        if (!res.ok) continue;
  1201→        const buf = await res.arrayBuffer();
  1202→        const ext = name.split(".").pop()!.toLowerCase();
  1203→        const mime =
  1204→          ext === "pdf"
  1205→            ? "application/pdf"
  1206→            : ext === "png"
  1207→              ? "image/png"
  1208→              : "application/vnd.openxmlformats-officedocument.wordprocessingml.document";
  1209→        fetched.push(new File([buf], name, { type: mime }));
  1210→      }
  1211→      if (fetched.length > 0) addFiles(fetched);
  1212→    } catch {
  1213→      toast.error("Contoh dokumen tidak dapat dimuat.");
  1214→    }
  1215→  }, [addFiles]);
  1216→
  1217→  // clamp preview page
  1218→  useEffect(() => {
  1219→    const total = activeFile?.pageCount ?? 1;
  1220→    if (page > total) setPage(Math.max(1, total));
  1221→  }, [activeFile?.pageCount, page]);
  1222→
  1223→  const summary = useMemo(() => {
  1224→    const paperShort = paper.shortLabel;
  1225→    const color = settings.colorMode === "color" ? "Berwarna" : "Hitam Putih";
  1226→    const copies = settings.copies;
  1227→    if (photoMode) {
  1228→      return `${photoPlacements.length} Foto • ${photoPreset.short} • ${paperShort} • ${color} • ${copies} Salinan`;
  1229→    }
  1230→    const pagesCount = selectedPages.length;
  1231→    const base = `${paperShort} • ${color} • ${pagesCount} Hal × ${copies} Salinan`;
  1232→    return settings.duplex ? `${base} • Bolak-Balik` : base;
  1233→  }, [
  1234→    photoMode,
  1235→    photoPlacements.length,
  1236→    photoPreset.short,
  1237→    paper.shortLabel,
  1238→    settings.colorMode,
  1239→    settings.copies,
  1240→    settings.duplex,
  1241→    selectedPages.length,
  1242→  ]);
  1243→
  1244→  return {

  1265→    // photo mode state & derived
  1266→    photoMode,

  1269→    photoSettings,

  1275→    // actions
