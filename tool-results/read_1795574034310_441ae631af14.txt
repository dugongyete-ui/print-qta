    83→function normRotation(rot: number): number {
    84→  const r = Math.round(rot / 90) * 90;
    85→  return ((r % 360) + 360) % 360;
    86→}
    87→
    88→function swappedBBox(rotation: number): boolean {
    89→  const r = normRotation(rotation);
    90→  return r === 90 || r === 270;
    91→}

    93→/** Visual bounding box (mm) of a placement, accounting for rotation. */
    94→function placementBBox(p: {
    95→  xMm: number;
    96→  yMm: number;
    97→  wMm: number;
    98→  hMm: number;
    99→  rotation: number;
   100→}): { x: number; y: number; w: number; h: number } {
   101→  const swapped = swappedBBox(p.rotation);
   102→  const w = swapped ? p.hMm : p.wMm;
   103→  const h = swapped ? p.wMm : p.hMm;
   104→  const cx = p.xMm + p.wMm / 2;
   105→  const cy = p.yMm + p.hMm / 2;
   106→  return { x: cx - w / 2, y: cy - h / 2, w, h };
   107→}
   108→
   109→/**
   110→ * Manual-edit clamp: allow the photo to extend beyond the paper edges
   111→ * ("full-bleed" — the preview stage and the print rasterizer both crop at
   112→ * the sheet boundary, so what the user sees is exactly what prints). Only
   113→ * the photo's center is kept on the paper so an oversized photo can never
   114→ * be dragged completely off the sheet and lost.
   115→ */
   116→function clampPlacementBleed(
   117→  p: PhotoPlacement,
   118→  paperWmm: number,
   119→  paperHmm: number,
   120→): PhotoPlacement {
   121→  const cx = Math.min(Math.max(p.xMm + p.wMm / 2, 0), paperWmm);
   122→  const cy = Math.min(Math.max(p.yMm + p.hMm / 2, 0), paperHmm);
   123→  return { ...p, xMm: cx - p.wMm / 2, yMm: cy - p.hMm / 2 };
   124→}
   125→
   126→/** Keep the visual bbox fully on the paper by clamping the rotation center. */
   127→function clampPlacement(
   128→  p: PhotoPlacement,
   129→  paperWmm: number,
   130→  paperHmm: number,
   131→): PhotoPlacement {
   132→  const bb = placementBBox(p);
   133→  const cx = p.xMm + p.wMm / 2;
   134→  const cy = p.yMm + p.hMm / 2;
   135→  const ncx = Math.min(
   136→    Math.max(cx, bb.w / 2),
   137→    Math.max(bb.w / 2, paperWmm - bb.w / 2),
   138→  );
   139→  const ncy = Math.min(
   140→    Math.max(cy, bb.h / 2),
   141→    Math.max(bb.h / 2, paperHmm - bb.h / 2),
   142→  );
   143→  return { ...p, xMm: ncx - p.wMm / 2, yMm: ncy - p.hMm / 2 };
   144→}

   146→function toBase64(bytes: Uint8Array): string {
   147→  let binary = "";
   148→  const chunk = 0x8000;
   149→  for (let i = 0; i < bytes.length; i += chunk) {
   150→    binary += String.fromCharCode(
   151→      ...Array.from(bytes.subarray(i, i + chunk)),
   152→    );
   153→  }
   154→  return btoa(binary);
   155→}

   161→function loadPdfTask(file: LoadedFile): Promise<PDFDocumentLoadingTask> {
   162→  const cached = pdfTaskCache.get(file.id);
   163→  if (cached) return cached;
   164→  const task = (async () => {
   165→    const pdfjs = await import("pdfjs-dist");
   166→    pdfjs.GlobalWorkerOptions.workerSrc = "/pdf.worker.min.mjs";
   167→    // pdf.js may transfer the buffer to the worker — always hand it a copy.
   168→    const data = file.bytes!.slice();
   169→    return pdfjs.getDocument({ data });
   170→  })();
   171→  pdfTaskCache.set(file.id, task);
   172→  return task;
   173→}

   175→async function loadPdfDoc(file: LoadedFile): Promise<PDFDocumentProxy> {
   176→  const task = await loadPdfTask(file);
   177→  return task.promise;
   178→}

   180→async function inspectPdf(file: LoadedFile): Promise<Partial<LoadedFile>> {
   181→  try {
   182→    const doc = await loadPdfDoc(file);
   183→    const page1 = await doc.getPage(1);
   184→    const vp = page1.getViewport({ scale: 1 });
   185→    const wMm = vp.width * PT_TO_MM;
   186→    const hMm = vp.height * PT_TO_MM;
   187→    const matched = matchPaper(wMm, hMm);
   188→    return {
   189→      status: "ready",
   190→      pageCount: doc.numPages,
   191→      docWidthMm: wMm,
   192→      docHeightMm: hMm,
   193→      docLabel: matched ? getPaper(matched).shortLabel : sizeLabel(wMm, hMm),
   194→    };
   195→  } catch {
   196→    pdfTaskCache.delete(file.id);
   197→    return {
   198→      status: "error",
   199→      error: "Dokumen PDF rusak, terproteksi, atau tidak dapat dibaca.",
   200→    };
   201→  }
   202→}

   204→async function inspectImage(file: LoadedFile): Promise<Partial<LoadedFile>> {
   205→  if (!file.bytes) {
   206→    return { status: "error", error: "Data gambar kosong." };
   207→  }
   208→  const url = URL.createObjectURL(
   209→    new Blob([file.bytes.slice().buffer as ArrayBuffer], { type: file.mime }),
   210→  );
   211→  try {
   212→    const img = await new Promise<HTMLImageElement>((resolve, reject) => {
   213→      const el = new Image();
   214→      el.onload = () => resolve(el);
   215→      el.onerror = () => reject(new Error("decode"));
   216→      el.src = url;
   217→    });
   218→    return {
   219→      status: "ready",
   220→      url,
   221→      pageCount: 1,
   222→      docWidthMm: img.naturalWidth * PX96_TO_MM,
   223→      docHeightMm: img.naturalHeight * PX96_TO_MM,
   224→      docLabel: `${img.naturalWidth} × ${img.naturalHeight} px`,
   225→    };
   226→  } catch {
   227→    URL.revokeObjectURL(url);
   228→    return { status: "error", error: "Gambar tidak dapat dibaca atau rusak." };
   229→  }
   230→}

   232→async function renderDocx(file: LoadedFile): Promise<DocxRender> {
   233→  const cached = docxRenderCache.get(file.id);
   234→  if (cached) return cached;
   235→  const task = (async () => {
   236→    const docxPreview = await import("docx-preview");
   237→    const host = document.createElement("div");
   238→    host.style.cssText =
   239→      "position:fixed;left:-100000px;top:0;width:1200px;visibility:hidden;";
   240→    document.body.appendChild(host);
   241→    try {
   242→      await docxPreview.renderAsync(
   243→        new Blob([file.bytes!.slice().buffer as ArrayBuffer], { type: file.mime }),
   244→        host,
   245→        host,
   246→        {
   247→          inWrapper: true,
   248→          breakPages: true,
   249→          experimental: true,
   250→          renderHeaders: true,
   251→          renderFooters: true,
   252→          useBase64URL: true,
   253→          ignoreWidth: false,
   254→          ignoreHeight: false,
   255→        },
   256→      );
   257→      const wrapper = host.querySelector(".docx-wrapper");
   258→      const sections = wrapper
   259→        ? Array.from(wrapper.querySelectorAll("section.docx"))
   260→        : Array.from(host.querySelectorAll("section.docx"));
   261→      if (sections.length === 0) throw new Error("no-sections");
   262→      const first = sections[0] as HTMLElement;
   263→      const wpx = first.offsetWidth || Math.round(210 * (96 / 25.4));
   264→      const hpx = first.offsetHeight || Math.round(297 * (96 / 25.4));
   265→      return {
   266→        sectionsHtml: sections.map((s) => (s as HTMLElement).outerHTML),
   267→        pageWpx: wpx,
   268→        pageHpx: hpx,
   269→        pageWmm: wpx * PX96_TO_MM,
   270→        pageHmm: hpx * PX96_TO_MM,
   271→        pageCount: sections.length,
   272→      } satisfies DocxRender;
   273→    } finally {
   274→      host.remove();
   275→    }
   276→  })();
   277→  docxRenderCache.set(file.id, task);
   278→  return task;
   279→}

   281→async function convertViaServer(file: LoadedFile): Promise<string> {
   282→  const cached = docHtmlCache.get(file.id);
   283→  if (cached) return cached;
   284→  const task = (async () => {
   285→    const form = new FormData();
   286→    form.append(
   287→      "file",
   288→      new Blob([file.bytes!.slice().buffer as ArrayBuffer], { type: file.mime }),
   289→      file.name,
   290→    );
   291→    const res = await fetch("/api/convert-docx", { method: "POST", body: form });
   292→    if (!res.ok) {
   293→      const body = (await res.json().catch(() => null)) as { message?: string } | null;
   294→      throw new Error(body?.message ?? "Konversi dokumen gagal di server.");
   295→    }
   296→    const body = (await res.json()) as { html: string };
   297→    return body.html;
   298→  })();
   299→  docHtmlCache.set(file.id, task);
   300→  return task;
   301→}

   303→async function inspectWord(file: LoadedFile): Promise<Partial<LoadedFile>> {
   304→  // 1st choice: client-side docx-preview (true pagination + layout)
   305→  if (file.kind === "docx") {
   306→    try {
   307→      const render = await renderDocx(file);
   308→      const matched = matchPaper(render.pageWmm, render.pageHmm);
   309→      return {
   310→        status: "ready",
   311→        pageCount: render.pageCount,
   312→        docWidthMm: render.pageWmm,
   313→        docHeightMm: render.pageHmm,
   314→        docLabel: matched ? getPaper(matched).shortLabel : sizeLabel(render.pageWmm, render.pageHmm),
   315→      };
   316→    } catch {
   317→      docxRenderCache.delete(file.id);
   318→      // fall through to server-side mammoth
   319→    }
   320→  }
   321→  // 2nd choice: server conversion (mammoth / LibreOffice headless)
   322→  try {
   323→    await convertViaServer(file);
   324→    return {
   325→      status: "ready",
   326→      continuous: true,
   327→      pageCount: 1,
   328→      docWidthMm: 210,
   329→      docHeightMm: 297,
   330→      docLabel: "Halaman Mengalir",
   331→    };
   332→  } catch (err) {
   333→    return {
   334→      status: "error",
   335→      error:
   336→        err instanceof Error
   337→          ? err.message
   338→          : "Konversi dokumen gagal. Simpan ulang sebagai .docx atau PDF.",
   339→    };
   340→  }
   341→}

   349→export function usePrintStation() {
   350→  const [files, setFiles] = useState<LoadedFile[]>([]);
   351→  const [activeId, setActiveId] = useState<string | null>(null);
   352→  const [settings, setSettings] = useState<PrintSettings>(DEFAULT_SETTINGS);
   353→  const [photoMode, setPhotoModeState] = useState(false);
   354→  const [photoPlacements, setPhotoPlacements] = useState<PhotoPlacement[]>([]);
   355→  /** Word-like multi-selection — order matters: last id is the primary. */
   356→  const [photoSelectedIds, setPhotoSelectedIds] = useState<string[]>([]);
   357→  const [photoSettings, setPhotoSettings] = useState<PhotoSettings>(DEFAULT_PHOTO_SETTINGS);
   358→  const [page, setPage] = useState(1);
   359→  const [zoom, setZoom] = useState<"fit" | "100">("fit");
   360→  const [showMargins, setShowMargins] = useState(false);
   361→  const [printing, setPrinting] = useState(false);
   362→  const [duplexAssist, setDuplexAssist] = useState<DuplexAssist | null>(null);
   363→  const [spooler, setSpooler] = useState<SpoolerState>({
   364→    state: "checking",
   365→    info: null,
   366→  });
   367→
   368→  const activeFile = useMemo(
   369→    () => files.find((f) => f.id === activeId) ?? null,
   370→    [files, activeId],
   371→  );
   372→  const activeFileRef = useRef(activeFile);
   373→  activeFileRef.current = activeFile;
   374→  const settingsRef = useRef(settings);
   375→  settingsRef.current = settings;
   376→  const filesRef = useRef(files);
   377→  filesRef.current = files;
   378→  const photoModeRef = useRef(photoMode);
   379→  photoModeRef.current = photoMode;
   380→  const photoPlacementsRef = useRef(photoPlacements);
   381→  photoPlacementsRef.current = photoPlacements;
   382→  const photoSelectedId = photoSelectedIds[photoSelectedIds.length - 1] ?? null;
   383→  const photoSelectedIdsRef = useRef(photoSelectedIds);
   384→  photoSelectedIdsRef.current = photoSelectedIds;
   385→  const photoSelectedIdRef = useRef<string | null>(photoSelectedId);
   386→  photoSelectedIdRef.current = photoSelectedId;
   387→  const photoSettingsRef = useRef(photoSettings);
   388→  photoSettingsRef.current = photoSettings;
   389→
   390→  // -- derived paper geometry -----------------------------------------------
   391→  const paper = getPaper(settings.paperId);
   392→  const { wMm: paperWmm, hMm: paperHmm } = paperDims(
   393→    settings.paperId,
   394→    settings.orientation,
   395→  );
   396→  const paperDimsRef = useRef({ wMm: paperWmm, hMm: paperHmm });
   397→  paperDimsRef.current = { wMm: paperWmm, hMm: paperHmm };
   398→  // Assigned after the photo actions below — used by addFiles to auto-place images.
   399→  const intakeImageRef = useRef<(itemId: string) => void>(() => {});
   400→  // True when the user hand-edited the sheet since the last auto-grid arrange;
   401→  // intake auto-arrange never reshuffles a hand-tuned layout.
   402→  const layoutTouchedRef = useRef(false);
   403→  const intakeTimerRef = useRef<number | null>(null);
   404→
   405→  const contentScale = useMemo(() => {
   406→    const f = activeFile;
   407→    if (!f || f.docWidthMm == null || f.docWidthMm <= 0) return 1;
   408→    if (settings.scaleMode !== "fit" || f.continuous) return 1;
   409→    const paperW = paperWmm;
   410→    const paperH = paperHmm;
   411→    return Math.min(paperW / f.docWidthMm, paperH / (f.docHeightMm || 1));
   412→  }, [activeFile, settings.scaleMode, paperWmm, paperHmm]);
   413→
   414→  const paperMismatch = useMemo(() => {
   415→    const f = activeFile;
   416→    if (!f || f.docWidthMm == null || f.continuous) return false;
   417→    return matchPaper(f.docWidthMm, f.docHeightMm ?? 0) !== settings.paperId;
   418→  }, [activeFile, settings.paperId]);
   419→
   420→  const contentOverflows = useMemo(() => {
   421→    const f = activeFile;
   422→    if (!f || f.docWidthMm == null || f.continuous) return false;
   423→    const s = settings.scaleMode === "fit" ? contentScale : 1;
   424→    return (
   425→      f.docWidthMm * s > paperWmm + 0.5 || (f.docHeightMm ?? 0) * s > paperHmm + 0.5
   426→    );
   427→  }, [activeFile, contentScale, settings.scaleMode, paperWmm, paperHmm]);
   428→
   429→  // -- page range -------------------------------------------------------------
   430→  const selectedPages = useMemo(() => {
   431→    const f = activeFile;
   432→    if (!f || f.status !== "ready" || !f.pageCount) return [];
   433→    if (settings.rangeMode === "all") {
   434→      return Array.from({ length: f.pageCount }, (_, i) => i + 1);
   435→    }
   436→    const parsed = parsePageRange(settings.rangeText, f.pageCount);
   437→    return parsed.valid ? parsed.pages : [];
   438→  }, [activeFile, settings.rangeMode, settings.rangeText]);
   439→
   440→  const rangeInvalid =
   441→    settings.rangeMode === "custom" &&
   442→    !parsePageRange(settings.rangeText, activeFile?.pageCount ?? 1).valid;
   443→
   444→  const selectedPagesRef = useRef<number[]>([]);
   445→  selectedPagesRef.current = selectedPages;
   446→
   447→  const photoReady =
   448→    photoPlacements.length > 0 &&
   449→    photoPlacements.every((p) => {
   450→      const it = files.find((f) => f.id === p.itemId);
   451→      return !!it && it.kind === "image" && it.status === "ready" && !!it.url;
   452→    });
   453→  const photoPreset = getPhotoPreset(photoSettings.presetId);
   454→  const photoSelected =
   455→    photoPlacements.find((p) => p.id === photoSelectedId) ?? null;
   456→
   457→  /**
   458→   * Max copies of the current preset that fit on one sheet (best of portrait
   459→   * / landscape orientation), clamped by the engine's placement cap.
   460→   */
   461→  const photoCapacity = useMemo(() => {
   462→    const w = photoPreset.group === "custom" ? photoSettings.customWMm : photoPreset.wMm;
   463→    const h = photoPreset.group === "custom" ? photoSettings.customHMm : photoPreset.hMm;
   464→    const margin = photoSettings.marginMm;
   465→    const gap = photoSettings.gapMm;
   466→    const printW = paperWmm - 2 * margin;
   467→    const printH = paperHmm - 2 * margin;
   468→    const capFor = (pw: number, ph: number) => {
   469→      if (pw > printW + PHOTO_TOL || ph > printH + PHOTO_TOL) return 0;
   470→      return (
   471→        Math.max(1, Math.floor((printW + gap) / (pw + gap))) *
   472→        Math.max(1, Math.floor((printH + gap) / (ph + gap)))
   473→      );
   474→    };
   475→    return Math.min(Math.max(capFor(w, h), capFor(h, w)), MAX_PLACEMENTS);
   476→  }, [photoPreset, photoSettings, paperWmm, paperHmm]);
   477→
   478→  const photoBeyondMargin = photoPlacements.some((p) => {
   479→    const bb = placementBBox(p);
   480→    return (
   481→      bb.x < photoSettings.marginMm - PHOTO_TOL ||
   482→      bb.y < photoSettings.marginMm - PHOTO_TOL ||
   483→      bb.x + bb.w > paperWmm - photoSettings.marginMm + PHOTO_TOL ||
   484→      bb.y + bb.h > paperHmm - photoSettings.marginMm + PHOTO_TOL
   485→    );
   486→  });
   487→
   488→  const canPrint = photoMode
   489→    ? photoReady && !printing
   490→    : !!activeFile &&
   491→      activeFile.status === "ready" &&
   492→      selectedPages.length > 0 &&
   493→      !printing;
   494→
   495→  // -- spooler detection ------------------------------------------------------
   496→  useEffect(() => {
   497→    let alive = true;
   498→    fetch("/api/print")
   499→      .then((r) => r.json())
   500→      .then((body: { spooler: { type: "ipp" | "lp"; printer: string | null } | null }) => {
   501→        if (!alive) return;
   502→        setSpooler(
   503→          body.spooler
   504→            ? { state: "available", info: body.spooler }
   505→            : { state: "unavailable", info: null },
   506→        );
   507→      })
   508→      .catch(() => alive && setSpooler({ state: "unavailable", info: null }));
   509→    return () => {
   510→      alive = false;
   511→    };
   512→  }, []);
   513→
   514→  // -- auto-fit to loaded tray ------------------------------------------------
   515→  const autoFitKey = useRef<string | null>(null);
   516→  useEffect(() => {
   517→    const f = activeFile;
   518→    if (!f || f.status !== "ready" || f.docWidthMm == null || f.continuous) return;
   519→    const key = `${f.id}:${settings.paperId}`;
   520→    if (autoFitKey.current === key) return;
   521→    const matched = matchPaper(f.docWidthMm, f.docHeightMm ?? 0);
   522→    if (matched !== settings.paperId) {
   523→      autoFitKey.current = key;
   524→      setSettings((s) => (s.scaleMode === "fit" ? s : { ...s, scaleMode: "fit" }));
   525→    }
   526→  }, [activeFile, settings.paperId]);
   527→
   528→  // -- file intake ------------------------------------------------------------
   529→  const patchFile = useCallback((id: string, patch: Partial<LoadedFile>) => {
   530→    setFiles((prev) => prev.map((f) => (f.id === id ? { ...f, ...patch } : f)));
   531→  }, []);
   532→
   533→  const addFiles = useCallback(
   534→    (incoming: File[]) => {
   535→      for (const file of incoming) {
   536→        const ext = (file.name.split(".").pop() ?? "").toLowerCase();
   537→        const accepted = ACCEPTED_EXT[ext];
   538→        if (!accepted) {
   539→          toast.error(`Format .${ext || "?"} tidak didukung.`, {

  1031→   * with every distinct ready photo on the sheet — changing the count never
  1032→   * collapses the sheet into copies of a hand-picked photo (use "Isi Penuh"
  1033→   * for that).
  1034→   */
  1035→  const arrangePhotoCopies = useCallback(
  1036→    (
  1037→      count: number,
  1038→      options?: {
  1039→        dims?: { wMm: number; hMm: number };
  1040→        auto?: boolean;
  1041→        /** Grow the slot count so every distinct uploaded photo keeps a slot
  1042→         * (several customers, one collective sheet). Intake-driven arranges
  1043→         * always set this so a newly added face photo is never dropped. */
  1044→        coverAll?: boolean;
  1045→        /** Explicit margin override — used when margin changes and re-tile
  1046→         * happens in the same tick (borderless toggle), where the settings
  1047→         * ref is still stale. */
  1048→        marginMm?: number;
  1049→        gapMm?: number;
  1050→        alignV?: GridAlignV;
  1051→        alignH?: GridAlignH;
  1052→      },
  1053→    ) => {
  1054→      const list = photoPlacementsRef.current;
  1055→      if (list.length === 0) return;
  1056→      const ps = photoSettingsRef.current;
  1057→      const { wMm: paperW, hMm: paperH } = paperDimsRef.current;
  1058→      const preset = getPhotoPreset(ps.presetId);
  1059→      const presetW = preset.group === "custom" ? ps.customWMm : preset.wMm;
  1060→      const presetH = preset.group === "custom" ? ps.customHMm : preset.hMm;
  1061→      const pw = Math.min(options?.dims ? options.dims.wMm : presetW, paperW);
  1062→      const ph = Math.min(options?.dims ? options.dims.hMm : presetH, paperH);
  1063→      const margin = options?.marginMm ?? ps.marginMm;
  1064→      const gap = options?.gapMm ?? ps.gapMm;
  1065→      const alignV = options?.alignV ?? ps.alignV;
  1066→      const alignH = options?.alignH ?? ps.alignH;
  1067→      const printW = paperW - 2 * margin;
  1068→      const printH = paperH - 2 * margin;
  1069→
  1070→      const fitFor = (w: number, h: number) => {
  1071→        if (w > printW + PHOTO_TOL || h > printH + PHOTO_TOL) return null;
  1072→        const cols = Math.max(1, Math.floor((printW + gap) / (w + gap)));
  1073→        const rows = Math.max(1, Math.floor((printH + gap) / (h + gap)));
  1074→        return { cols, rows, cap: cols * rows };
  1075→      };

  1077→      const portrait = fitFor(pw, ph);
  1078→      const landscape = fitFor(ph, pw);

  1081→      // rotating a pas-foto frame would crop faces. Only fall back to the

  1084→      if (portrait && landscape) {

  1087→            ? false
  1088→            : landscape.cap >= n0
  1089→              ? true
  1090→              : landscape.cap > portrait.cap;

  1100→          });

  1113→          (f) =>
  1114→            f.id === p.itemId &&
  1115→            f.kind === "image" &&
  1116→            f.status === "ready" &&
  1117→            !!f.url,
  1118→        );
  1119→        if (it) {
  1120→          seen.add(p.itemId);
  1121→          sources.push(p.itemId);
  1122→        }
  1123→      }
  1124→      if (sources.length === 0) sources = [list[0].itemId];
  1125→
  1126→      const baseTotal = Math.min(n0, fit.cap, MAX_PLACEMENTS);
  1127→      let total = baseTotal;
  1128→      // Intake-driven arrange ("coverAll"): never drop a customer's photo —
  1129→      // when more distinct photos are on the sheet than slots, the grid grows
  1130→      // to fit all of them (up to sheet capacity).

  1132→      if (options?.coverAll && sources.length > total) {
  1133→        total = Math.min(sources.length, fit.cap, MAX_PLACEMENTS);
  1134→        grown = total > baseTotal;
  1135→      }

  1137→

  1139→      // preferring the wider option on ties (6 → 3×2, 9 → 3×3, 8 → 4×2).

  1144→        if (r > fit.rows) continue;

  1146→        if (score < bestScore || (score === bestScore && c > cols)) {
  1147→          bestScore = score;
  1148→          cols = c;
  1149→        }
  1150→      }

  1152→      const h = useLandscape ? pw : ph;
  1153→      const rows = Math.ceil(total / cols);
  1154→

  1157→        alignV === "top"
  1158→          ? margin
  1159→          : alignV === "bottom"

  1168→            ? margin
  1169→            : alignH === "right"

  1176→            id: newPhotoId(),
  1177→            itemId,
  1178→            xMm: gx + c * (w + gap),
  1179→            yMm: gy + r * (h + gap),

  1181→            hMm: h,
  1182→            rotation: 0,
  1183→            fit: src ? src.fit : "cover",
  1184→            cropX: src?.cropX ?? 0,
  1185→            cropY: src?.cropY ?? 0,
  1186→          });
  1187→        }
  1188→      }

  1190→      setPhotoSelectedIds([]);
  1191→      layoutTouchedRef.current = false;

  1193→        alignV === "top" ? "rata atas" : alignV === "bottom" ? "rata bawah" : "di tengah"
  1194→      } • jarak ${gap} mm`;

  1196→        if (options?.coverAll) {
  1197→          toast.warning(`Lembar penuh — ${overCap} foto tidak muat`, {
  1198→            description: `Hanya ${total} dari ${sources.length} foto bisa disusun ukuran preset ini. Pilih preset lebih kecil atau kertas lebih besar agar semua foto ikut.`,
  1199→          });
  1200→        } else {
  1201→          toast.info(
  1202→            `${total} foto tersusun dari ${sources.length} foto yang diunggah`,
  1203→            {
  1204→              description: `Naikkan jumlah foto ke ${sources.length} agar semua foto ikut tersusun.`,
  1205→            },
  1206→          );
  1207→        }
  1208→        return;
  1209→      }

  1213→        });

  1219→            "Kurangi jumlah foto, pilih preset lebih kecil, atau ganti kertas lebih besar.",

  1227→      }

  1716→    }

  1718→    const s = settingsRef.current;
  1719→    const pages = selectedPagesRef.current;
