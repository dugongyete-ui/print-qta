    93→    93→/** Visual bounding box of a placement (accounts for 90-step rotation). */
    94→    94→function bboxOf(p: PhotoPlacement): { x: number; y: number; w: number; h: number } {
    95→    95→  const rot = ((Math.round(p.rotation / 90) * 90) % 360 + 360) % 360;
    96→    96→  const swapped = rot === 90 || rot === 270;
    97→    97→  const w = swapped ? p.hMm : p.wMm;
    98→    98→  const h = swapped ? p.wMm : p.hMm;
    99→    99→  const cx = p.xMm + p.wMm / 2;
   100→   100→  const cy = p.yMm + p.hMm / 2;
   101→   101→  return { x: cx - w / 2, y: cy - h / 2, w, h };
   102→   102→}

   104→   104→/** Resize with center-based rotation: the opposite anchor stays fixed. */
   105→   105→function resizePlacement(
   106→   106→  orig: PhotoPlacement,
   107→   107→  dir: HandleDir,
   108→   108→  dxMm: number,
   109→   109→  dyMm: number,
   110→   110→  lockAspect: boolean,
   111→   111→): { xMm: number; yMm: number; wMm: number; hMm: number } {
   112→   112→  const theta = (orig.rotation * Math.PI) / 180;
   113→   113→  const c = Math.cos(theta);
   114→   114→  const s = Math.sin(theta);
   115→   115→  // screen delta -> placement-local delta (rotate by -theta)
   116→   116→  const lx = c * dxMm + s * dyMm;
   117→   117→  const ly = -s * dxMm + c * dyMm;
   118→   118→
   119→   119→  let dw = 0;
   120→   120→  let dh = 0;
   121→   121→  if (dir.includes("e")) dw = lx;
   122→   122→  if (dir.includes("w")) dw = -lx;
   123→   123→  if (dir.includes("s")) dh = ly;
   124→   124→  if (dir.includes("n")) dh = -ly;
   125→   125→
   126→   126→  const ratio = orig.hMm / orig.wMm;
   127→   127→  if (lockAspect && dw !== 0 && dh !== 0) {
   128→   128→    if (Math.abs(dw) / orig.wMm > Math.abs(dh) / orig.hMm) dh = dw * ratio;
   129→   129→    else dw = dh / ratio;
   130→   130→  }
   131→   131→  let w2 = Math.max(MIN_PHOTO_MM, orig.wMm + dw);
   132→   132→  let h2 = Math.max(MIN_PHOTO_MM, orig.hMm + dh);
   133→   133→  if (lockAspect && dw !== 0 && dh !== 0) {
   134→   134→    if (Math.abs(dw) / orig.wMm >= Math.abs(dh) / orig.hMm) h2 = w2 * ratio;
   135→   135→    else w2 = h2 / ratio;
   136→   136→  }
   137→   137→
   138→   138→  // anchor = opposite handle point in local (center-relative) coords
   139→   139→  const fx = dir.includes("w") ? orig.wMm / 2 : dir.includes("e") ? -orig.wMm / 2 : 0;
   140→   140→  const fy = dir.includes("n") ? orig.hMm / 2 : dir.includes("s") ? -orig.hMm / 2 : 0;
   141→   141→  const cx0 = orig.xMm + orig.wMm / 2;
   142→   142→  const cy0 = orig.yMm + orig.hMm / 2;
   143→   143→  const fpx = cx0 + (c * fx - s * fy);
   144→   144→  const fpy = cy0 + (s * fx + c * fy);
   145→   145→  const nfx = dir.includes("w") ? w2 / 2 : dir.includes("e") ? -w2 / 2 : 0;
   146→   146→  const nfy = dir.includes("n") ? h2 / 2 : dir.includes("s") ? -h2 / 2 : 0;
   147→   147→  const ncx = fpx - (c * nfx - s * nfy);
   148→   148→  const ncy = fpy - (s * nfx + c * nfy);
   149→   149→  return { xMm: ncx - w2 / 2, yMm: ncy - h2 / 2, wMm: w2, hMm: h2 };
   150→   150→}

   152→   152→function attachWindowListeners(
   153→   153→  onMove: (ev: PointerEvent) => void,
   154→   154→  onUp: (ev?: Event) => void,
   155→   155→): void {
   156→   156→  window.addEventListener("pointermove", onMove);
   157→   157→  window.addEventListener("pointerup", onUp);
   158→   158→  window.addEventListener("pointercancel", onUp);
   159→   159→  window.addEventListener("blur", onUp);
   160→   160→}

   162→   162→function detachWindowListeners(
   163→   163→  onMove: (ev: PointerEvent) => void,
   164→   164→  onUp: (ev?: Event) => void,
   165→   165→): void {
   166→   166→  window.removeEventListener("pointermove", onMove);

   168→   168→  window.removeEventListener("pointercancel", onUp);
   169→   169→  window.removeEventListener("blur", onUp);
   170→   170→}

   172→   172→export function PhotoCanvas(props: PhotoCanvasProps) {
   173→   173→  const {
   174→   174→    paperWmm,
   175→   175→    paperHmm,
   176→   176→    k,
   177→   177→    placements,
   178→   178→    files,
   179→   179→    selectedIds,
   180→   180→    marginMm,
   181→   181→    lockAspect,
   182→   182→    cutLines,
   183→   183→    grayscale,
   184→   184→    adjust,
   185→   185→    multiMode,
   186→   186→    onSelect,
   187→   187→    onSelectIds,
   188→   188→    onSelectAll,
   189→   189→    onUpdateMany,
   190→   190→    onRemove,
   191→   191→    onDuplicate,
   192→   192→    onChangePhoto,
   193→   193→    onCropPhoto,
   194→   194→  } = props;
   195→   195→
   196→   196→  const gestureRef = useRef<Gesture | null>(null);
   197→   197→  const placementsRef = useRef(placements);
   198→   198→  const kRef = useRef(k);
   199→   199→  const lockRef = useRef(lockAspect);
   200→   200→  const paperRef = useRef({ wMm: paperWmm, hMm: paperHmm });
   201→   201→  const selectedIdsRef = useRef(selectedIds);
   202→   202→  const multiModeRef = useRef(multiMode);
   203→   203→  // Sync latest props into refs after commit — gesture/keyboard handlers read
   204→   204→  // them at event time (never during render).
   205→   205→  useEffect(() => {
   206→   206→    placementsRef.current = placements;
   207→   207→    kRef.current = k;
   208→   208→    lockRef.current = lockAspect;
   209→   209→    paperRef.current = { wMm: paperWmm, hMm: paperHmm };
   210→   210→    selectedIdsRef.current = selectedIds;
   211→   211→    multiModeRef.current = multiMode;
   212→   212→  }, [placements, k, lockAspect, paperWmm, paperHmm, selectedIds, multiMode]);
   213→   213→  const [guides, setGuides] = useState<{ v: number | null; h: number | null }>({
   214→   214→    v: null,
   215→   215→    h: null,
   216→   216→  });
   217→   217→  const [marquee, setMarquee] = useState<{
   218→   218→    x: number;
   219→   219→    y: number;
   220→   220→    w: number;
   221→   221→    h: number;
   222→   222→  } | null>(null);
   223→   223→
   224→   224→  const primaryId = selectedIds.length > 0 ? selectedIds[selectedIds.length - 1] : null;
   225→   225→
   226→   226→  const sheetW = Math.round(paperWmm * k);
   227→   227→  const sheetH = Math.round(paperHmm * k);
   228→   228→
   229→   229→  // -- pointer gesture engine ------------------------------------------------
   230→   230→  const endGesture = useCallback(() => {
   231→   231→    gestureRef.current = null;
   232→   232→    setGuides({ v: null, h: null });
   233→   233→    setMarquee(null);
   234→   234→  }, []);
   235→   235→
   236→   236→  /** Group move: every id in `ids` translates by the same mm delta. */
   237→   237→  const beginMoveGesture = useCallback(
   238→   238→    (
   239→   239→      ids: string[],
   240→   240→      grabId: string,
   241→   241→      e: React.PointerEvent,
   242→   242→      tapToggleId: string | null,
   243→   243→    ) => {
   244→   244→      e.stopPropagation();
   245→   245→      e.preventDefault();
   246→   246→      const origins = new Map<string, { x: number; y: number }>();
   247→   247→      for (const p of placementsRef.current) {
   248→   248→        if (ids.includes(p.id)) origins.set(p.id, { x: p.xMm, y: p.yMm });
   249→   249→      }
   250→   250→      gestureRef.current = {
   251→   251→        kind: "move",
   252→   252→        ids,
   253→   253→        grabId,
   254→   254→        px0: e.clientX,
   255→   255→        py0: e.clientY,
   256→   256→        origins,
   257→   257→        tapToggleId,
   258→   258→        movedPx: 0,
   259→   259→      };
   260→   260→      const onMove = (ev: PointerEvent) => {
   261→   261→        const cur = gestureRef.current;
   262→   262→        if (!cur || cur.kind !== "move") return;
   263→   263→        const dxPx = ev.clientX - cur.px0;
   264→   264→        const dyPx = ev.clientY - cur.py0;
   265→   265→        cur.movedPx = Math.max(cur.movedPx, Math.abs(dxPx), Math.abs(dyPx));
   266→   266→        const scale = kRef.current || 1;
   267→   267→        let dx = dxPx / scale;
   268→   268→        let dy = dyPx / scale;
   269→   269→        // smart snapping (Word-like guides) — anchored on the grabbed photo
   270→   270→        const { wMm: paperW, hMm: paperH } = paperRef.current;
   271→   271→        const tolerance = 6 / scale;
   272→   272→        const me = placementsRef.current.find((p) => p.id === cur.grabId);
   273→   273→        const grabOrigin = cur.origins.get(cur.grabId);
   274→   274→        if (me && grabOrigin) {
   275→   275→          const moved = { ...me, xMm: grabOrigin.x + dx, yMm: grabOrigin.y + dy };
   276→   276→          const bb = bboxOf(moved);
   277→   277→          const tx: number[] = [0, marginMm, paperW - marginMm, paperW / 2, paperW];
   278→   278→          const ty: number[] = [0, marginMm, paperH - marginMm, paperH / 2, paperH];
   279→   279→          for (const o of placementsRef.current) {
   280→   280→            if (cur.ids.includes(o.id)) continue;
   281→   281→            const ob = bboxOf(o);
   282→   282→            tx.push(ob.x, ob.x + ob.w, ob.x + ob.w / 2);
   283→   283→            ty.push(ob.y, ob.y + ob.h, ob.y + ob.h / 2);
   284→   284→          }
   285→   285→          let bestX: { d: number; t: number } | null = null;
   286→   286→          let bestY: { d: number; t: number } | null = null;
   287→   287→          for (const t of tx) {
   288→   288→            for (const e2 of [bb.x, bb.x + bb.w, bb.x + bb.w / 2]) {
   289→   289→              const d = t - e2;
   290→   290→              if (Math.abs(d) <= tolerance && (!bestX || Math.abs(d) < Math.abs(bestX.d))) {
   291→   291→                bestX = { d, t };
   292→   292→              }
   293→   293→            }
   294→   294→          }
   295→   295→          for (const t of ty) {
   296→   296→            for (const e2 of [bb.y, bb.y + bb.h, bb.y + bb.h / 2]) {
   297→   297→              const d = t - e2;
   298→   298→              if (Math.abs(d) <= tolerance && (!bestY || Math.abs(d) < Math.abs(bestY.d))) {
   299→   299→                bestY = { d, t };
   300→   300→              }
   301→   301→            }
   302→   302→          }
   303→   303→          if (bestX) dx += bestX.d;
   304→   304→          if (bestY) dy += bestY.d;
   305→   305→          setGuides({ v: bestX ? bestX.t : null, h: bestY ? bestY.t : null });
   306→   306→        }
   307→   307→        onUpdateMany(
   308→   308→          cur.ids
   309→   309→            .filter((id) => cur.origins.has(id))
   310→   310→            .map((id) => {
   311→   311→              const o = cur.origins.get(id)!;
   312→   312→              return { id, patch: { xMm: o.x + dx, yMm: o.y + dy } };
   313→   313→            }),
   314→   314→        );
   315→   315→      };
   316→   316→      const onUp = () => {
   317→   317→        detachWindowListeners(onMove, onUp);
   318→   318→        const cur = gestureRef.current;
   319→   319→        if (
   320→   320→          cur &&
   321→   321→          cur.kind === "move" &&
   322→   322→          cur.tapToggleId &&
   323→   323→          cur.movedPx < 4
   324→   324→        ) {
   325→   325→          // touch "Pilih Banyak": a tap (no drag) toggles the photo in
   326→   326→          onSelect(cur.tapToggleId, true);
   327→   327→        }
   328→   328→        endGesture();
   329→   329→      };
   330→   330→      attachWindowListeners(onMove, onUp);
   331→   331→    },
   332→   332→    [onUpdateMany, onSelect, marginMm, endGesture],
   333→   333→  );
   334→   334→
   335→   335→  const beginMove = useCallback(
   336→   336→    (p: PhotoPlacement, e: React.PointerEvent) => {
   337→   337→      const additive = e.shiftKey || e.ctrlKey || e.metaKey;
   338→   338→      if (additive) {
   339→   339→        // Ctrl/Shift+click toggles into the multi-selection (Word-like);
   340→   340→        // no drag so building a group never accidentally moves photos.
   341→   341→        e.stopPropagation();
   342→   342→        e.preventDefault();
   343→   343→        onSelect(p.id, true);
   344→   344→        return;
   345→   345→      }
   346→   346→      const sel = selectedIdsRef.current;
   347→   347→      if (multiModeRef.current) {
   348→   348→        if (sel.includes(p.id)) {
   349→   349→          beginMoveGesture([...sel], p.id, e, null);
   350→   350→        } else {
   351→   351→          // tap adds it to the selection, drag moves just this photo
   352→   352→          beginMoveGesture([p.id], p.id, e, p.id);
   353→   353→        }
   354→   354→        return;
   355→   355→      }
   356→   356→      if (sel.includes(p.id)) {
   357→   357→        // already in the (possibly multi) selection — drag moves the group
   358→   358→        beginMoveGesture([...sel], p.id, e, null);
   359→   359→      } else {
   360→   360→        onSelect(p.id);
   361→   361→        beginMoveGesture([p.id], p.id, e, null);
   362→   362→      }
   363→   363→    },
   364→   364→    [beginMoveGesture, onSelect],
   365→   365→  );
   366→   366→
   367→   367→  const beginResize = useCallback(
   368→   368→    (p: PhotoPlacement, dir: HandleDir, e: React.PointerEvent) => {
   369→   369→      e.stopPropagation();
   370→   370→      e.preventDefault();
   371→   371→      gestureRef.current = {
   372→   372→        kind: "resize",
   373→   373→        id: p.id,
   374→   374→        dir,
   375→   375→        px0: e.clientX,
   376→   376→        py0: e.clientY,
   377→   377→        orig: p,
   378→   378→      };
   379→   379→      const onMove = (ev: PointerEvent) => {
   380→   380→        const cur = gestureRef.current;
   381→   381→        if (!cur || cur.kind !== "resize") return;
   382→   382→        const scale = kRef.current || 1;
   383→   383→        const dx = (ev.clientX - cur.px0) / scale;
   384→   384→        const dy = (ev.clientY - cur.py0) / scale;
   385→   385→        const r = resizePlacement(cur.orig, cur.dir, dx, dy, lockRef.current);
   386→   386→        onUpdateMany([{ id: cur.id, patch: r }]);
   387→   387→      };
   388→   388→      const onUp = () => {
   389→   389→        detachWindowListeners(onMove, onUp);
   390→   390→        endGesture();
   391→   391→      };
   392→   392→      attachWindowListeners(onMove, onUp);
   393→   393→    },
   394→   394→    [onUpdateMany, endGesture],
   395→   395→  );
   396→   396→
   397→   397→  // -- marquee lasso on the empty sheet --------------------------------------
   398→   398→  const onSheetPointerDown = useCallback(
   399→   399→    (e: React.PointerEvent<HTMLDivElement>) => {
   400→   400→      if (e.target !== e.currentTarget) return;
   401→   401→      if (e.pointerType === "mouse" && e.button !== 0) return;
   402→   402→      const rect = e.currentTarget.getBoundingClientRect();
   403→   403→      const scale = kRef.current || 1;
   404→   404→      const x0 = (e.clientX - rect.left) / scale;
   405→   405→      const y0 = (e.clientY - rect.top) / scale;
   406→   406→      gestureRef.current = {
   407→   407→        kind: "marquee",
   408→   408→        rect,
   409→   409→        x0,
   410→   410→        y0,
   411→   411→        px0: e.clientX,
   412→   412→        py0: e.clientY,
   413→   413→        moved: false,
   414→   414→      };
   415→   415→      const onMove = (ev: PointerEvent) => {
   416→   416→        const cur = gestureRef.current;
   417→   417→        if (!cur || cur.kind !== "marquee") return;
   418→   418→        if (
   419→   419→          Math.abs(ev.clientX - cur.px0) > 4 ||
   420→   420→          Math.abs(ev.clientY - cur.py0) > 4
   421→   421→        ) {
   422→   422→          cur.moved = true;
   423→   423→        }
   424→   424→        const x = (ev.clientX - cur.rect.left) / scale;
   425→   425→        const y = (ev.clientY - cur.rect.top) / scale;
   426→   426→        setMarquee({
   427→   427→          x: Math.min(x, cur.x0),
   428→   428→          y: Math.min(y, cur.y0),
   429→   429→          w: Math.abs(x - cur.x0),
   430→   430→          h: Math.abs(y - cur.y0),
   431→   431→        });
   432→   432→      };
   433→   433→      const onUp = (ev?: Event) => {
   434→   434→        detachWindowListeners(onMove, onUp);
   435→   435→        const cur = gestureRef.current;
   436→   436→        if (cur && cur.kind === "marquee" && cur.moved && ev) {
   437→   437→          const pe = ev as PointerEvent;
   438→   438→          const x1 = (pe.clientX - cur.rect.left) / scale;
   439→   439→          const y1 = (pe.clientY - cur.rect.top) / scale;
   440→   440→          const rx = Math.min(x1, cur.x0);
