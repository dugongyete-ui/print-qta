    84→    84→function NumField({
    85→    85→  label,
    86→    86→  value,
    87→    87→  onCommit,
    88→    88→  min,
    89→    89→  max,
    90→    90→  disabled,
    91→    91→}: {
    92→    92→  label: string;
    93→    93→  value: number;
    94→    94→  onCommit: (v: number) => void;
    95→    95→  min: number;
    96→    96→  max: number;
    97→    97→  disabled?: boolean;
    98→    98→}) {
    99→    99→  const fmt = (v: number) => String(Math.round(v * 10) / 10);
   100→   100→  const commit = (raw: string) => {
   101→   101→    const parsed = Number(raw.replace(",", "."));
   102→   102→    if (!Number.isFinite(parsed)) return;
   103→   103→    onCommit(Math.min(Math.max(parsed, min), max));
   104→   104→  };
   105→   105→  // Uncontrolled input keyed by the external value: dragging / resizing on the
   106→   106→  // canvas remounts it with fresh numbers, while local typing stays untouched.
   107→   107→  return (
   108→   108→    <label className="flex min-w-0 flex-1 flex-col gap-1">
   109→   109→      <span className="font-mono text-[9.5px] uppercase tracking-[0.06em] text-[#6B7280]">
   110→   110→        {label}
   111→   111→      </span>
   112→   112→      <input
   113→   113→        key={`${label}-${fmt(value)}`}
   114→   114→        type="number"
   115→   115→        inputMode="decimal"
   116→   116→        step="0.5"
   117→   117→        defaultValue={fmt(value)}
   118→   118→        disabled={disabled}
   119→   119→        onBlur={(e) => commit(e.target.value)}
   120→   120→        onKeyDown={(e) => {
   121→   121→          if (e.key === "Enter") {
   122→   122→            e.preventDefault();
   123→   123→            (e.target as HTMLInputElement).blur();
   124→   124→          }
   125→   125→        }}
   126→   126→        aria-label={label}
   127→   127→        className="h-8 w-full rounded-md border border-[#E5E7EB] bg-white px-2 font-mono text-[11.5px] tabular-nums text-[#111827] outline-none transition-colors focus:border-[#003399] disabled:opacity-40"
   128→   128→      />
   129→   129→    </label>
   130→   130→  );
   131→   131→}
